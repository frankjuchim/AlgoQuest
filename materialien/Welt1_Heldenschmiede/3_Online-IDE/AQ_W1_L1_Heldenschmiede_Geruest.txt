class Wuerfel {

    // wuerfeln(seiten: Ganzzahl): Ganzzahl
    // Liefert eine ganzzahlige Zufallszahl von 1 bis seiten.
    // Math.random() liefert eine Kommazahl von 0 (inklusive) bis 1 (exklusive).
    public static int wuerfeln(int seiten) {
        return (int) (Math.random() * seiten) + 1;
    }
}

// ======================= Level 1.1 – Der Held ========================

class L11HeldTest {

    @Test
    public void basis_konstruktor_name() {
        Held h = new Held("Aria");
        assertEquals("Aria", h.getName(), "getName() muss den Namen aus dem Konstruktor liefern");
    }

    @Test
    public void basis_konstruktor_startwerte() {
        Held h = new Held("Aria");
        assertEquals(100, h.getLeben(), "Ein neuer Held startet mit 100 Leben");
        assertEquals(100, h.getMaxLeben(), "Ein neuer Held hat maxLeben 100");
        assertEquals(10, h.getStaerke(), "Ein neuer Held hat Staerke 10");
        assertEquals(0, h.getGold(), "Ein neuer Held startet mit 0 Gold");
    }

    @Test
    public void basis_nimmSchaden_normal() {
        Held h = new Held("Aria");
        h.nimmSchaden(30);
        assertEquals(70, h.getLeben(), "Nach nimmSchaden(30) muss der Held 70 Leben haben");
    }

    @Test
    public void basis_istBesiegt_neuerHeld() {
        Held h = new Held("Aria");
        assertFalse(h.istBesiegt(), "Ein neuer Held mit 100 Leben ist nicht besiegt");
    }

    @Test
    public void erweiterung_nimmSchaden_nichtUnterNull() {
        Held h = new Held("Aria");
        h.nimmSchaden(150);
        assertEquals(0, h.getLeben(), "nimmSchaden darf leben nicht unter 0 senken");
    }

    @Test
    public void erweiterung_nimmSchaden_genauAufNull() {
        Held h = new Held("Aria");
        h.nimmSchaden(100);
        assertEquals(0, h.getLeben(), "Nach nimmSchaden(100) muss der Held genau 0 Leben haben");
        assertTrue(h.istBesiegt(), "Ein Held mit 0 Leben ist besiegt");
    }

    @Test
    public void erweiterung_istBesiegt_beiEinemLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(99);
        assertFalse(h.istBesiegt(), "Mit 1 Leben ist der Held noch NICHT besiegt");
    }

    @Test
    public void erweiterung_heilen_normal() {
        Held h = new Held("Aria");
        h.nimmSchaden(40);
        h.heilen(25);
        assertEquals(85, h.getLeben(), "100 - 40 + 25 = 85: heilen muss die Punkte addieren");
    }

    @Test
    public void erweiterung_heilen_nichtUeberMaxLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(10);
        h.heilen(50);
        assertEquals(100, h.getLeben(), "heilen darf leben nicht ueber maxLeben erhoehen");
    }

    @Test
    public void erweiterung_erhalteGold_positiv() {
        Held h = new Held("Aria");
        h.erhalteGold(15);
        assertEquals(15, h.getGold(), "Nach erhalteGold(15) muss der Held 15 Gold haben");
        h.erhalteGold(5);
        assertEquals(20, h.getGold(), "Gold muss sich aufsummieren: 15 + 5 = 20");
    }

    @Test
    public void experte_erhalteGold_nullUndNegativ() {
        Held h = new Held("Aria");
        h.erhalteGold(10);
        h.erhalteGold(0);
        assertEquals(10, h.getGold(), "erhalteGold(0) darf nichts veraendern");
        h.erhalteGold(-7);
        assertEquals(10, h.getGold(), "erhalteGold mit negativem Betrag darf kein Gold abziehen");
    }

    @Test
    public void experte_heilen_genauAufMaxLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(30);
        h.heilen(30);
        assertEquals(100, h.getLeben(), "70 + 30 = 100 - genau maxLeben ist erlaubt");
        h.heilen(0);
        assertEquals(100, h.getLeben(), "heilen(0) darf nichts veraendern");
    }

    @Test
    public void experte_mehrereTreffer() {
        Held h = new Held("Aria");
        h.nimmSchaden(40);
        h.nimmSchaden(40);
        assertEquals(20, h.getLeben(), "Zwei Treffer mit je 40: 100 - 80 = 20");
        assertFalse(h.istBesiegt(), "Mit 20 Leben ist der Held nicht besiegt");
        h.nimmSchaden(40);
        assertEquals(0, h.getLeben(), "Dritter Treffer: leben darf nicht unter 0 fallen");
        assertTrue(h.istBesiegt(), "Nach drei Treffern mit je 40 ist der Held besiegt");
    }

    @Test
    public void experte_zweiHeldenUnabhaengig() {
        Held a = new Held("Aria");
        Held b = new Held("Brann");
        a.nimmSchaden(50);
        a.erhalteGold(30);
        assertEquals(100, b.getLeben(), "Schaden an Aria darf Brann nicht treffen - jedes Objekt hat eigene Attribute (kein static!)");
        assertEquals(0, b.getGold(), "Gold von Aria darf nicht bei Brann landen - jedes Objekt hat eigene Attribute");
        assertEquals("Brann", b.getName(), "Jeder Held behaelt seinen eigenen Namen");
    }
}

    @Test
    public void erweiterung_monsterIstBesiegt() {

    }

    @Test
    public void erweiterung_heldGreiftMonsterAn() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        h.angreifen(m);
        assertEquals(20, m.getLeben(), "Aria (Angriffswert 10) greift an: Goblin hat danach 30 - 10 = 20 Leben");
        assertEquals(100, h.getLeben(), "Der Angreifer selbst darf beim Angreifen keinen Schaden nehmen");
    }

    @Test
    public void erweiterung_monsterGreiftHeldAn() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        m.angreifen(h);
        assertEquals(94, h.getLeben(), "Goblin (Staerke 6) greift an: Aria hat danach 100 - 6 = 94 Leben");
    }

    @Test
    public void experte_kampfBisZumSieg() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        int runde = 0;
        while (!m.istBesiegt() && !h.istBesiegt() && runde < 10) {
            runde++;
            h.angreifen(m);
            if (!m.istBesiegt()) {
                m.angreifen(h);
            }
        }
        assertEquals(3, runde, "Aria braucht genau 3 Runden fuer den Goblin (30 Leben, 10 Schaden pro Treffer)");
        assertTrue(m.istBesiegt(), "Nach dem Kampf muss der Goblin besiegt sein");
        assertEquals(88, h.getLeben(), "Der Goblin trifft nur in Runde 1 und 2: 100 - 2 * 6 = 88");
    }

    public void experte_zweiVariablenEinObjekt() {

    @Test
    public void experte_heilenMitEigenemMaxLeben() {
