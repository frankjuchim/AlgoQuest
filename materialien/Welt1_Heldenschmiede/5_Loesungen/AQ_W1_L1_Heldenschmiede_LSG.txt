class Wuerfel {

    // wuerfeln(seiten: Ganzzahl): Ganzzahl
    // Liefert eine ganzzahlige Zufallszahl von 1 bis seiten.
    // Math.random() liefert eine Kommazahl von 0 (inklusive) bis 1 (exklusive).
    public static int wuerfeln(int seiten) {
        return (int) (Math.random() * seiten) + 1;
    }
}

// ========================= Level 1.1 – 1.4: Held =====================

class Held {

    // Level 1.1
    private String name;
    private int leben;
    private int maxLeben;
    private int staerke;
    private int gold;

    // Level 1.3
    private Gegenstand[] inventar;
    private int anzahl;               // Anzahl der belegten Plaetze, zugleich naechster freier Index

    // Level 1.4
    private Waffe waffe;              // anfangs null = keine Waffe


    // Level 1.1: Standard-Held
    public Held(String name) {
        this.name = name;
        maxLeben = 100;
        leben = 100;
        staerke = 10;
        gold = 0;
        inventar = new Gegenstand[5];
        anzahl = 0;
        waffe = null;
    }

    // Level 1.2: ueberladener Konstruktor (gleicher Name, andere Parameterliste).
    // ACHTUNG: Auch hier das Inventar anlegen - sonst stuerzt aufnehmen ab.
    public Held(String name, int maxLeben, int staerke) {
        this.name = name;
        this.maxLeben = maxLeben;
        leben = maxLeben;
        this.staerke = staerke;
        gold = 0;
        inventar = new Gegenstand[5];
        anzahl = 0;
        waffe = null;
    }


    // ---------- Level 1.1: Getter ----------

    public String getName() {
        return name;
    }

    public int getLeben() {
        return leben;
    }

    public int getMaxLeben() {
        return maxLeben;
    }

    public int getStaerke() {
        return staerke;
    }

    public int getGold() {
        return gold;
    }


    // ---------- Level 1.1: Zustand veraendern ----------

    public void erhalteGold(int betrag) {
        if (betrag > 0) {
            gold = gold + betrag;
        }
    }

    public boolean istBesiegt() {
        return leben <= 0;
    }

    public void nimmSchaden(int schaden) {
        leben = leben - schaden;
        if (leben < 0) {
            leben = 0;
        }
    }

    public void heilen(int punkte) {
        leben = leben + punkte;
        if (leben > maxLeben) {       // maxLeben, NICHT die feste Zahl 100
            leben = maxLeben;
        }
    }


    // ---------- Level 1.2 (erweitert in 1.4): Kampf ----------

    // Level 1.2 zunaechst nur:  return staerke;
    // Level 1.4: Schaden der Waffe dazu - mit null-Abfrage!
    public int angriffswert() {
        if (waffe == null) {
            return staerke;
        }
        return staerke + waffe.getSchaden();
    }

    // Der Held ruft eine Methode des Monsters auf (Botschaft an ein
    // anderes Objekt). Direkter Zugriff auf m.leben ist nicht moeglich,
    // weil leben in Monster private ist.
    public void angreifen(Monster m) {
        m.nimmSchaden(angriffswert());
    }


    // ---------- Level 1.3: Inventar ----------

    public int getAnzahl() {
        return anzahl;
    }

    // Zuerst die Abfragen (null, voll), dann ans Ende einfuegen.
    // inventar.length statt fester 5 - dann bleibt die Methode richtig,
    // wenn das Inventar spaeter groesser wird.
    public boolean aufnehmen(Gegenstand g) {
        if (g == null || anzahl >= inventar.length) {
            return false;
        }
        inventar[anzahl] = g;
        anzahl++;
        return true;
    }

    // Schleife nur bis anzahl - 1: dahinter stehen null-Eintraege.
    public int gesamtWert() {
        int summe = 0;
        for (int i = 0; i < anzahl; i++) {
            summe = summe + inventar[i].getWert();
        }
        return summe;
    }

    // Leer-Abfrage zuerst, dann mit dem ersten Gegenstand starten.
    // Echtes > (nicht >=): Bei Gleichstand bleibt der erste Gegenstand.
    public Gegenstand wertvollsterGegenstand() {
        if (anzahl == 0) {
            return null;
        }
        Gegenstand bester = inventar[0];
        for (int i = 1; i < anzahl; i++) {
            if (inventar[i].getWert() > bester.getWert()) {
                bester = inventar[i];
            }
        }
        return bester;
    }

    // Zeichenketten IMMER mit equals vergleichen (Abitur!).
    // Hier ist name der PARAMETER (gesuchter Name), nicht das Attribut.
    public boolean enthaelt(String name) {
        for (int i = 0; i < anzahl; i++) {
            if (inventar[i].getName().equals(name)) {
                return true;
            }
        }
        return false;
    }


    // ---------- Level 1.4: Ausruestung und Polymorphie ----------

    public Waffe getWaffe() {
        return waffe;
    }

    public void ausruesten(Waffe w) {
        waffe = w;                    // ersetzt eine alte Waffe; null = Waffe ablegen
    }

    // heilen wiederverwenden - dann gilt die maxLeben-Grenze automatisch.
    public void trinken(Trank t) {
        heilen(t.getHeilwert());
    }

    // Polymorphie: inventar[i] hat den statischen Typ Gegenstand. Welche
    // beschreibung() ausgefuehrt wird, entscheidet aber das Objekt zur
    // Laufzeit (dynamisches Binden): Waffe -> "Waffe: ...",
    // Trank -> "Trank: ...", Schluessel -> geerbte Version.
    public void inventarAusgeben() {
        println("Inventar von " + name + " (" + anzahl + " von " + inventar.length + " Plaetzen):");
        for (int i = 0; i < anzahl; i++) {
            println("  " + inventar[i].beschreibung());
        }
    }
}


// ======================== Level 1.2: Monster =========================

class Monster {

    private String name;
    private int leben;
    private int staerke;

    public Monster(String name, int leben, int staerke) {
        this.name = name;
        this.leben = leben;
        this.staerke = staerke;
    }

    public String getName() {
        return name;
    }

    public int getLeben() {
        return leben;
    }

    public int getStaerke() {
        return staerke;
    }

    public void nimmSchaden(int schaden) {
        leben = leben - schaden;
        if (leben < 0) {
            leben = 0;
        }
    }

    public boolean istBesiegt() {
        return leben <= 0;
    }

    public void angreifen(Held h) {
        h.nimmSchaden(staerke);
    }
}


// ======================= Level 1.3: Gegenstand =======================

class Gegenstand {

    private String name;
    private int wert;                 // Wert in Gold

    public Gegenstand(String name, int wert) {
        this.name = name;             // this.name = Attribut, name = Parameter
        this.wert = wert;
    }

    public String getName() {
        return name;
    }

    public int getWert() {
        return wert;
    }

    // z. B. "Fackel (5 Gold)"
    public String beschreibung() {
        return name + " (" + wert + " Gold)";
    }
}


// ================== Level 1.4: Unterklassen von Gegenstand ==========

class Waffe extends Gegenstand {

    private int schaden;              // nur das NEUE Attribut - name und wert erbt Waffe

    public Waffe(String name, int wert, int schaden) {
        super(name, wert);            // muss die ERSTE Anweisung sein
        this.schaden = schaden;
    }

    public int getSchaden() {
        return schaden;
    }

    // ueberschreibt beschreibung() aus Gegenstand.
    // name und wert sind in Gegenstand private -> nicht direkt zugreifbar.
    // super.beschreibung() liefert den Teil "Schwert (50 Gold)".
    public String beschreibung() {
        return "Waffe: " + super.beschreibung() + ", Schaden " + schaden;
    }
}


class Trank extends Gegenstand {

    private int heilwert;

    public Trank(String name, int wert, int heilwert) {
        super(name, wert);
        this.heilwert = heilwert;
    }

    public int getHeilwert() {
        return heilwert;
    }

    // ueberschreibt beschreibung() aus Gegenstand
    public String beschreibung() {
        return "Trank: " + super.beschreibung() + ", heilt " + heilwert;
    }
}


class Schluessel extends Gegenstand {

    private int tuer;                 // Nummer der Tuer, die er oeffnet

    public Schluessel(String name, int wert, int tuer) {
        super(name, wert);
        this.tuer = tuer;
    }

    public int getTuer() {
        return tuer;
    }

    // beschreibung() wird hier absichtlich NICHT ueberschrieben:
    // Ein Schluessel benutzt die geerbte Version aus Gegenstand.
}

class L11HeldTest {

    @Test
    public void basis_konstruktor_name() {
        Held h = new Held("Aria");
        assertEquals("Aria", h.getName(), "getName() muss den Namen aus dem Konstruktor liefern");
    }

    @Test
    public void basis_konstruktor_startwerte() {
        Held h = new Held("Aria");
        assertEquals(100, h.getLeben(), "Ein neuer Held startet mit 100 Leben");
        assertEquals(100, h.getMaxLeben(), "Ein neuer Held hat maxLeben 100");
        assertEquals(10, h.getStaerke(), "Ein neuer Held hat Staerke 10");
        assertEquals(0, h.getGold(), "Ein neuer Held startet mit 0 Gold");
    }

    @Test
    public void basis_nimmSchaden_normal() {
        Held h = new Held("Aria");
        h.nimmSchaden(30);
        assertEquals(70, h.getLeben(), "Nach nimmSchaden(30) muss der Held 70 Leben haben");
    }

    @Test
    public void basis_istBesiegt_neuerHeld() {
        Held h = new Held("Aria");
        assertFalse(h.istBesiegt(), "Ein neuer Held mit 100 Leben ist nicht besiegt");
    }

    @Test
    public void erweiterung_nimmSchaden_nichtUnterNull() {
        Held h = new Held("Aria");
        h.nimmSchaden(150);
        assertEquals(0, h.getLeben(), "nimmSchaden darf leben nicht unter 0 senken");
    }

    @Test
    public void erweiterung_nimmSchaden_genauAufNull() {
        Held h = new Held("Aria");
        h.nimmSchaden(100);
        assertEquals(0, h.getLeben(), "Nach nimmSchaden(100) muss der Held genau 0 Leben haben");
        assertTrue(h.istBesiegt(), "Ein Held mit 0 Leben ist besiegt");
    }

    @Test
    public void erweiterung_istBesiegt_beiEinemLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(99);
        assertFalse(h.istBesiegt(), "Mit 1 Leben ist der Held noch NICHT besiegt");
    }

    @Test
    public void erweiterung_heilen_normal() {
        Held h = new Held("Aria");
        h.nimmSchaden(40);
        h.heilen(25);
        assertEquals(85, h.getLeben(), "100 - 40 + 25 = 85: heilen muss die Punkte addieren");
    }

    @Test
    public void erweiterung_heilen_nichtUeberMaxLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(10);
        h.heilen(50);
        assertEquals(100, h.getLeben(), "heilen darf leben nicht ueber maxLeben erhoehen");
    }

    @Test
    public void erweiterung_erhalteGold_positiv() {
        Held h = new Held("Aria");
        h.erhalteGold(15);
        assertEquals(15, h.getGold(), "Nach erhalteGold(15) muss der Held 15 Gold haben");
        h.erhalteGold(5);
        assertEquals(20, h.getGold(), "Gold muss sich aufsummieren: 15 + 5 = 20");
    }

    @Test
    public void experte_erhalteGold_nullUndNegativ() {
        Held h = new Held("Aria");
        h.erhalteGold(10);
        h.erhalteGold(0);
        assertEquals(10, h.getGold(), "erhalteGold(0) darf nichts veraendern");
        h.erhalteGold(-7);
        assertEquals(10, h.getGold(), "erhalteGold mit negativem Betrag darf kein Gold abziehen");
    }

    @Test
    public void experte_heilen_genauAufMaxLeben() {
        Held h = new Held("Aria");
        h.nimmSchaden(30);
        h.heilen(30);
        assertEquals(100, h.getLeben(), "70 + 30 = 100 - genau maxLeben ist erlaubt");
        h.heilen(0);
        assertEquals(100, h.getLeben(), "heilen(0) darf nichts veraendern");
    }

    @Test
    public void experte_mehrereTreffer() {
        Held h = new Held("Aria");
        h.nimmSchaden(40);
        h.nimmSchaden(40);
        assertEquals(20, h.getLeben(), "Zwei Treffer mit je 40: 100 - 80 = 20");
        assertFalse(h.istBesiegt(), "Mit 20 Leben ist der Held nicht besiegt");
        h.nimmSchaden(40);
        assertEquals(0, h.getLeben(), "Dritter Treffer: leben darf nicht unter 0 fallen");
        assertTrue(h.istBesiegt(), "Nach drei Treffern mit je 40 ist der Held besiegt");
    }

    @Test
    public void experte_zweiHeldenUnabhaengig() {
        Held a = new Held("Aria");
        Held b = new Held("Brann");
        a.nimmSchaden(50);
        a.erhalteGold(30);
        assertEquals(100, b.getLeben(), "Schaden an Aria darf Brann nicht treffen - jedes Objekt hat eigene Attribute (kein static!)");
        assertEquals(0, b.getGold(), "Gold von Aria darf nicht bei Brann landen - jedes Objekt hat eigene Attribute");
        assertEquals("Brann", b.getName(), "Jeder Held behaelt seinen eigenen Namen");
    }
}


// ================== Level 1.2 – Monster und Kampf ===================

class L12MonsterTest {

    @Test
    public void basis_heldKonstruktorMitWerten() {
        Held h = new Held("Brann", 150, 14);
        assertEquals("Brann", h.getName(), "Der zweite Konstruktor muss den Namen setzen");
        assertEquals(150, h.getMaxLeben(), "Brann (maxLeben 150, Staerke 14) muss maxLeben 150 haben");
        assertEquals(150, h.getLeben(), "Der Held startet mit vollem Leben: leben = maxLeben");
        assertEquals(14, h.getStaerke(), "Brann (maxLeben 150, Staerke 14) muss Staerke 14 haben");
        assertEquals(0, h.getGold(), "Auch mit dem zweiten Konstruktor startet der Held mit 0 Gold");
    }

    @Test
    public void basis_monsterKonstruktor() {
        Monster m = new Monster("Goblin", 30, 6);
        assertEquals("Goblin", m.getName(), "getName() muss den Namen des Monsters liefern");
        assertEquals(30, m.getLeben(), "Der Goblin (30 Leben, Staerke 6) muss 30 Leben haben");
        assertEquals(6, m.getStaerke(), "Der Goblin (30 Leben, Staerke 6) muss Staerke 6 haben");
    }

    @Test
    public void basis_monsterNimmSchaden() {
        Monster m = new Monster("Goblin", 30, 6);
        m.nimmSchaden(10);
        assertEquals(20, m.getLeben(), "Nach nimmSchaden(10) muss der Goblin 20 Leben haben");
    }

    @Test
    public void erweiterung_monsterNimmSchaden_nichtUnterNull() {
        Monster m = new Monster("Goblin", 30, 6);
        m.nimmSchaden(45);
        assertEquals(0, m.getLeben(), "nimmSchaden darf leben des Monsters nicht unter 0 senken");
    }

    @Test
    public void erweiterung_monsterIstBesiegt() {
        Monster m = new Monster("Goblin", 30, 6);
        assertFalse(m.istBesiegt(), "Ein neues Monster ist nicht besiegt");
        m.nimmSchaden(30);
        assertTrue(m.istBesiegt(), "Ein Monster mit 0 Leben ist besiegt");
    }

    @Test
    public void erweiterung_angriffswertOhneWaffe() {
        Held a = new Held("Aria");
        Held b = new Held("Brann", 150, 14);
        assertEquals(10, a.angriffswert(), "Ohne Waffe ist der Angriffswert gleich der Staerke (10)");
        assertEquals(14, b.angriffswert(), "Ohne Waffe ist der Angriffswert gleich der Staerke (14)");
    }

    @Test
    public void erweiterung_heldGreiftMonsterAn() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        h.angreifen(m);
        assertEquals(20, m.getLeben(), "Aria (Angriffswert 10) greift an: Goblin hat danach 30 - 10 = 20 Leben");
        assertEquals(100, h.getLeben(), "Der Angreifer selbst darf beim Angreifen keinen Schaden nehmen");
    }

    @Test
    public void erweiterung_monsterGreiftHeldAn() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        m.angreifen(h);
        assertEquals(94, h.getLeben(), "Goblin (Staerke 6) greift an: Aria hat danach 100 - 6 = 94 Leben");
    }

    @Test
    public void experte_kampfBisZumSieg() {
        Held h = new Held("Aria");
        Monster m = new Monster("Goblin", 30, 6);
        int runde = 0;
        while (!m.istBesiegt() && !h.istBesiegt() && runde < 10) {
            runde++;
            h.angreifen(m);
            if (!m.istBesiegt()) {
                m.angreifen(h);
            }
        }
        assertEquals(3, runde, "Aria braucht genau 3 Runden fuer den Goblin (30 Leben, 10 Schaden pro Treffer)");
        assertTrue(m.istBesiegt(), "Nach dem Kampf muss der Goblin besiegt sein");
        assertEquals(88, h.getLeben(), "Der Goblin trifft nur in Runde 1 und 2: 100 - 2 * 6 = 88");
    }

    @Test
    public void experte_zweiVariablenEinObjekt() {
        Held a = new Held("Aria");
        Held b = a;
        b.nimmSchaden(30);
        assertTrue(a == b, "a und b verweisen auf dasselbe Objekt");
        assertEquals(70, a.getLeben(), "b = a kopiert nur die Referenz: Schaden ueber b trifft auch a");
    }

    @Test
    public void experte_heilenMitEigenemMaxLeben() {
        Held h = new Held("Brann", 150, 14);
        h.nimmSchaden(20);
        h.heilen(100);
        assertEquals(150, h.getLeben(), "heilen muss maxLeben benutzen, nicht die feste Zahl 100");
    }

    @Test
    public void experte_zweiMonsterUnabhaengig() {
        Monster m1 = new Monster("Goblin", 30, 6);
        Monster m2 = new Monster("Ork", 60, 11);
        m1.nimmSchaden(30);
        assertEquals(60, m2.getLeben(), "Schaden am Goblin darf den Ork nicht treffen");
        assertFalse(m2.istBesiegt(), "Der Ork ist nicht besiegt, nur weil der Goblin besiegt ist");
        assertEquals("Ork", m2.getName(), "Jedes Monster behaelt seinen eigenen Namen");
    }
}

class L13InventarTest {

    @Test
    public void basis_gegenstandKonstruktor() {
        Gegenstand g = new Gegenstand("Fackel", 5);
        assertEquals("Fackel", g.getName(), "getName() muss den Namen des Gegenstands liefern");
        assertEquals(5, g.getWert(), "getWert() muss den Wert in Gold liefern");
    }

    @Test
    public void basis_gegenstandBeschreibung() {
        Gegenstand g = new Gegenstand("Fackel", 5);
        assertEquals("Fackel (5 Gold)", g.beschreibung(), "beschreibung() muss genau so aussehen: Fackel (5 Gold)");
    }

    @Test
    public void basis_inventarAnfangsLeer() {
        Held h = new Held("Aria");
        assertEquals(0, h.getAnzahl(), "Ein neuer Held hat 0 Gegenstaende im Inventar");
    }

    @Test
    public void basis_aufnehmenEinGegenstand() {
        Held h = new Held("Aria");
        boolean ok = h.aufnehmen(new Gegenstand("Fackel", 5));
        assertTrue(ok, "aufnehmen muss true liefern, wenn noch Platz ist");
        assertEquals(1, h.getAnzahl(), "Nach einem aufnehmen muss getAnzahl() 1 liefern");
    }

    @Test
    public void erweiterung_aufnehmenBisVoll() {
        Held h = new Held("Aria");
        for (int i = 1; i <= 5; i++) {
            boolean ok = h.aufnehmen(new Gegenstand("Stein " + i, 1));
            assertTrue(ok, "Gegenstand Nr. " + i + " muss noch ins Inventar passen (Platz fuer 5)");
        }
        assertEquals(5, h.getAnzahl(), "Nach 5 aufnehmen muss getAnzahl() 5 liefern");
    }

    @Test
    public void erweiterung_aufnehmenWennVoll() {

//  HÄUFIGE SCHÜLERFEHLER (und welcher Test sie aufdeckt)
//
//  Level 1.1 – Held
//  - Parameter verdeckt Attribut: "name = name;" statt "this.name = name;"
//    -> das Attribut bleibt null (basis_konstruktor_name).
//  - Lokale Variable statt Attribut im Konstruktor: "int leben = 100;"
//    -> Attribut bleibt 0 (basis_konstruktor_startwerte).
//  - Attribute static deklariert -> alle Helden teilen sich Leben und
//    Gold (experte_zweiHeldenUnabhaengig). Gute Gelegenheit fuer
//    "Klasse vs. Objekt".
//  - nimmSchaden: Untergrenze vergessen oder "if (leben < 0)" VOR dem
//    Abziehen geprueft (erweiterung_nimmSchaden_nichtUnterNull).
//  - istBesiegt mit "leben == 0" statt "leben <= 0": laeuft hier wegen
//    der Untergrenze zwar, ist aber nicht robust - ansprechen.
//  - heilen: Obergrenze fest mit 100 statt maxLeben -> faellt erst in
//    Level 1.2 auf (experte_heilenMitEigenemMaxLeben).
//  - erhalteGold ohne "betrag > 0"-Abfrage (experte_erhalteGold_nullUndNegativ).
//
//  Level 1.2 – Monster und Kampf
//  - Zweiter Konstruktor setzt leben nicht (leben bleibt 0, Held sofort
//    besiegt) oder vergisst gold (basis_heldKonstruktorMitWerten).
//  - Monster.angreifen greift direkt auf h.leben zu -> Kompilierfehler,
//    weil leben private ist. Richtig: h.nimmSchaden(staerke). Kapselung!
//  - Held.angreifen benutzt staerke statt angriffswert() -> faellt erst
//    in Level 1.4 auf (experte_angreifenMitWaffe).
//  - Referenz-Raetsel: "Held b = a;" wird fuer eine Kopie gehalten
//    (experte_zweiVariablenEinObjekt). Objektkarten zeichnen lassen!
//
//  Level 1.3 – Gegenstand und Inventar
//  - Inventar nur im ersten Konstruktor angelegt -> NullPointerException
//    bei aufnehmen (experte_inventarMitZweitemKonstruktor).
//  - aufnehmen ohne Voll-Abfrage -> Indexfehler beim 6. Gegenstand
//    (erweiterung_aufnehmenWennVoll).
//  - aufnehmen ohne null-Abfrage -> null landet im Inventar, spaeter
//    NullPointerException (erweiterung_aufnehmenNull).
//  - aufnehmen sucht mit Schleife den ersten freien Platz und vergisst
//    danach den Abbruch -> Gegenstand belegt alle freien Plaetze.
//  - Schleifen bis inventar.length statt bis anzahl ->
//    NullPointerException bei null-Eintraegen (erweiterung_gesamtWert,
//    erweiterung_enthaeltLeer, experte_wertvollsterNormal).
//  - wertvollsterGegenstand: ">=" statt ">" -> bei Gleichstand der
//    letzte (experte_wertvollsterGleichstand); Start mit bester = null
//    und danach bester.getWert() -> NullPointerException; Leer-Abfrage
//    vergessen -> Indexfehler bzw. Zugriff auf null (experte_wertvollsterLeer).
//  - enthaelt mit == statt equals: In der Online-IDE gruen, in echtem
//    Java und im Abitur falsch. Der Test experte_enthaeltMitEquals baut
//    den Namen zur Laufzeit zusammen - in echtem Java wuerde == dort
//    scheitern, die Online-IDE vergleicht aber auch mit == inhaltlich.
//    Deshalb im Gespraech bzw. beim Code-Review pruefen!
//  - enthaelt mit contains statt equals -> "Fack" wird gefunden
//    (experte_enthaeltNurExakterName).
//  - enthaelt mit "else return false;" in der Schleife -> nur der erste
//    Gegenstand wird geprueft (erweiterung_enthaelt, Fall "Seil").
//
//  Level 1.4 – Vererbung
//  - super(name, wert) fehlt oder steht nicht an erster Stelle ->
//    Kompilierfehler (Gegenstand hat keinen parameterlosen Konstruktor).
//  - Attribute name und wert in der Unterklasse noch einmal angelegt
//    ("Verdecken") - laeuft, ist aber falsch modelliert; im Klassen-
//    diagramm stehen geerbte Attribute NICHT noch einmal.
//  - beschreibung() in Waffe greift direkt auf name/wert zu ->
//    Kompilierfehler (private). Loesung: super.beschreibung() oder
//    getName()/getWert().
//  - Leerzeichen/Komma im Text falsch -> Test zeigt erwarteten und
//    tatsaechlichen Text; genau vergleichen lassen.
//  - beschreibung() in Schluessel unnoetig ueberschrieben
//    (erweiterung_schluesselBeschreibungGeerbt).
//  - angriffswert ohne null-Abfrage -> NullPointerException fuer Helden
//    ohne Waffe (erweiterung_angriffswertOhneWaffe aus L1.2,
//    experte_waffeAblegen).
//  - ausruesten erhoeht staerke um den Schaden -> Waffenwechsel addiert
//    (experte_waffeWechseln).
//  - trinken: "leben = leben + t.getHeilwert();" ohne Obergrenze statt
//    heilen() wiederzuverwenden (experte_trinkenNichtUeberMaxLeben).
//  - Waffe/Trank gelten nicht als Gegenstand ("passt nicht ins Inventar")
//    -> genau das klaert experte_polymorphieImInventar: Eine Waffe IST
//    ein Gegenstand; welche beschreibung() laeuft, entscheidet das
//    Objekt (dynamisches Binden).
//
//  ABI-BEZUG (Abi-Check)
//  - null-Abfrage: aufnehmen(null), wertvollsterGegenstand() bei leerem
//    Inventar, angriffswert() ohne Waffe. Im Abitur fuehrt ein fehlender
//    Leer-/null-Fall regelmaessig zu BE-Abzug.
//  - equals: Zeichenketten IMMER mit equals vergleichen (enthaelt). Die
//    Online-IDE verzeiht == - das Abitur nicht.
//  - Schleifengrenzen: bei teilweise gefuellter Reihung bis anzahl - 1,
//    nicht bis length - 1.
//  - Signatur genau wie gefordert, return in jedem Pfad
//    (wertvollsterGegenstand liefert in jedem Fall etwas).
//  - Klassendiagramm (EH 2025): Konstruktoren mit "c", Sichtbarkeit +/-,
//    Vererbungspfeil mit hohler Spitze von Waffe/Trank/Schluessel zu
//    Gegenstand, Assoziation Held - Gegenstand als einfache Linie ohne
//    Kardinalitaet.
//  - Objektkarten: Referenz-Raetsel (a und b zeigen auf dieselbe
//    Objektkarte) und Held mit waffe = null bzw. Pfeil auf eine
//    Waffe-Objektkarte - beides typische Abi-Teilaufgaben (AFB I/II).
//  - Papier-Compiler-Kandidaten: aufnehmen und wertvollsterGegenstand
//    erst handschriftlich, dann unveraendert abtippen.

Held aria = new Held("Aria");
Monster goblin = new Monster("Goblin", 30, 6);

println("Ein " + goblin.getName() + " versperrt " + aria.getName() + " den Weg!");

int runde = 0;
// runde < 20 ist eine Notbremse, damit das Programm auch dann endet,
// wenn noch nicht alle Methoden fertig sind.
while (!aria.istBesiegt() && !goblin.istBesiegt() && runde < 20) {
    runde++;
    println("--- Runde " + runde + " ---");
    aria.angreifen(goblin);
    println(aria.getName() + " trifft. " + goblin.getName() + " hat noch " + goblin.getLeben() + " Leben.");
    if (!goblin.istBesiegt()) {
        int wurf = Wuerfel.wuerfeln(6);
        if (wurf == 1) {
            println(goblin.getName() + " wuerfelt eine 1 und verfehlt!");
        } else {
            goblin.angreifen(aria);
            println(goblin.getName() + " trifft. " + aria.getName() + " hat noch " + aria.getLeben() + " Leben.");
        }
    }
}

if (goblin.istBesiegt()) {
    println(aria.getName() + " siegt nach " + runde + " Runden!");
    aria.erhalteGold(15);
    println("Beute: 15 Gold. " + aria.getName() + " hat jetzt " + aria.getGold() + " Gold.");
} else {
    if (aria.istBesiegt()) {
        println(aria.getName() + " wurde besiegt ...");
    } else {
        println("Kein Sieger nach 20 Runden - sind alle Methoden schon fertig?");
    }
}
