// ---------------------------------------------------------------------
//  GERÜST – unveraendert
// ---------------------------------------------------------------------
//  Die Klasse Anzeige hilft beim Ausgeben von Reihungen. Die Tests
//  benutzen sie, um dir in der Meldung die ganze Reihung zu zeigen,
//  z. B. "[erwartet: {4, 3, 2, 1}, tatsaechlich: {1, 2, 3, 4}]".
//  Du darfst sie auch selbst im Hauptprogramm benutzen.
// ---------------------------------------------------------------------

class Anzeige {

    // text(r: Reihung von Ganzzahl): Zeichenkette
    // Liefert die Reihung als Text, z. B. "{4, 8, 15}".
    // Leere Reihung -> "{}", null (noch keine Reihung) -> "null".
    public static String text(int[] r) {
        if (r == null) {
            return "null";
        }
        String s = "{";
        for (int i = 0; i < r.length; i++) {
            if (i > 0) {
                s = s + ", ";
            }
            s = s + r[i];
        }
        return s + "}";
    }
}

// ===================== Level 0.2 – Die Werkstatt ======================

class Werkstatt {

    // ===================== Stufe BASIS =====================

    // istGerade(n: Ganzzahl): Wahrheitswert
    public static boolean istGerade(int n) {
        return n % 2 == 0;
    }

    // restGold(gold: Ganzzahl, preis: Ganzzahl): Ganzzahl
    public static int restGold(int gold, int preis) {
        return gold % preis;
    }

    // kistenBenoetigt(anzahl: Ganzzahl, proKiste: Ganzzahl): Ganzzahl
    // Aufrunden ohne Math: Ist ein Rest da, braucht man eine Kiste mehr.
    // Gleichwertig: return (anzahl + proKiste - 1) / proKiste;
    public static int kistenBenoetigt(int anzahl, int proKiste) {
        int kisten = anzahl / proKiste;
        if (anzahl % proKiste != 0) {
            kisten = kisten + 1;
        }
        return kisten;
    }


    // ================== Stufe ERWEITERUNG ==================

    // schadensstufe(schaden: Ganzzahl): Zeichenkette
    public static String schadensstufe(int schaden) {
        if (schaden <= 0) {
            return "kein";
        } else if (schaden < 10) {
            return "leicht";
        } else if (schaden < 30) {
            return "mittel";
        } else {
            return "schwer";
        }
    }

    // durchschnitt(a: Ganzzahl, b: Ganzzahl, c: Ganzzahl): Kommazahl
    // 3.0 macht die Division zur Kommazahl-Division.
    // Gleichwertig: return (double) (a + b + c) / 3;
    public static double durchschnitt(int a, int b, int c) {
        return (a + b + c) / 3.0;
    }

    // quersumme(n: Ganzzahl): Ganzzahl
    // n wird hier veraendert - das ist erlaubt: n ist eine KOPIE des
    // Arguments, die Variable des Aufrufers bleibt unveraendert
    // (siehe Wert-Raetsel im Hauptprogramm).
    public static int quersumme(int n) {
        int summe = 0;
        while (n > 0) {
            summe = summe + n % 10;
            n = n / 10;
        }
        return summe;
    }


    // ==================== Stufe EXPERTE ====================

    // potenz(basis: Ganzzahl, exponent: Ganzzahl): Ganzzahl
    public static int potenz(int basis, int exponent) {
        int ergebnis = 1;
        for (int i = 0; i < exponent; i++) {
            ergebnis = ergebnis * basis;
        }
        return ergebnis;
    }

    // ggT(a: Ganzzahl, b: Ganzzahl): Ganzzahl
    // Euklidischer Algorithmus (Divisionsvariante).
    public static int ggT(int a, int b) {
        while (b != 0) {
            int rest = a % b;
            a = b;
            b = rest;
        }
        return a;
    }

    // istPrimzahl(n: Ganzzahl): Wahrheitswert
    // Es genuegt, Teiler bis zur Wurzel zu pruefen: t * t <= n.
    public static boolean istPrimzahl(int n) {
        if (n < 2) {
            return false;
        }
        for (int t = 2; t * t <= n; t++) {
            if (n % t == 0) {
                return false;
            }
        }
        return true;
    }
}


// =================== Level 0.3 – Die Vorratskammer ===================

class Vorrat {

    // ===================== Stufe BASIS =====================

    // anzahlLeer(r: Reihung von Ganzzahl): Ganzzahl
    public static int anzahlLeer(int[] r) {
        int anzahl = 0;
        for (int i = 0; i < r.length; i++) {
            if (r[i] == 0) {
                anzahl++;
            }
        }
        return anzahl;
    }

    // letzterWert(r: Reihung von Ganzzahl): Ganzzahl
    public static int letzterWert(int[] r) {
        if (r.length == 0) {
            return -1;
        }
        return r[r.length - 1];
    }

    // enthaelt(r: Reihung von Ganzzahl, x: Ganzzahl): Wahrheitswert
    public static boolean enthaelt(int[] r, int x) {
        for (int wert : r) {
            if (wert == x) {
                return true;
            }
        }
        return false;
    }


    // ================== Stufe ERWEITERUNG ==================

    // umgedreht(r: Reihung von Ganzzahl): Reihung von Ganzzahl
    public static int[] umgedreht(int[] r) {
        int[] neu = new int[r.length];
        for (int i = 0; i < r.length; i++) {
            neu[i] = r[r.length - 1 - i];
        }
        return neu;
    }

    // verdopple(r: Reihung von Ganzzahl): nichts
    // Zuweisung an r[i] - mit for-each ginge das NICHT.
    public static void verdopple(int[] r) {
        for (int i = 0; i < r.length; i++) {
            r[i] = r[i] * 2;
        }
    }

    // anzahlAnstiege(r: Reihung von Ganzzahl): Ganzzahl
    // Start bei i = 1, damit r[i - 1] existiert.
    public static int anzahlAnstiege(int[] r) {
        int anzahl = 0;
        for (int i = 1; i < r.length; i++) {
            if (r[i] > r[i - 1]) {
                anzahl++;
            }
        }
        return anzahl;
    }


    // ==================== Stufe EXPERTE ====================

    // zaehleZeichen(s: Zeichenkette, c: Zeichen): Ganzzahl
    public static int zaehleZeichen(String s, char c) {
        int anzahl = 0;
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == c) {
                anzahl++;
            }
        }
        return anzahl;
    }

    // ohneVokale(s: Zeichenkette): Zeichenkette
    // Nur charAt, length und + (kein replace, kein toLowerCase).
    public static String ohneVokale(String s) {
        String ergebnis = "";
        for (int i = 0; i < s.length(); i++) {
            char z = s.charAt(i);
            boolean vokal = z == 'a' || z == 'e' || z == 'i' || z == 'o' || z == 'u'
                    || z == 'A' || z == 'E' || z == 'I' || z == 'O' || z == 'U';
            if (!vokal) {
                ergebnis = ergebnis + z;
            }
        }
        return ergebnis;
    }

    // beginntMit(s: Zeichenkette, anfang: Zeichenkette): Wahrheitswert
    // Gleichwertig nach der Laengenabfrage:
    //     return s.substring(0, anfang.length()).equals(anfang);
    public static boolean beginntMit(String s, String anfang) {
        if (anfang.length() > s.length()) {
            return false;
        }
        for (int i = 0; i < anfang.length(); i++) {
            if (s.charAt(i) != anfang.charAt(i)) {
                return false;
            }
        }
        return true;
    }
}


// ==================== Level 0.4 – Die Zauberschule ===================

class Zauber {

    // ===================== Stufe BASIS =====================

    // heilwert(stufe: Ganzzahl): Ganzzahl
    public static int heilwert(int stufe) {
        return stufe * 15;
    }

    // istStaerker(a: Ganzzahl, b: Ganzzahl): Wahrheitswert
    public static boolean istStaerker(int a, int b) {
        return a > b;
    }

    // spruchtext(name: Zeichenkette, stufe: Ganzzahl): Zeichenkette
    public static String spruchtext(String name, int stufe) {
        return name + " (Stufe " + stufe + ")";
    }

    // max2(a: Ganzzahl, b: Ganzzahl): Ganzzahl
    public static int max2(int a, int b) {
        if (a > b) {
            return a;
        } else {
            return b;
        }
    }


    // ================== Stufe ERWEITERUNG ==================
    // (Im Geruest steht hier nur die Abi-Signatur - der Methodenkopf ist
    //  Teil der Aufgabe.)

    // manaKosten(stufe: Ganzzahl, verstaerkt: Wahrheitswert): Ganzzahl
    public static int manaKosten(int stufe, boolean verstaerkt) {
        int kosten = 10 * stufe;
        if (verstaerkt) {
            kosten = kosten * 2;
        }
        return kosten;
    }

    // kannZaubern(mana: Ganzzahl, stufe: Ganzzahl, verstaerkt: Wahrheitswert): Wahrheitswert
    public static boolean kannZaubern(int mana, int stufe, boolean verstaerkt) {
        return mana >= manaKosten(stufe, verstaerkt);
    }

    // max3(a: Ganzzahl, b: Ganzzahl, c: Ganzzahl): Ganzzahl
    public static int max3(int a, int b, int c) {
        return max2(max2(a, b), c);
    }


    // ==================== Stufe EXPERTE ====================

    // runden(wert: Kommazahl): Ganzzahl
    // Fuer wert >= 0: erst 0.5 addieren, dann abschneiden.
    public static int runden(double wert) {
        return (int) (wert + 0.5);
    }

    // wuerfelImBereich(von: Ganzzahl, bis: Ganzzahl): Ganzzahl
    // bis - von + 1 moegliche Ergebnisse; Klammern um den Cast beachten!
    public static int wuerfelImBereich(int von, int bis) {
        return (int) (Math.random() * (bis - von + 1)) + von;
    }

    // zauberkette(stufe: Ganzzahl): Zeichenkette
    public static String zauberkette(int stufe) {
        String kette = "";
        for (int i = 1; i <= stufe; i++) {
            if (i > 1) {
                kette = kette + "-";
            }
            kette = kette + i;
        }
        return kette;
    }
}


// ================ Level 0.5 – Die Zauberschule, Teil II ==============

class ZauberII {

    // ===================== Stufe BASIS =====================

    // fuelle(r: Reihung von Ganzzahl, wert: Ganzzahl): nichts
    public static void fuelle(int[] r, int wert) {
        for (int i = 0; i < r.length; i++) {
            r[i] = wert;
        }
    }

    // erhoeheAlle(r: Reihung von Ganzzahl, bonus: Ganzzahl): nichts
    public static void erhoeheAlle(int[] r, int bonus) {
        for (int i = 0; i < r.length; i++) {
            r[i] = r[i] + bonus;
        }
    }

    // erhoehteKopie(r: Reihung von Ganzzahl, bonus: Ganzzahl): Reihung von Ganzzahl
    public static int[] erhoehteKopie(int[] r, int bonus) {
        int[] neu = new int[r.length];
        for (int i = 0; i < r.length; i++) {
            neu[i] = r[i] + bonus;
        }
        return neu;
    }


    // ================== Stufe ERWEITERUNG ==================

    // istImBereich(x: Ganzzahl, von: Ganzzahl, bis: Ganzzahl): Wahrheitswert
    public static boolean istImBereich(int x, int von, int bis) {
        return von <= x && x <= bis;
    }

    // zaehleImBereich(r: Reihung von Ganzzahl, von: Ganzzahl, bis: Ganzzahl): Ganzzahl
    public static int zaehleImBereich(int[] r, int von, int bis) {
        int anzahl = 0;
        for (int x : r) {
            if (istImBereich(x, von, bis)) {
                anzahl++;
            }
        }
        return anzahl;
    }

    // indexDesMinimums(r: Reihung von Ganzzahl): Ganzzahl
    public static int indexDesMinimums(int[] r) {
        if (r.length == 0) {
            return -1;
        }
        int minIndex = 0;
        for (int i = 1; i < r.length; i++) {
            if (r[i] < r[minIndex]) {
                minIndex = i;
            }
        }
        return minIndex;
    }

    // tauscheErstesUndLetztes(r: Reihung von Ganzzahl): nichts
    public static void tauscheErstesUndLetztes(int[] r) {
        if (r.length >= 2) {
            int hilf = r[0];
            r[0] = r[r.length - 1];
            r[r.length - 1] = hilf;
        }
    }


    // ==================== Stufe EXPERTE ====================

    // zweitgroesster(r: Reihung von Ganzzahl): Ganzzahl
    // Zerlegung in zwei Durchlaeufe: erst das Maximum, dann das groesste
    // Element, das kleiner als das Maximum ist.
    public static int zweitgroesster(int[] r) {
        if (r.length < 2) {
            return -1;
        }
        int max = r[0];
        for (int x : r) {
            if (x > max) {
                max = x;
            }
        }
        boolean gefunden = false;
        int zweiter = 0;
        for (int x : r) {
            if (x < max) {
                if (!gefunden || x > zweiter) {
                    zweiter = x;
                    gefunden = true;
                }
            }
        }
        if (!gefunden) {
            return -1;
        }
        return zweiter;
    }

    // verschmelze(a: Reihung von Ganzzahl, b: Reihung von Ganzzahl): Reihung von Ganzzahl
    // Reissverschluss: immer das kleinere der beiden vorderen Elemente
    // nehmen, danach den Rest der anderen Reihung anhaengen.
    public static int[] verschmelze(int[] a, int[] b) {
        int[] neu = new int[a.length + b.length];
        int i = 0;
        int j = 0;
        int k = 0;
        while (i < a.length && j < b.length) {
            if (a[i] <= b[j]) {
                neu[k] = a[i];
                i++;
            } else {
                neu[k] = b[j];
                j++;
            }
            k++;
        }
        while (i < a.length) {
            neu[k] = a[i];
            i++;
            k++;
        }
        while (j < b.length) {
            neu[k] = b[j];
            j++;
            k++;
        }
        return neu;
    }
}


// ================= Papier-Compiler 0 – Werkstattpruefung ===============

class Werkstattpruefung {

    // anzahlUeberDurchschnitt(r: Reihung von Ganzzahl): Ganzzahl
    public static int anzahlUeberDurchschnitt(int[] r) {
        if (r.length == 0) {
            return 0;
        }
        int summe = 0;
        for (int x : r) {
            summe = summe + x;
        }
        double schnitt = (double) summe / r.length;
        int anzahl = 0;
        for (int x : r) {
            if (x > schnitt) {
                anzahl++;
            }
        }
        return anzahl;
    }
}

class L02WerkstattTest {

    @Test
    public void basis_istGerade_positiv() {
        assertTrue(Werkstatt.istGerade(4), "4 ist gerade: istGerade(4) muss true liefern");
        assertFalse(Werkstatt.istGerade(7), "7 ist ungerade: istGerade(7) muss false liefern");
        assertTrue(Werkstatt.istGerade(0), "0 ist gerade (0 % 2 ergibt 0)");
    }

    @Test
    public void basis_istGerade_negativ() {
        assertTrue(Werkstatt.istGerade(-6), "-6 ist gerade: istGerade(-6) muss true liefern");
        assertFalse(Werkstatt.istGerade(-3), "-3 ist ungerade - Achtung: -3 % 2 ergibt in Java -1, nicht 1! Pruefe besser auf n % 2 == 0");
    }

    @Test
    public void basis_restGold() {
        assertEquals(5, Werkstatt.restGold(50, 15), "50 Gold, Trank kostet 15: 3 Traenke (45 Gold), Rest 5 - Tipp: Operator %");
        assertEquals(0, Werkstatt.restGold(45, 15), "45 Gold, Trank kostet 15: genau 3 Traenke, Rest 0");
        assertEquals(10, Werkstatt.restGold(10, 15), "10 Gold reichen fuer keinen Trank zu 15 Gold: Es bleiben alle 10 Gold uebrig");
    }

    @Test
    public void basis_restGold_randfaelle() {
        assertEquals(2, Werkstatt.restGold(8, 3), "8 Gold, Trank kostet 3: 2 Traenke (6 Gold), Rest 2");
        assertEquals(0, Werkstatt.restGold(0, 7), "Wer 0 Gold hat, behaelt 0 Gold");
        assertEquals(0, Werkstatt.restGold(9, 1), "Kostet ein Trank 1 Gold, bleibt nichts uebrig");
    }

    @Test
    public void basis_kistenBenoetigt_passtGenau() {
        assertEquals(2, Werkstatt.kistenBenoetigt(10, 5), "10 Traenke, 5 pro Kiste: genau 2 Kisten");
        assertEquals(3, Werkstatt.kistenBenoetigt(12, 4), "12 Traenke, 4 pro Kiste: genau 3 Kisten - keine Kiste zu viel!");
    }

    @Test
    public void basis_kistenBenoetigt_aufrunden() {
        assertEquals(3, Werkstatt.kistenBenoetigt(11, 5), "11 Traenke, 5 pro Kiste: 2 volle Kisten + 1 Kiste fuer den Rest = 3 (aufrunden!)");
        assertEquals(3, Werkstatt.kistenBenoetigt(14, 5), "14 Traenke, 5 pro Kiste: 3 Kisten");
        assertEquals(1, Werkstatt.kistenBenoetigt(1, 5), "Auch ein einzelner Trank braucht eine Kiste");
    }

    @Test
    public void basis_kistenBenoetigt_randfaelle() {
        assertEquals(0, Werkstatt.kistenBenoetigt(0, 5), "Keine Traenke - keine Kiste");
        assertEquals(7, Werkstatt.kistenBenoetigt(7, 1), "7 Traenke, 1 pro Kiste: 7 Kisten");
    }

// =====================================================================
//  HÄUFIGE SCHÜLERFEHLER (Lehrerhinweise) – in Klammern der Test, der
//  den Fehler aufdeckt (diese Zuordnungen sind per Mutationstest geprueft)
// =====================================================================
//
//  Level 0.2 – Die Werkstatt
//  - Casting-Falle / ganzzahlige Division: "(a + b + c) / 3" liefert
//    2.0 statt 2.333..., "(double) ((a + b + c) / 3)" castet zu spaet
//    (erweiterung_durchschnitt_kommazahl). Richtig: / 3.0 oder
//    (double) (a + b + c) / 3. Merke: Der Cast wirkt nur auf das, was
//    direkt dahinter steht.
//  - (int) schneidet ab, rundet nicht: kistenBenoetigt mit
//    "(int) ((double) anzahl / proKiste)" oder ohne Rest-Abfrage ->
//    11 Traenke in 2 Kisten (basis_kistenBenoetigt_aufrunden); "immer
//    + 1" -> eine Kiste zu viel (basis_kistenBenoetigt_passtGenau).
//  - istGerade mit "n % 2 == 1" bzw. "!(n % 2 == 1)": In Java ist
//    -3 % 2 == -1 (basis_istGerade_negativ). Immer auf == 0 pruefen.
//  - = statt ==: "if (r[i] = 0)" ist in Java ein Kompilierfehler
//    ("incompatible types: int cannot be converted to boolean") - gut
//    zum Einueben der Fehlermeldung.
//  - schadensstufe: Grenzen <= 10 / <= 30 statt < 10 / < 30
//    (erweiterung_schadensstufe_grenzen); "== 0" statt "<= 0" vergisst
//    negativen Schaden (erweiterung_schadensstufe_kein).
//  - quersumme mit "while (n > 9)" -> vorderste Ziffer fehlt
//    (erweiterung_quersumme); n % 10 und n / 10 vertauscht ->
//    Endlosschleife.
//  - potenz: Start mit ergebnis = 0 oder Schleife mit i <= exponent
//    (experte_potenz, experte_potenz_exponentNull).
//  - ggT: am Ende b statt a zurueckgegeben (b ist dann 0) (experte_ggT).
//  - istPrimzahl: Grenze t * t < n statt <= n -> 9, 25, 49 gelten als
//    Primzahl (experte_istPrimzahl_quadratzahlen); n < 2 vergessen
//    (experte_istPrimzahl_kleinerZwei); "else return true;" in der
//    Schleife -> nur der Teiler 2 wird geprueft (experte_istPrimzahl_
//    quadratzahlen).
//  - Fehlender Rueckgabepfad: "return true;" nach der Schleife fehlt ->
//    Kompilierfehler "missing return statement". Im Abitur: BE-Abzug,
//    weil die Methode nicht in jedem Fall etwas liefert.
//
//  Level 0.3 – Die Vorratskammer
//  - Off-by-one: "i < r.length - 1" -> letztes Element fehlt
//    (basis_anzahlLeer, basis_enthaelt, erweiterung_anzahlAnstiege,
//    experte_zaehleZeichen_randfaelle); "i <= r.length" bzw.
//    r[r.length] -> ArrayIndexOutOfBoundsException (basis_anzahlLeer,
//    basis_letzterWert).
//  - Leer-Abfrage vergessen: letzterWert({}) -> Indexfehler
//    (basis_letzterWert_randfaelle).
//  - enthaelt mit "else return false;" in der Schleife -> nur das erste
//    Element wird geprueft (basis_enthaelt).
//  - umgedreht: "int[] neu = r;" kopiert nur die Referenz -> Werte
//    werden ueberschrieben (erweiterung_umgedreht); in-place umdrehen
//    und r zurueckgeben -> Original veraendert
//    (erweiterung_umgedreht_originalUnveraendert); r[r.length - i] ->
//    Indexfehler.
//  - verdopple mit for-each "for (int x : r) { x = x * 2; }" -> x ist
//    eine Kopie, r bleibt gleich (erweiterung_verdopple).
//  - anzahlAnstiege: Start bei i = 0 -> r[-1] (Indexfehler); ">=" statt
//    ">" zaehlt gleiche Nachbarn (erweiterung_anzahlAnstiege_randfaelle).
//  - ohneVokale: nur Kleinbuchstaben geprueft (experte_ohneVokale,
//    "Oase"); "&&" statt "||" in der Vokal-Bedingung -> nichts wird
//    entfernt. toLowerCase oder replace sind im Abitur NICHT erlaubt -
//    die Tests merken das nicht -> Code-Review!
//  - beginntMit: contains statt Anfangsvergleich findet "ball" in
//    "Feuerball" (experte_beginntMit); fehlende Laengenabfrage ->
//    Indexfehler bei beginntMit("Feu", "Feuer") (experte_beginntMit_
//    randfaelle); "else return true;" in der Schleife -> nur das erste
//    Zeichen wird verglichen (experte_beginntMit, "Fenster"). substring(...) == anfang statt equals:
//    in der Online-IDE gruen (== vergleicht dort inhaltlich), in echtem
//    Java und im Abitur falsch -> Code-Review.
//
//  Level 0.4 – Die Zauberschule (Methoden)
//  - Methodenkopf falsch uebersetzt (haeufigster Fehler in DS 4):
//    "public static manaKosten(...)" ohne Rueckgabetyp, "Ganzzahl" statt
//    int, "void" bei Methoden mit Rueckgabe, Parametertypen vergessen
//    ("int max3(a, b, c)"), "boolean" als Rueckgabetyp von manaKosten.
//    Nach dem Freischalten meldet der Compiler die Testzeile - die
//    Schueler sollen den Kopf dann wieder mit der Abi-Signatur
//    vergleichen, nicht den Test aendern.
//  - Freischalten: nur EINEN der zwei Bloecke einer Stufe geoeffnet
//    (dann laeuft der Test, aber das Hauptprogramm fehlt - unschaedlich)
//    oder "/*" bzw. "*/" einzeln geloescht -> Kompilierfehler weit
//    unten. Immer genau die ganze Markerzeile loeschen.
//  - heilwert mit + statt *, spruchtext ohne Leerzeichen oder mit festen
//    Werten (basis_heilwert, basis_spruchtext) - Parameter benutzen!
//  - istStaerker mit >= (basis_istStaerker: Gleichstand).
//  - max2 ohne else-Zweig/ohne zweites return -> Kompilierfehler
//    "missing return statement" (fehlender Rueckgabepfad).
//  - manaKosten ignoriert verstaerkt oder addiert statt zu verdoppeln
//    (erweiterung_manaKosten).
//  - kannZaubern mit > statt >= (erweiterung_kannZaubern: genau 30
//    Mana); rechnet die Kosten neu statt manaKosten aufzurufen und
//    vergisst dabei verstaerkt (erweiterung_kannZaubern_passtZuMana-
//    Kosten).
//  - max3 vergisst c (erweiterung_max3) oder startet mit 0 statt mit
//    einem der Werte (erweiterung_max3_gleicheUndNegative).
//  - Casting-Falle runden: "(int) wert" schneidet ab (experte_runden);
//    "(int) wert + 0.5" ist ein double -> Kompilierfehler "possible
//    lossy conversion". Math.round ist verboten - die Tests merken es
//    nicht -> Code-Review.
//  - wuerfelImBereich: ohne + 1 nie die obere Grenze
//    (experte_wuerfelImBereich_grenzen); "(int) Math.random() * ..." ->
//    der Cast macht aus 0.xx sofort 0, Ergebnis immer von (dieselbe
//    Testmeldung); ohne + von bzw. "* bis" -> Werte ausserhalb des
//    Bereichs; "gerundete" Variante (... + 0.5) trifft die Grenzen nur
//    halb so oft (experte_wuerfelImBereich_gleichverteilt).

//    ab 0 (experte_zauberkette).
//
//  Level 0.5 – Die Zauberschule, Teil II (Wert vs. Referenz)
//  - Parameter ueberschreiben bringt nichts: "r = new int[r.length];"
//    in fuelle -> die neue Reihung ist ausserhalb unsichtbar, das
//    Original bleibt (basis_fuelle). Genauso bei int-Parametern: Eine
//    Methode kann die Variable des Aufrufers nie aendern (Wert-Raetsel
//    im Hauptprogramm: jahr bleibt 2026, obwohl quersumme n auf 0

//    doppeltes Maximum zaehlt ({3, 9, 5, 9} -> 9); Start mit zweiter = 0

//    (experte_verschmelze_eineSeiteZuerstLeer); "||" statt "&&" in der
//    Reissverschluss-Bedingung -> Indexfehler; nur aneinanderhaengen.
//
//  Papier-Compiler 0 – anzahlUeberDurchschnitt
//  - Durchschnitt als Ganzzahl ("int schnitt = summe / r.length;") oder
//    Cast zu spaet: Bei positiven Werten zufaellig richtig (zwischen
//    abgerundetem und echtem Durchschnitt liegt keine Ganzzahl), bei
//    negativem Durchschnitt falsch: -3 / 2 = -1 statt -1.5
//    (durchschnittIstKommazahl). Im Abitur trotzdem immer Kommazahl!
//  - >= statt > (echtGroesser, einWert); durch r.length - 1 geteilt
//    (echtGroesser); Summe mit Off-by-one oder return in der Schleife
//    ("Rueckgabe zu frueh") (normalfall).
//  - Leer-Abfrage vergessen: Mit Kommazahl-Division entsteht 0.0 / 0 =
//    NaN, jeder Vergleich ist false -> Ergebnis 0, der Test leereReihung
//    ist dann trotzdem gruen. Im Abitur gibt es fuer die fehlende
//    Abfrage dennoch Abzug -> beim Papier-Compiler ansprechen.
//
//  Was die Tests NICHT erkennen koennen (-> Code-Review / Abi-Check):
//  - verbotene Hilfsmittel mit richtigem Ergebnis: Math.round in
//    runden, Math.pow in potenz, toLowerCase/replace in ohneVokale;
//  - == statt equals bei Zeichenketten (Online-IDE vergleicht inhaltlich);
