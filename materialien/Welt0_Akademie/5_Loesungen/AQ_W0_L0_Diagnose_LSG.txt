class Diagnose {

    // ===================== Stufe BASIS =====================

    // Akkumulator-Muster: Startwert 0, jedes Element dazuaddieren.
    // Leere Reihung: Die Schleife laeuft gar nicht -> 0. Kein Sonderfall noetig.
    public static int summe(int[] r) {
        int gesamt = 0;
        for (int i = 0; i < r.length; i++) {
            gesamt = gesamt + r[i];
        }
        return gesamt;
    }

    // Startwert ist das ERSTE Element, nicht 0 (sonst falsch bei nur
    // negativen Zahlen). Deshalb darf die Schleife bei 1 beginnen.
    // Vorbedingung laut Aufgabe: r ist nicht leer.
    public static int maximum(int[] r) {
        int max = r[0];
        for (int i = 1; i < r.length; i++) {
            if (r[i] > max) {
                max = r[i];
            }
        }
        return max;
    }

    // Zaehler-Muster: nur erhoehen, wenn die Bedingung erfuellt ist.
    // Das return steht NACH der Schleife.
    public static int zaehle(int[] r, int x) {
        int anzahl = 0;
        for (int i = 0; i < r.length; i++) {
            if (r[i] == x) {
                anzahl++;
            }
        }
        return anzahl;
    }


    // ================== Stufe ERWEITERUNG ==================

    // Lineare Suche: Beim ersten Treffer sofort zurueckgeben - dadurch
    // wird automatisch das ERSTE Vorkommen geliefert.
    // Das "return -1" steht NACH der Schleife: erst wenn ALLE Elemente
    // geprueft sind, steht fest, dass x nicht vorkommt.
    public static int indexVon(int[] r, int x) {
        for (int i = 0; i < r.length; i++) {
            if (r[i] == x) {
                return i;
            }
        }
        return -1;
    }

    // Nachbarvergleich r[i] mit r[i + 1]: Die Schleife darf nur bis
    // r.length - 2 laufen (Bedingung i < r.length - 1), sonst greift
    // r[i + 1] hinter das Ende der Reihung.
    // Leere Reihung: r.length - 1 = -1, die Schleife laeuft nicht -> true.
    // Gegenbeispiel finden -> sofort false; erst am Ende true.
    public static boolean istAufsteigendSortiert(int[] r) {
        for (int i = 0; i < r.length - 1; i++) {
            if (r[i] > r[i + 1]) {
                return false;
            }
        }
        return true;
    }

    // Zwei Fallen: (1) Division durch 0 bei leerer Reihung,
    // (2) Ganzzahl / Ganzzahl rechnet ganzzahlig (3 / 2 = 1).
    // Deshalb vorher abfragen und vor der Division in double umwandeln.
    public static double durchschnitt(int[] r) {
        if (r.length == 0) {
            return 0.0;
        }
        int gesamt = summe(r);                 // Methode aus Basis wiederverwenden
        return (double) gesamt / r.length;
    }


    // ==================== Stufe EXPERTE ====================

    // Nur charAt und length (Abitur-Liste der Zeichenketten-Operationen).
    // Gross- und Kleinbuchstaben werden einzeln abgefragt, weil
    // toLowerCase im Abitur NICHT zur Verfuegung steht.
    public static int anzahlVokale(String s) {
        int anzahl = 0;
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u'
                    || c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U') {
                anzahl++;
            }
        }
        return anzahl;
    }

    // Zwei Indizes laufen von aussen nach innen. Sobald ein Paar nicht
    // passt -> false. Treffen sich die Indizes, ist alles geprueft.
    // "" (links = 0, rechts = -1) und ein Zeichen (0, 0): Die Schleife
    // laeuft nicht -> true. Bei ungerader Laenge wird das mittlere
    // Zeichen nicht verglichen - das ist richtig so.
    public static boolean istPalindrom(String s) {
        int links = 0;
        int rechts = s.length() - 1;
        while (links < rechts) {
            if (s.charAt(links) != s.charAt(rechts)) {
                return false;
            }
            links++;
            rechts--;
        }
        return true;
    }

    // Alternative mit Zaehlschleife (ebenfalls korrekt):
    //     for (int i = 0; i < s.length() / 2; i++) {
    //         if (s.charAt(i) != s.charAt(s.length() - 1 - i)) {
    //             return false;
    //         }
    //     }
    //     return true;
}

class DiagnoseBasisTest {

    @Test
    public void basis_summe_normal() {
        int[] r = {3, 5, 2};
        assertEquals(10, Diagnose.summe(r), "summe({3, 5, 2}) muss 10 ergeben");
    }

    @Test
    public void basis_summe_mitNegativenZahlen() {
        int[] r = {-4, 4, -2};
        assertEquals(-2, Diagnose.summe(r), "summe({-4, 4, -2}) muss -2 ergeben - negative Zahlen zaehlen mit");
    }

    @Test
    public void basis_summe_einElement() {
        int[] r = {7};
        assertEquals(7, Diagnose.summe(r), "summe({7}) muss 7 ergeben");
    }

    @Test
    public void basis_summe_leereReihung() {
        int[] r = new int[0];
        assertEquals(0, Diagnose.summe(r), "Die Summe einer leeren Reihung ist 0");
    }

    @Test
    public void basis_maximum_inDerMitte() {
        int[] r = {3, 9, 2};
        assertEquals(9, Diagnose.maximum(r), "maximum({3, 9, 2}) muss 9 ergeben");
    }

    @Test
    public void basis_maximum_amAnfangUndAmEnde() {
        int[] vorne = {9, 1, 2};
        int[] hinten = {1, 2, 9};
        assertEquals(9, Diagnose.maximum(vorne), "maximum({9, 1, 2}) muss 9 ergeben - steht das Maximum ganz vorne?");
        assertEquals(9, Diagnose.maximum(hinten), "maximum({1, 2, 9}) muss 9 ergeben - laeuft deine Schleife bis zum letzten Element?");
    }

    @Test
    public void basis_maximum_nurNegativeZahlen() {
        int[] r = {-5, -2, -9};
        assertEquals(-2, Diagnose.maximum(r), "maximum({-5, -2, -9}) muss -2 ergeben - starte nicht mit max = 0, sondern mit r[0]");
    }

    @Test
    public void basis_zaehle_mehrfach() {
        int[] r = {1, 2, 1, 1};
        assertEquals(3, Diagnose.zaehle(r, 1), "zaehle({1, 2, 1, 1}, 1) muss 3 ergeben");
    }

    @Test
    public void basis_zaehle_nichtEnthalten() {
        int[] r = {1, 2, 3};
        assertEquals(0, Diagnose.zaehle(r, 5), "zaehle({1, 2, 3}, 5) muss 0 ergeben");
    }

    @Test
    public void basis_zaehle_leereReihung() {
        int[] r = new int[0];
        assertEquals(0, Diagnose.zaehle(r, 1), "In einer leeren Reihung kommt nichts vor - Ergebnis 0");
    }
}


class DiagnoseErweiterungTest {

    @Test
    public void erweiterung_indexVon_gefunden() {
        int[] r = {4, 7, 9};
        assertEquals(1, Diagnose.indexVon(r, 7), "indexVon({4, 7, 9}, 7) muss 1 ergeben - der Index beginnt bei 0");
    }

    @Test
    public void erweiterung_indexVon_erstesVorkommen() {
        int[] r = {5, 3, 5, 3};
        assertEquals(0, Diagnose.indexVon(r, 5), "indexVon({5, 3, 5, 3}, 5) muss 0 ergeben - gesucht ist das ERSTE Vorkommen");
        assertEquals(1, Diagnose.indexVon(r, 3), "indexVon({5, 3, 5, 3}, 3) muss 1 ergeben - gesucht ist das ERSTE Vorkommen");
    }

    @Test
    public void erweiterung_indexVon_amEnde() {
        int[] r = {4, 7, 9};
        assertEquals(2, Diagnose.indexVon(r, 9), "indexVon({4, 7, 9}, 9) muss 2 ergeben - wird das letzte Element geprueft?");
    }

    @Test
    public void erweiterung_indexVon_nichtGefunden() {
        int[] r = {4, 7, 9};
        assertEquals(-1, Diagnose.indexVon(r, 8), "indexVon({4, 7, 9}, 8) muss -1 ergeben, weil 8 nicht vorkommt");
    }

    @Test
    public void erweiterung_indexVon_leereReihung() {
        int[] r = new int[0];
        assertEquals(-1, Diagnose.indexVon(r, 1), "In einer leeren Reihung muss indexVon -1 liefern");
    }

    @Test
    public void erweiterung_sortiert_ja() {
        int[] r = {1, 3, 7, 12};
        assertTrue(Diagnose.istAufsteigendSortiert(r), "{1, 3, 7, 12} ist aufsteigend sortiert");
    }

    @Test
    public void erweiterung_sortiert_gleicheNachbarn() {
        int[] r = {2, 2, 5};
        assertTrue(Diagnose.istAufsteigendSortiert(r), "{2, 2, 5} gilt als aufsteigend sortiert - gleiche Nachbarn sind erlaubt");
    }

    @Test
    public void erweiterung_sortiert_nein() {
        int[] mitte = {1, 5, 3, 7};
        int[] vorne = {5, 1, 2, 3};
        int[] hinten = {1, 2, 3, 0};
        assertFalse(Diagnose.istAufsteigendSortiert(mitte), "{1, 5, 3, 7} ist NICHT sortiert");
        assertFalse(Diagnose.istAufsteigendSortiert(vorne), "{5, 1, 2, 3} ist NICHT sortiert - die Unordnung steht ganz vorne");
        assertFalse(Diagnose.istAufsteigendSortiert(hinten), "{1, 2, 3, 0} ist NICHT sortiert - wird das letzte Paar verglichen?");
    }

    @Test
    public void erweiterung_sortiert_leerUndEinElement() {
        int[] leer = new int[0];
        int[] eins = {42};
        assertTrue(Diagnose.istAufsteigendSortiert(leer), "Eine leere Reihung gilt als sortiert");
        assertTrue(Diagnose.istAufsteigendSortiert(eins), "Eine Reihung mit einem Element gilt als sortiert");
    }

    @Test
    public void erweiterung_durchschnitt_ganzzahlig() {
        int[] r = {2, 4, 6};
        assertEquals(4.0, Diagnose.durchschnitt(r), "durchschnitt({2, 4, 6}) muss 4.0 ergeben");
    }

    @Test
    public void erweiterung_durchschnitt_mitKomma() {
        int[] r = {1, 2};
        assertEquals(1.5, Diagnose.durchschnitt(r), "durchschnitt({1, 2}) muss 1.5 ergeben - Vorsicht: Ganzzahl / Ganzzahl rechnet ganzzahlig!");
    }

    @Test
    public void erweiterung_durchschnitt_leereReihung() {
        int[] r = new int[0];
        assertEquals(0.0, Diagnose.durchschnitt(r), "Der Durchschnitt einer leeren Reihung soll 0.0 sein - teile nie durch 0");
    }
}

    }

// =====================================================================
//  HÄUFIGE SCHÜLERFEHLER (und welcher Test sie aufdeckt)
//
//  Stufe BASIS
//  - Schleifengrenze i <= r.length -> Indexfehler beim letzten Durchlauf
//    (Laufzeitfehler in allen Tests mit Reihung).
//  - maximum: Startwert max = 0 statt r[0] -> basis_maximum_nurNegativeZahlen
//    liefert 0 statt -2.
//  - maximum: Schleife bis r.length - 1 (exklusiv) -> letztes Element
//    fehlt, basis_maximum_amAnfangUndAmEnde schlaegt fehl.
//  - zaehle/summe: return INNERHALB der Schleife -> Ergebnis nach dem
//    ersten Element.
//  - Akkumulator in der Schleife deklariert (int summe = 0; im Rumpf)
//    -> wird jedes Mal zurueckgesetzt bzw. ist nach der Schleife
//    unbekannt.
//
//  Stufe ERWEITERUNG
//  - indexVon: "else return -1;" in der Schleife -> nur das erste
//    Element wird geprueft (erweiterung_indexVon_gefunden).
//  - indexVon: kein vorzeitiger Abbruch, sondern "pos = i" -> liefert
//    das LETZTE Vorkommen (erweiterung_indexVon_erstesVorkommen).
//  - istAufsteigendSortiert: Bedingung i < r.length -> r[i + 1] greift
//    hinter das Ende (Laufzeitfehler).
//  - istAufsteigendSortiert: "if (r[i] <= r[i + 1]) return true;" ->
//    Entscheidung schon nach dem ersten Paar (erweiterung_sortiert_nein,
//    Fall "hinten").
//  - istAufsteigendSortiert mit ">=" statt ">" im Abbruch -> gleiche Nachbarn
//    werden als unsortiert erkannt (erweiterung_sortiert_gleicheNachbarn).
//  - durchschnitt: summe / r.length ohne Umwandlung -> 1.0 statt 1.5
//    (erweiterung_durchschnitt_mitKomma). Auch (double) (summe / r.length)
//    hilft NICHT - dann wird erst ganzzahlig geteilt und danach umgewandelt.
//  - durchschnitt: keine Leer-Abfrage -> Division durch 0
//    (erweiterung_durchschnitt_leereReihung; bei double-Division NaN).
//
//  Stufe EXPERTE
//  - anzahlVokale: nur Kleinbuchstaben abgefragt
//    (experte_vokale_grossbuchstaben).
//  - anzahlVokale mit toLowerCase(): laeuft in der Online-IDE, ist im
//    Abitur aber keine erlaubte Zeichenketten-Operation -> im Gespraech
//    ansprechen, auch wenn der Test gruen ist.
//  - istPalindrom: Gegenindex s.length() - i statt s.length() - 1 - i
//    -> Indexfehler.
//  - istPalindrom: Zeichenkette umdrehen und mit == vergleichen. In der
//    Online-IDE vergleicht == Zeichenketten zwar inhaltlich, in echtem
//    Java und im Abitur gilt aber: Zeichenketten IMMER mit equals.
//  - istPalindrom: "return true" schon nach dem ersten passenden Paar
//    (experte_palindrom_nein, Fall "abca").
//
//  ABI-BEZUG
//  Genau dieser Aufgabentyp ("Implementieren Sie die Methode ... auf
//  einer Reihung") steht in fast jedem niedersaechsischen Abitur
//  (AFB I/II), handschriftlich. Daher nach dem Testrunner immer der
//  Abi-Check:
//    [ ] Signatur genau wie gefordert, return in JEDEM Pfad
//    [ ] Leer-Abfrage bzw. Vorbedingung beachtet (leere Reihung, "")
//    [ ] Schleifengrenzen 0 bis r.length - 1, bei Nachbarvergleich
//        nur bis r.length - 2
//    [ ] nur erlaubte Zeichenketten-Operationen, Vergleich mit equals
//    [ ] unsichere Stellen kurz kommentieren
//  Anschlussaufgabe [PAPIER]: indexVon als Struktogramm (Kopf
//  indexVon(r: Reihung von Ganzzahl, x: Ganzzahl): Ganzzahl) und eine
//  Tracetabelle zu istAufsteigendSortiert({1, 2, 3, 0}).
// =====================================================================


// ---------------------------------------------------------------------
//  Programmstart – Demo
// ---------------------------------------------------------------------

int[] schatz = {12, 5, 30, 5, 9};

println("Summe:      " + Diagnose.summe(schatz));                    // 61
println("Maximum:    " + Diagnose.maximum(schatz));                  // 30
println("Anzahl 5:   " + Diagnose.zaehle(schatz, 5));                // 2
println("Index 30:   " + Diagnose.indexVon(schatz, 30));             // 2
println("Sortiert?   " + Diagnose.istAufsteigendSortiert(schatz));   // false
println("Schnitt:    " + Diagnose.durchschnitt(schatz));             // 12.2
println("Vokale:     " + Diagnose.anzahlVokale("Das Dungeon der Datenstrukturen"));  // 10
println("Palindrom?  " + Diagnose.istPalindrom("rentner"));          // true
