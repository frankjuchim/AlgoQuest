// ---------------------------------------------------------------------
//  GERÜST – hier nichts veraendern
// ---------------------------------------------------------------------
//  Die Klasse Anzeige hilft beim Ausgeben von Reihungen. Die Tests
//  benutzen sie, um dir in der Meldung die ganze Reihung zu zeigen,
//  z. B. "[erwartet: {1, 2, 3}, tatsaechlich: {3, 1, 2}]".
//  Du darfst sie auch selbst im Hauptprogramm benutzen.
// ---------------------------------------------------------------------

class Anzeige {

    // text(r: Reihung von Ganzzahl): Zeichenkette
    // Liefert die Reihung als Text, z. B. "{4, 8, 15}".
    // Leere Reihung -> "{}", null (noch keine Reihung) -> "null".
    public static String text(int[] r) {
        if (r == null) {
            return "null";
        }
        String s = "{";
        for (int i = 0; i < r.length; i++) {
            if (i > 0) {
                s = s + ", ";
            }
            s = s + r[i];
        }
        return s + "}";
    }
}

    // Baut die Karte aus Textzeilen: feld[z][s] = zeilen[z].charAt(s).
    // Alle Zeilen muessen gleich lang sein.
    public Karte(String[] zeilen) {
        feld = new char[zeilen.length][zeilen[0].length()];
        for (int z = 0; z < zeilen.length; z++) {
            for (int s = 0; s < zeilen[z].length(); s++) {
                feld[z][s] = zeilen[z].charAt(s);
            }
        }
        schritte = 0;
    }

    public int getZeilen() {
        return feld.length;
    }

    public int getSpalten() {
        return feld[0].length;
    }

    public char getFeld(int z, int s) {
        return feld[z][s];
    }

    public void setFeld(int z, int s, char c) {
        feld[z][s] = c;
    }

    // Die Karte als Text: Zeilen mit "\n" getrennt (ohne "\n" am Ende).
    // println(karte.alsText()) zeichnet die Karte auf die Konsole.
    public String alsText() {
        String text = "";
        for (int z = 0; z < feld.length; z++) {
            if (z > 0) {
                text = text + "\n";
            }
            for (int s = 0; s < feld[z].length; s++) {

// Hilfsklasse fuer Tests und Hauptprogramm: die vier Testkarten.
// Jeder Aufruf liefert eine NEUE Karte (frisch vernebelt).
class Kartenbuch {

    // Die offene Hoehle (5 Zeilen, 7 Spalten), Start (1, 1), Ziel (3, 5):
    //       0123456
    //    0  #######
    //    1  #S....#
    //    2  #..#..#
    //    3  #....Z#
    //    4  #######
    public static Karte hoehle() {
        String[] zeilen = {
            "#######",
            "#S....#",
            "#..#..#",
            "#....Z#",
            "#######"
        };
        return new Karte(zeilen);
    }

    // Das Labyrinth (9 Zeilen, 9 Spalten), Start (1, 1), Ziel (5, 7).
    // Es gibt einen Weg, mehrere Sackgassen und unten rechts eine
    // abgeschlossene Kammer (nur UEBER ECK mit dem Gang verbunden).
    //       012345678
    //    0  #########
    //    1  #S#...#.#
    //    2  #.#.#.#.#
    //    3  #...#...#
    //    4  #.###.#.#
    //    5  #...#.#Z#
    //    6  ###.##.##
    //    7  #...#...#
    //    8  #########
    public static Karte labyrinth() {
        String[] zeilen = {
            "#########",
            "#S#...#.#",
            "#.#.#.#.#",
            "#...#...#",
            "#.###.#.#",
            "#...#.#Z#",
            "###.##.##",
            "#...#...#",
            "#########"
        };
        return new Karte(zeilen);
    }

    // Der Kerker (6 Zeilen, 8 Spalten), Start (1, 1), Ziel (2, 5).
    // KEIN Weg: Das Ziel liegt in einer eigenen Kammer.
    //       01234567
    //    0  ########
    //    1  #S..#..#
    //    2  #.#.#Z.#
    //    3  #...####
    //    4  #.#....#
    //    5  ########
    public static Karte kerker() {
        String[] zeilen = {
            "########",
            "#S..#..#",
            "#.#.#Z.#",
            "#...####",
            "#.#....#",
            "########"
        };
        return new Karte(zeilen);
    }

    // Die Lichtung (3 Zeilen, 4 Spalten), Start (1, 1), Ziel (2, 3).
    // OHNE Randmauer: Wer nicht prueft, ob (z, s) im Feld liegt, stuerzt ab.
    //       0123
    //    0  ..#.
    //    1  .S..
    //    2  #..Z
    public static Karte lichtung() {
        String[] zeilen = {
            "..#.",
            ".S..",
            "#..Z"
        };
        return new Karte(zeilen);
    }

    // Erwartete Karte als Text (Zeilen mit "\n" verbunden) - wie alsText().
    public static String text(String[] zeilen) {
        String t = "";
        for (int i = 0; i < zeilen.length; i++) {
            if (i > 0) {
                t = t + "\n";
            }
            t = t + zeilen[i];
        }
        return t;
    }
}
