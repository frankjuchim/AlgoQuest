// Die Werkstatt stellt fuer jede Deko die Turtle auf (Startpunkt,
// Startrichtung, Farbe), setzt die Zaehler auf 0, ruft deine Methode
// auf und gibt danach die Zaehlerstaende aus.
//
// Startrichtungen (die Turtle blickt anfangs nach RECHTS):
//   baum, baum3    Start unten in der Mitte, Drehung +90 -> Blick nach oben
//   koch           Start links oben, keine Drehung -> Blick nach rechts,
//                  die Zacken zeigen nach UNTEN (Eiszapfen)
//   schneeflocke   Start links, keine Drehung -> Blick nach rechts
//   sierpinski     Start links unten, KEINE Drehung -> Blick nach rechts,
//                  die Spitze des Dreiecks zeigt nach oben
//   hbaum          Start in der Mitte, keine Drehung -> Blick nach rechts
class Werkstatt {

    private static boolean weltBereit = false;

    // Zeichnet eine Deko und gibt Aufrufe und Striche aus.
    // deko: "baum", "baum3", "koch", "schneeflocke", "sierpinski", "hbaum"
    static void zeichne(String deko, int tiefe) {
        if (!pruefe(deko, tiefe)) {
            return;
        }
        welt();
        Fraktal.aufrufe = 0;
        Fraktal.striche = 0;
        Turtle t = aufstellen(deko, 0);
        starte(deko, t, tiefe);
        println("Deko '" + deko + "', Tiefe " + tiefe);
        println("Aufrufe: " + Fraktal.aufrufe);
        println("Striche: " + Fraktal.striche);
    }

    // Zaehlt fuer die Tiefen 0 bis bisTiefe die Aufrufe und Striche und
    // gibt eine Tabelle aus. Gezeichnet wird dabei AUSSERHALB des
    // sichtbaren Bereichs - du siehst also nichts, bekommst aber schnell
    // alle Zahlen. Erst selbst vorhersagen, dann nachpruefen!
    static void tabelle(String deko, int bisTiefe) {
        if (!pruefe(deko, bisTiefe)) {
            return;
        }
        welt();
        println("Deko '" + deko + "'");
        println("Tiefe | Aufrufe | Striche");
        for (int tiefe = 0; tiefe <= bisTiefe; tiefe++) {
            Fraktal.aufrufe = 0;
            Fraktal.striche = 0;
            Turtle t = aufstellen(deko, 5000);
            starte(deko, t, tiefe);
            println("  " + tiefe + "   | " + Fraktal.aufrufe + " | " + Fraktal.striche);
        }
    }


    // =============== intern ===============

    private static void welt() {
        if (!weltBereit) {
            new World(800, 700);
            weltBereit = true;
        }
    }

    // Prueft den Namen und begrenzt die Tiefe (sonst rechnet die
    // Online-IDE minutenlang).
    private static boolean pruefe(String deko, int tiefe) {
        int max = maxTiefe(deko);
        if (max < 0) {
            println("Unbekannte Deko '" + deko + "'. Erlaubt: baum, baum3, koch, schneeflocke, sierpinski, hbaum");
            return false;
        }
        if (tiefe < 0 || tiefe > max) {
            println("Tiefe " + tiefe + " geht nicht: fuer '" + deko + "' sind 0 bis " + max + " erlaubt.");
            return false;
        }
        return true;
    }

    private static int maxTiefe(String deko) {
        if (deko.equals("baum")) {
            return 12;
        }
        if (deko.equals("baum3")) {
            return 8;
        }
        if (deko.equals("koch")) {
            return 6;
        }
        if (deko.equals("schneeflocke")) {
            return 5;
        }
        if (deko.equals("sierpinski")) {
            return 7;
        }
        if (deko.equals("hbaum")) {
            return 10;
        }
        return -1;
    }

    // Stellt die Turtle auf. verschiebung > 0 schiebt alles nach rechts
    // aus dem sichtbaren Bereich (fuer die Tabelle).
    private static Turtle aufstellen(String deko, int verschiebung) {
        Turtle t;
        if (deko.equals("baum") || deko.equals("baum3")) {
            t = new Turtle(400 + verschiebung, 680);
            t.turn(90);                          // Blick nach oben
            t.setBorderColor("#C8A165");
        } else if (deko.equals("koch")) {
            t = new Turtle(100 + verschiebung, 150);
            t.setBorderColor("#6FB7E9");
        } else if (deko.equals("schneeflocke")) {
            t = new Turtle(150 + verschiebung, 470);
            t.setBorderColor("#6FB7E9");
        } else if (deko.equals("sierpinski")) {
            t = new Turtle(150 + verschiebung, 630);
            t.setBorderColor("#E8743B");     // KEINE Startdrehung
        } else {
            t = new Turtle(400 + verschiebung, 350);
            t.setBorderColor("#7FBF5F");
        }
        return t;
    }

    private static void starte(String deko, Turtle t, int tiefe) {
        if (deko.equals("baum")) {
            Fraktal.baum(t, 150, tiefe);
        } else if (deko.equals("baum3")) {
            Fraktal.baum3(t, 130, tiefe);
        } else if (deko.equals("koch")) {
            Fraktal.koch(t, 600, tiefe);
        } else if (deko.equals("schneeflocke")) {
            Fraktal.schneeflocke(t, 500, tiefe);
        } else if (deko.equals("sierpinski")) {
            Fraktal.sierpinski(t, 500, tiefe);
        } else {
            Fraktal.hBaum(t, 300, tiefe);
        }
    }
}

    // Die zweite Drehung ist -180, damit sich beide exakt aufheben und
    // der Blickwinkel unveraendert bleibt.
    static void zurueck(Turtle t, double laenge) {
        t.penUp();
        t.turn(180);
        t.forward(laenge);
        t.turn(-180);
        t.penDown();

    // ---------------- Deko 1: Baum (fertiges Beispiel) ----------------

    static void baum(Turtle t, double laenge, int tiefe) {
        aufrufe++;

            return;                            // Basisfall
