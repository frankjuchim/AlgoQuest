// ---------------------------------------------------------------------
//  GERÜST – unveraendert
// ---------------------------------------------------------------------
//  Die Klasse Anzeige hilft beim Ausgeben von Reihungen. Die Tests
//  benutzen sie, um dir in der Meldung die ganze Reihung zu zeigen,
//  z. B. "[erwartet: {1, 2, 3}, tatsaechlich: {3, 1, 2}]".
//  Du darfst sie auch selbst im Hauptprogramm benutzen.
// ---------------------------------------------------------------------

class Anzeige {

    // text(r: Reihung von Ganzzahl): Zeichenkette
    // Liefert die Reihung als Text, z. B. "{4, 8, 15}".
    // Leere Reihung -> "{}", null (noch keine Reihung) -> "null".
    public static String text(int[] r) {
        if (r == null) {
            return "null";
        }
        String s = "{";
        for (int i = 0; i < r.length; i++) {
            if (i > 0) {
                s = s + ", ";
            }
            s = s + r[i];
        }
        return s + "}";
    }
}


// ---------------------------------------------------------------------
//  LÖSUNGEN
// ---------------------------------------------------------------------
//  Ueber jeder Methode steht ihre Signatur in Abi-Notation:
//      name(p: Typ, q: Typ): Rueckgabetyp
//  Neu in Welt 3:
//      Zeichen                          char
//      Reihung von Reihung von Zeichen  char[][]   (2D-Reihung)
//
//  So sieht jede rekursive Methode aus (Beispiel fakultaet):
//
//      fakultaet(n: Ganzzahl): Ganzzahl
//      +-----------------------------------------------------+
//      |                       n = 0 ?                       |
//      |           wahr           |          falsch          |
//      +--------------------------+--------------------------+
//      | gib 1 zurueck            | gib n * fakultaet(n - 1) |
//      | // Basisfall             | zurueck                  |
//      |                          | // Rekursionsschritt     |
//      +-----------------------------------------------------+
// ---------------------------------------------------------------------

// ===================== Level 3.1 – Rekursion (DS 18) ===================
//  Alle Methoden sind static: Aufruf mit Klassenname, z. B.
//  Rekursion.fakultaet(5). ALLE Methoden dieser Klasse sind
//  REKURSIV - ohne Schleife!

class Rekursion {

    // VORGEGEBEN: zaehlt die Aufrufe von fibonacci (siehe Zaehl-Regeln).
    public static int aufrufe = 0;

    // ===================== Stufe BASIS =====================

    // fakultaet(n: Ganzzahl): Ganzzahl
    // n! = 1 * 2 * ... * n fuer n >= 0. Es gilt 0! = 1 und n! = n * (n - 1)!.
    // REKURSIV - ohne Schleife.
    public static int fakultaet(int n) {
        if (n == 0) {
            return 1;                                   // Basisfall
        }
        return n * fakultaet(n - 1);                    // Rekursionsschritt
    }

    // summeBis(n: Ganzzahl): Ganzzahl
    // 0 + 1 + 2 + ... + n fuer n >= 0, also summeBis(0) = 0.
    // REKURSIV - ohne Schleife.
    public static int summeBis(int n) {
        if (n == 0) {
            return 0;                                   // Basisfall
        }
        return n + summeBis(n - 1);                     // Rekursionsschritt
    }

    // potenz(basis: Ganzzahl, exponent: Ganzzahl): Ganzzahl
    // basis hoch exponent fuer exponent >= 0; potenz(x, 0) = 1.
    // REKURSIV - ohne Schleife (und ohne Math.pow).
    public static int potenz(int basis, int exponent) {
        if (exponent == 0) {
            return 1;                                   // Basisfall
        }
        return basis * potenz(basis, exponent - 1);     // Rekursionsschritt
    }


    // ================== Stufe ERWEITERUNG ==================

    // anzahlZiffern(n: Ganzzahl): Ganzzahl
    // Anzahl der Ziffern von n >= 0, z. B. 4711 -> 4. Achtung: 0 hat EINE
    // Ziffer. Tipp: n / 10 schneidet die letzte Ziffer ab.
    // REKURSIV - ohne Schleife.
    public static int anzahlZiffern(int n) {
        if (n < 10) {
            return 1;                                   // Basisfall: einstellig
        }
        return 1 + anzahlZiffern(n / 10);               // Rekursionsschritt
    }

    // istPalindrom(s: Zeichenkette): Wahrheitswert
    // true, wenn s vorwaerts und rueckwaerts gleich ist ("otto", "level").
    // Exakter Zeichenvergleich (gross/klein zaehlt). "" und Zeichenketten
    // mit 1 Zeichen sind Palindrome. Idee: erstes und letztes Zeichen
    // gleich UND der Teil dazwischen ist ein Palindrom.
    // Nur length(), charAt(i) und substring(a, b) benutzen.
    // REKURSIV - ohne Schleife.
    public static boolean istPalindrom(String s) {
        if (s.length() <= 1) {
            return true;                                // Basisfall
        }
        if (s.charAt(0) != s.charAt(s.length() - 1)) {
            return false;                               // Basisfall: Raender verschieden
        }
        return istPalindrom(s.substring(1, s.length() - 1));   // Rekursionsschritt
    }

    // summeAb(r: Reihung von Ganzzahl, i: Ganzzahl): Ganzzahl
    // Summe r[i] + r[i + 1] + ... + r[r.length - 1].
    // Basisfall: i == r.length (keine Elemente mehr) -> 0.
    // Die ganze Reihung: summeAb(r, 0).
    // REKURSIV - ohne Schleife.
    public static int summeAb(int[] r, int i) {
        if (i == r.length) {
            return 0;                                   // Basisfall
        }
        return r[i] + summeAb(r, i + 1);                // Rekursionsschritt
    }


    // =================== Stufe EXPERTE =====================

    // maximumAb(r: Reihung von Ganzzahl, i: Ganzzahl): Ganzzahl
    // Groesster Wert unter r[i], r[i + 1], ..., r[r.length - 1]
    // (0 <= i < r.length, die Reihung ist also nicht leer).
    // REKURSIV - ohne Schleife.
    public static int maximumAb(int[] r, int i) {
        if (i == r.length - 1) {
            return r[i];                                // Basisfall: letztes Element
        }
        int maxRest = maximumAb(r, i + 1);              // Rekursionsschritt
        if (r[i] > maxRest) {
            return r[i];
        }
        return maxRest;
    }

    // dual(n: Ganzzahl): Zeichenkette
    // Dualdarstellung von n >= 0 ohne fuehrende Nullen:
    // dual(0) = "0", dual(1) = "1", dual(6) = "110", dual(13) = "1101".
    // Tipp: Die letzte Dualziffer ist n % 2, der Rest davor ist dual(n / 2).
    // REKURSIV - ohne Schleife.
    public static String dual(int n) {
        if (n < 2) {
            return "" + n;                              // Basisfall: "0" oder "1"
        }
        return dual(n / 2) + (n % 2);                   // Rekursionsschritt
    }

    // fibonacci(n: Ganzzahl): Ganzzahl
    // fib(0) = 0, fib(1) = 1, fib(n) = fib(n - 1) + fib(n - 2) fuer n >= 2.
    // BAUMREKURSIV: Im Rekursionsschritt ruft sich die Methode ZWEIMAL auf.
    // Erste Anweisung: aufrufe um 1 erhoehen (bei JEDEM Aufruf).
    // Beispiel: aufrufe = 0; fibonacci(5) liefert 5, danach ist aufrufe = 15.
    // REKURSIV - ohne Schleife.
    public static int fibonacci(int n) {
        aufrufe++;
        if (n < 2) {
            return n;                                   // Basisfall: fib(0) = 0, fib(1) = 1
        }
        return fibonacci(n - 1) + fibonacci(n - 2);     // Rekursionsschritt (zwei Aufrufe)
    }
}


// ============ Level 3.2 – Die Dungeon-Karte (DS 19 und 20) ============
//  Klassenkarte Karte (Notation wie im Abitur):
//
//      Karte
//      ---------------------------------------------------------------
//      - feld: Reihung von Reihung von Zeichen       // feld[z][s]
//      + schritte: Ganzzahl                          // Aufrufe von findeWeg
//      ---------------------------------------------------------------
//      + c Karte(zeilen: Reihung von Zeichenkette)
//      + getZeilen(): Ganzzahl
//      + getSpalten(): Ganzzahl
//      + getFeld(z: Ganzzahl, s: Ganzzahl): Zeichen
//      + setFeld(z: Ganzzahl, s: Ganzzahl, c: Zeichen): nichts
//      + alsText(): Zeichenkette
//      + istImFeld(z: Ganzzahl, s: Ganzzahl): Wahrheitswert              Basis
//      + istBegehbar(z: Ganzzahl, s: Ganzzahl): Wahrheitswert            Basis
//      + anzahlWaende(): Ganzzahl                                        Basis
//      + anzahlBegehbareNachbarn(z: Ganzzahl, s: Ganzzahl): Ganzzahl     Erweiterung
//      + lueften(z: Ganzzahl, s: Ganzzahl): nichts                       Erweiterung
//      + anzahlGelueftet(): Ganzzahl                                     Erweiterung
//      + findeWeg(z: Ganzzahl, s: Ganzzahl): Wahrheitswert               Experte
//
//  Die Zeichen auf der Karte:
//      '#'  Wand                    '+'  geluefteter (besuchter) Gang
//      '.'  freier Gang (Nebel)     '*'  Feld auf dem gefundenen Weg
//      'S'  Start                   'x'  Sackgasse (besucht, kein Weg)
//      'Z'  Ziel
//
//  Beispiel: Die Karte aus den Zeilen {"#S.#", "#..Z"} hat 2 Zeilen und
//  4 Spalten; feld[0][1] ist 'S', feld[1][3] ist 'Z'.
//
//      Spalte s:   0 1 2 3
//      Zeile 0:    # S . #
//      Zeile 1:    # . . Z
//
//  Nachbarn von (z, s) - IMMER in dieser Reihenfolge:
//      oben (z - 1, s), rechts (z, s + 1), unten (z + 1, s), links (z, s - 1)
// ---------------------------------------------------------------------

class Karte {

    // ======== VORGEGEBEN – bis zur Linie "AB HIER ARBEITEST DU" nichts veraendern ========

    private char[][] feld;      // feld[z][s]: Zeile z, Spalte s
    public int schritte;        // zaehlt die Aufrufe von findeWeg (Experte)

    // Baut die Karte aus Textzeilen: feld[z][s] = zeilen[z].charAt(s).
    // Alle Zeilen muessen gleich lang sein.
    public Karte(String[] zeilen) {
        feld = new char[zeilen.length][zeilen[0].length()];
        for (int z = 0; z < zeilen.length; z++) {
            for (int s = 0; s < zeilen[z].length(); s++) {
                feld[z][s] = zeilen[z].charAt(s);
            }
        }
        schritte = 0;
    }

    public int getZeilen() {
        return feld.length;
    }

    public int getSpalten() {
        return feld[0].length;
    }

    public char getFeld(int z, int s) {
        return feld[z][s];
    }

    public void setFeld(int z, int s, char c) {
        feld[z][s] = c;
    }

    // Die Karte als Text: Zeilen mit "\n" getrennt (ohne "\n" am Ende).
    // println(karte.alsText()) zeichnet die Karte auf die Konsole.
    public String alsText() {
        String text = "";
        for (int z = 0; z < feld.length; z++) {
            if (z > 0) {
                text = text + "\n";
            }
            for (int s = 0; s < feld[z].length; s++) {
                text = text + feld[z][s];
            }
        }
        return text;
    }

    // ======== AB HIER ARBEITEST DU =============================================


    // ===================== Stufe BASIS =====================

    // istImFeld(z: Ganzzahl, s: Ganzzahl): Wahrheitswert
    // true, wenn (z, s) ein gueltiges Feld der Karte ist:
    // 0 <= z < Anzahl Zeilen und 0 <= s < Anzahl Spalten.
    public boolean istImFeld(int z, int s) {
        return z >= 0 && z < feld.length && s >= 0 && s < feld[0].length;
    }

    // istBegehbar(z: Ganzzahl, s: Ganzzahl): Wahrheitswert
    // true, wenn (z, s) im Feld liegt UND dort '.', 'S' oder 'Z' steht.
    // Ausserhalb der Karte -> false (KEIN Absturz!).
    public boolean istBegehbar(int z, int s) {
        if (!istImFeld(z, s)) {
            return false;                               // zuerst den Rand pruefen!
        }
        char c = feld[z][s];
        return c == '.' || c == 'S' || c == 'Z';
    }

    // anzahlWaende(): Ganzzahl
    // Anzahl der Felder mit '#' auf der ganzen Karte.
    // (Hier sind Schleifen erlaubt: zwei geschachtelte Zaehlschleifen.)
    public int anzahlWaende() {
        int anzahl = 0;
        for (int z = 0; z < feld.length; z++) {
            for (int s = 0; s < feld[z].length; s++) {
                if (feld[z][s] == '#') {
                    anzahl++;
                }
            }
        }
        return anzahl;
    }


    // ================== Stufe ERWEITERUNG ==================

    // anzahlBegehbareNachbarn(z: Ganzzahl, s: Ganzzahl): Ganzzahl
    // Wie viele der vier Nachbarn (oben, rechts, unten, links) von (z, s)
    // sind begehbar? Diagonale Felder zaehlen NICHT. Am Rand fehlen
    // Nachbarn - nutze istBegehbar, dann gibt es keinen Absturz.
    public int anzahlBegehbareNachbarn(int z, int s) {
        int anzahl = 0;
        if (istBegehbar(z - 1, s)) {                    // oben
            anzahl++;
        }
        if (istBegehbar(z, s + 1)) {                    // rechts
            anzahl++;
        }
        if (istBegehbar(z + 1, s)) {                    // unten
            anzahl++;
        }
        if (istBegehbar(z, s - 1)) {                    // links
            anzahl++;
        }
        return anzahl;
    }

    // lueften(z: Ganzzahl, s: Ganzzahl): nichts
    // Nebel lueften (Flood-Fill): Ist (z, s) begehbar ('.', 'S' oder 'Z'),
    // wird das Feld als besucht markiert und danach werden die vier
    // Nachbarn gelueftet (oben, rechts, unten, links). Sonst passiert
    // nichts (Basisfall: Wand, Rand, schon besucht).
    // Ergebnis: Alle von (z, s) aus erreichbaren '.' sind '+'.
    // 'S' und 'Z' sind am Ende wieder als 'S' und 'Z' zu sehen.
    // ACHTUNG Endlosrekursion: Markiere ein Feld, BEVOR du die Nachbarn
    // lueftest - sonst laufen zwei Nachbarn ewig hin und her. Auch S und Z
    // muessen waehrenddessen als besucht gelten (z. B. voruebergehend '+'
    // setzen und am Ende den alten Buchstaben zuruecksetzen).
    // REKURSIV - ohne Schleife.
    public void lueften(int z, int s) {
        if (!istBegehbar(z, s)) {
            return;                                     // Basisfall: Rand, Wand oder schon besucht
        }
        char alt = feld[z][s];
        feld[z][s] = '+';                               // ZUERST als besucht markieren
        lueften(z - 1, s);                              // oben
        lueften(z, s + 1);                              // rechts
        lueften(z + 1, s);                              // unten
        lueften(z, s - 1);                              // links
        if (alt == 'S' || alt == 'Z') {
            feld[z][s] = alt;                           // S und Z wieder sichtbar machen
        }
    }

    // anzahlGelueftet(): Ganzzahl
    // Anzahl der Felder mit '+' auf der ganzen Karte (Schleifen erlaubt).
    public int anzahlGelueftet() {
        int anzahl = 0;
        for (int z = 0; z < feld.length; z++) {
            for (int s = 0; s < feld[z].length; s++) {
                if (feld[z][s] == '+') {
                    anzahl++;
                }
            }
        }
        return anzahl;
    }


    // =================== Stufe EXPERTE – Kuer-DS 20 ===================

    // findeWeg(z: Ganzzahl, s: Ganzzahl): Wahrheitswert
    // BACKTRACKING: Sucht von (z, s) aus einen Weg zum Ziel 'Z'
    // (Nachbarn oben, rechts, unten, links). true, wenn Z erreichbar ist.
    //   1. schritte um 1 erhoehen (bei JEDEM Aufruf, siehe Zaehl-Regeln).
    //   2. Basisfaelle: ausserhalb der Karte -> false; Feld ist 'Z' -> true;
    //      Feld ist weder '.' noch 'S' (Wand oder schon besucht) -> false.
    //   3. Feld vorlaeufig als Weg markieren ('*'), dann nacheinander die
    //      Nachbarn versuchen. Liefert ein Nachbar true -> sofort true
    //      zurueckgeben (das Feld bleibt '*').
    //   4. Kein Nachbar fuehrt zum Ziel: Schritt zuruecknehmen - das Feld
    //      wird zur Sackgasse 'x' - und false zurueckgeben.
    // Am Ende: Weg-Felder '*', besuchte Sackgassen 'x', nie besuchte
    // Felder bleiben '.', und S und Z sind wieder 'S' und 'Z'.
    // Gibt es keinen Weg: false, alle besuchten '.' sind 'x'.
    // REKURSIV - ohne Schleife.
    public boolean findeWeg(int z, int s) {
        schritte++;
        if (!istImFeld(z, s)) {
            return false;                               // Basisfall: ausserhalb
        }
        char alt = feld[z][s];
        if (alt == 'Z') {
            return true;                                // Basisfall: Ziel gefunden
        }
        if (alt != '.' && alt != 'S') {
            return false;                               // Basisfall: Wand oder schon besucht
        }
        feld[z][s] = '*';                               // vorlaeufig auf den Weg setzen
        if (findeWeg(z - 1, s) || findeWeg(z, s + 1) || findeWeg(z + 1, s) || findeWeg(z, s - 1)) {
            if (alt == 'S') {
                feld[z][s] = 'S';                       // Start bleibt sichtbar
            }
            return true;
        }
        feld[z][s] = 'x';                               // Schritt zuruecknehmen: Sackgasse
        if (alt == 'S') {
            feld[z][s] = 'S';
        }
        return false;
    }
}


// ========= Level 3.3 – Teile und herrsche (DS 21 und 22) ==============
//  Teile und herrsche: Problem in Teilprobleme derselben Art zerlegen
//  (TEILEN), die Teile rekursiv loesen (HERRSCHEN) und die Teilloesungen
//  zusammensetzen (VERBINDEN). Alle Methoden sind static.
//  Reihungsabschnitte werden mit Indexgrenzen beschrieben:
//      von, bis bzw. lo, hi        -> beide Grenzen gehoeren DAZU
//      teilstueck(r, von, bis)     -> bis gehoert NICHT dazu (exklusiv)
// ---------------------------------------------------------------------

class TeileUndHerrsche {

    // VORGEGEBEN: zaehlt die Vergleiche r[i] < pivot in zerlege.
    public static int vergleiche = 0;

    // ===================== Stufe BASIS =====================

    // binaerRek(r: Reihung von Ganzzahl, x: Ganzzahl, lo: Ganzzahl, hi: Ganzzahl): Ganzzahl
    // Binaere Suche nach x im Bereich r[lo] ... r[hi] (beide einschliesslich),
    // r ist AUFSTEIGEND sortiert. Liefert den Index von x oder -1.
    // Basisfall: lo > hi (Bereich leer) -> -1. Sonst mid = (lo + hi) / 2;
    // r[mid] == x -> mid; sonst rekursiv NUR in der passenden Haelfte
    // weitersuchen (ohne mid).
    // REKURSIV - ohne Schleife.
    public static int binaerRek(int[] r, int x, int lo, int hi) {
        if (lo > hi) {
            return -1;                                  // Basisfall: Bereich leer
        }
        int mid = (lo + hi) / 2;
        if (r[mid] == x) {
            return mid;                                 // Basisfall: gefunden
        }
        if (r[mid] < x) {
            return binaerRek(r, x, mid + 1, hi);        // rechts weitersuchen
        }
        return binaerRek(r, x, lo, mid - 1);            // links weitersuchen
    }

    // binaer(r: Reihung von Ganzzahl, x: Ganzzahl): Ganzzahl
    // Sucht x in der ganzen Reihung: ruft binaerRek(r, x, 0, r.length - 1)
    // auf und liefert dessen Ergebnis.
    public static int binaer(int[] r, int x) {
        return binaerRek(r, x, 0, r.length - 1);
    }


    // ================== Stufe ERWEITERUNG ==================

    // maximumTH(r: Reihung von Ganzzahl, von: Ganzzahl, bis: Ganzzahl): Ganzzahl
    // Groesster Wert im Bereich r[von] ... r[bis] (von <= bis) nach
    // "Teile und herrsche":
    //   Basisfall: von == bis -> r[von].
    //   Sonst: mitte = (von + bis) / 2, Maximum der linken Haelfte
    //   von ... mitte und der rechten Haelfte mitte + 1 ... bis rekursiv
    //   bestimmen und den groesseren der beiden Werte zurueckgeben.
    // REKURSIV - ohne Schleife.
    public static int maximumTH(int[] r, int von, int bis) {
        if (von == bis) {
            return r[von];                              // Basisfall: ein Element
        }
        int mitte = (von + bis) / 2;                    // TEILEN
        int links = maximumTH(r, von, mitte);           // HERRSCHEN
        int rechts = maximumTH(r, mitte + 1, bis);
        if (links > rechts) {                           // VERBINDEN
            return links;
        }
        return rechts;
    }

    // teilstueck(r: Reihung von Ganzzahl, von: Ganzzahl, bis: Ganzzahl): Reihung von Ganzzahl
    // NEUE Reihung mit den Elementen r[von], r[von + 1], ..., r[bis - 1].
    // ACHTUNG: bis gehoert NICHT dazu, die neue Reihung hat also die
    // Laenge bis - von. von == bis -> leere Reihung. r bleibt unveraendert.
    // (Schleife erlaubt.)
    public static int[] teilstueck(int[] r, int von, int bis) {
        int[] t = new int[bis - von];
        for (int i = 0; i < t.length; i++) {
            t[i] = r[von + i];
        }
        return t;
    }

    // verschmelze(a: Reihung von Ganzzahl, b: Reihung von Ganzzahl): Reihung von Ganzzahl
    // a und b sind AUFSTEIGEND sortiert. Liefert eine NEUE aufsteigend
    // sortierte Reihung mit allen Elementen von a und b: Immer das
    // kleinere der beiden vordersten Elemente uebernehmen (bei Gleichstand
    // das aus a). Ist eine Reihung aufgebraucht, den Rest der anderen
    // anhaengen. (Schleifen erlaubt.)
    public static int[] verschmelze(int[] a, int[] b) {
        int[] erg = new int[a.length + b.length];
        int i = 0;
        int j = 0;
        int k = 0;
        while (i < a.length && j < b.length) {
            if (a[i] <= b[j]) {
                erg[k] = a[i];
                i++;
            } else {
                erg[k] = b[j];
                j++;
            }
            k++;
        }
        while (i < a.length) {                          // Rest von a
            erg[k] = a[i];
            i++;
            k++;
        }
        while (j < b.length) {                          // Rest von b
            erg[k] = b[j];
            j++;
            k++;
        }
        return erg;
    }

    // mergeSort(r: Reihung von Ganzzahl): Reihung von Ganzzahl
    // Liefert eine NEUE aufsteigend sortierte Reihung; r bleibt unveraendert.
    //   Basisfall: r.length <= 1 -> eine KOPIE von r zurueckgeben.
    //   Sonst: mitte = r.length / 2; links = teilstueck(r, 0, mitte),
    //   rechts = teilstueck(r, mitte, r.length); beide Teile rekursiv
    //   sortieren und mit verschmelze zusammenfuegen.
    // REKURSIV - ohne Schleife (die Schleifen stecken in teilstueck und
    // verschmelze).
    public static int[] mergeSort(int[] r) {
        if (r.length <= 1) {
            return teilstueck(r, 0, r.length);          // Basisfall: Kopie
        }
        int mitte = r.length / 2;                       // TEILEN
        int[] links = mergeSort(teilstueck(r, 0, mitte));          // HERRSCHEN
        int[] rechts = mergeSort(teilstueck(r, mitte, r.length));
        return verschmelze(links, rechts);              // VERBINDEN
    }


    // =================== Stufe EXPERTE =====================

    // quickSort(r: Reihung von Ganzzahl): nichts
    // Sortiert r AUFSTEIGEND in-place (r selbst wird veraendert):
    // ruft quickSort(r, 0, r.length - 1) auf.
    public static void quickSort(int[] r) {
        quickSort(r, 0, r.length - 1);
    }

    // quickSort(r: Reihung von Ganzzahl, lo: Ganzzahl, hi: Ganzzahl): nichts
    // Sortiert den Bereich r[lo] ... r[hi]:
    //   Nur falls lo < hi (sonst Basisfall - nichts zu tun):
    //   p = zerlege(r, lo, hi); dann quickSort(r, lo, p - 1) und
    //   quickSort(r, p + 1, hi). Das Pivot an Position p steht schon richtig.
    // REKURSIV - ohne Schleife.
    public static void quickSort(int[] r, int lo, int hi) {
        if (lo < hi) {
            int p = zerlege(r, lo, hi);                 // TEILEN
            quickSort(r, lo, p - 1);                    // HERRSCHEN links
            quickSort(r, p + 1, hi);                    // HERRSCHEN rechts
        }
    }

    // zerlege(r: Reihung von Ganzzahl, lo: Ganzzahl, hi: Ganzzahl): Ganzzahl
    // Zerlegt r[lo] ... r[hi] um das Pivot r[lo] (das LINKE Element):
    //   pivot <- r[lo]; grenze <- lo
    //   fuer i von lo + 1 bis hi:
    //       vergleiche um 1 erhoehen
    //       falls r[i] < pivot: grenze <- grenze + 1; tausche r[grenze] und r[i]
    //   tausche r[lo] und r[grenze]; gib grenze zurueck
    // Danach stehen links vom Pivot nur kleinere, rechts nur groessere oder
    // gleiche Werte. (Schleife erlaubt.)
    public static int zerlege(int[] r, int lo, int hi) {
        int pivot = r[lo];
        int grenze = lo;
        for (int i = lo + 1; i <= hi; i++) {
            vergleiche++;
            if (r[i] < pivot) {
                grenze++;
                int h = r[grenze];
                r[grenze] = r[i];
                r[i] = h;
            }
        }
        int hilf = r[lo];
        r[lo] = r[grenze];
        r[grenze] = hilf;
        return grenze;
    }
}
