//              liest sie aus, OHNE sie zu veraendern. Die Tests zeigen
//              dir damit in der Meldung den ganzen Inhalt, z. B.
//              "[erwartet: [Erz (5 kg), Holz (3 kg)], tatsaechlich: ...]".
//              Stapel werden von OBEN nach unten, Schlangen von VORNE
//              nach hinten angezeigt. Ausserdem: die Standard-Ladung
//              (sechs Kisten) und die Meldung bei Laufzeitfehlern.

//  Klassenkarte (Notation wie im Abitur):
//      Kiste
//      ------------------------------------------------
//      - name: Zeichenkette
//      - gewicht: Ganzzahl                     // in kg
//      ------------------------------------------------
//      + c Kiste(name: Zeichenkette, gewicht: Ganzzahl)
//      + getName(): Zeichenkette
//      + getGewicht(): Ganzzahl
//      + beschreibung(): Zeichenkette          // z. B. "Erz (5 kg)"

class Kiste {

    private String name;
    private int gewicht;

    public Kiste(String name, int gewicht) {
        this.name = name;
        this.gewicht = gewicht;
    }

    public String getName() {
        return name;
    }

    public int getGewicht() {
        return gewicht;
    }

    public String beschreibung() {
        return name + " (" + gewicht + " kg)";
    }
}


class Testhilfe {

    // ----- Strukturen bauen -----

    // Stapel aus Ganzzahlen: werte[0] zuerst, der letzte Wert liegt oben.
    public static Stack zahlenStapel(int[] werte) {
        Stack s = new Stack();
        for (int i = 0; i < werte.length; i++) {
            s.push(werte[i]);
        }
        return s;
    }

    // Schlange aus Ganzzahlen: werte[0] steht vorne.
    public static Queue zahlenSchlange(int[] werte) {
        Queue q = new Queue();
        for (int i = 0; i < werte.length; i++) {
            q.enqueue(werte[i]);
        }
        return q;
    }

    // Schlange aus Zeichenketten: namen[0] steht vorne.
    public static Queue namenSchlange(String[] namen) {
        Queue q = new Queue();
        for (int i = 0; i < namen.length; i++) {
            q.enqueue(namen[i]);
        }
        return q;
    }

    // Stapel aus Kisten: kisten[0] liegt unten, die letzte Kiste oben.
    public static Stack kistenStapel(Kiste[] kisten) {
        Stack s = new Stack();
        for (int i = 0; i < kisten.length; i++) {
            s.push(kisten[i]);
        }
        return s;
    }

    // Schlange aus Kisten: kisten[0] steht vorne.
    public static Queue kistenSchlange(Kiste[] kisten) {
        Queue q = new Queue();
        for (int i = 0; i < kisten.length; i++) {
            q.enqueue(kisten[i]);
        }
        return q;
    }

    // Dynamische Reihung aus Kisten in derselben Reihenfolge.
    public static DynArray kistenReihung(Kiste[] kisten) {
        DynArray d = new DynArray();
        for (int i = 0; i < kisten.length; i++) {
            d.append(kisten[i]);
        }
        return d;
    }

    // ----- Inhalt auslesen, OHNE die Struktur zu veraendern -----

    // Inhalt eines Stapels von oben nach unten. s ist danach wieder genau
    // wie vorher (dieselben Objekte, dieselbe Reihenfolge).
    public static DynArray stapelInhalt(Stack s) {
        if (s == null) {
            return null;
        }
        DynArray d = new DynArray();
        Stack hilf = new Stack();
        while (!s.isEmpty()) {
            Object o = s.pop();
            d.append(o);
            hilf.push(o);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return d;
    }

    // Inhalt einer Schlange von vorne nach hinten. q ist danach wieder
    // genau wie vorher.
    public static DynArray schlangeInhalt(Queue q) {
        if (q == null) {
            return null;
        }
        DynArray d = new DynArray();
        while (!q.isEmpty()) {
            d.append(q.dequeue());
        }
        for (int i = 0; i < d.getLength(); i++) {
            q.enqueue(d.getItem(i));
        }
        return d;
    }

    // ----- Texte fuer die Meldungen -----

    // Zahlen oder Zeichenketten als Text, z. B. "[4, 8, 15]".
    public static String text(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + d.getItem(i);
        }
        return t + "]";
    }

    // Kisten als Text, z. B. "[Erz (5 kg), Holz (3 kg)]".
    public static String kistenText(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            Kiste k = (Kiste) d.getItem(i);
            t = t + k.beschreibung();
        }
        return t + "]";
    }

    public static String stapelText(Stack s) {
        return text(stapelInhalt(s));
    }

    public static String schlangeText(Queue q) {
        return text(schlangeInhalt(q));
    }

    public static String kistenStapelText(Stack s) {
        return kistenText(stapelInhalt(s));
    }

    public static String kistenSchlangeText(Queue q) {
        return kistenText(schlangeInhalt(q));
    }

    // ----- Vergleich: dieselben Objekte in derselben Reihenfolge? -----

    public static boolean dieselben(DynArray a, DynArray b) {
        if (a == null || b == null) {
            return false;
        }
        if (a.getLength() != b.getLength()) {
            return false;
        }
        for (int i = 0; i < a.getLength(); i++) {
            if (a.getItem(i) != b.getItem(i)) {
                return false;
            }
        }
        return true;
    }

    // Eine Kiste als Text (auch null), z. B. "Erz (5 kg)".
    public static String kisteText(Kiste k) {
        if (k == null) {
            return "null";
        }
        return k.beschreibung();
    }

    // ----- Die Standard-Ladung des Hafens (fuer Tests, AB und Hauptprogramm) -----

    // Liefert bei JEDEM Aufruf sechs NEUE Kisten:
    //     Holz (3 kg), Erz (5 kg), Gold (8 kg), Salz (3 kg), Tuch (8 kg), Wein (6 kg)
    // Gesamtgewicht 33 kg, Durchschnitt 5,5 kg.
    public static Kiste[] ladung() {
        Kiste[] k = new Kiste[6];
        k[0] = new Kiste("Holz", 3);
        k[1] = new Kiste("Erz", 5);
        k[2] = new Kiste("Gold", 8);
        k[3] = new Kiste("Salz", 3);
        k[4] = new Kiste("Tuch", 8);
        k[5] = new Kiste("Wein", 6);
        return k;
    }

    // ----- Laufzeitfehler in einem Test verstaendlich melden -----

    // Die Tests fangen Laufzeitfehler (Exceptions) ab und melden sie mit
    // einem Hinweis. In einer richtigen Loesung tritt NIE ein Laufzeitfehler
    // auf - auch nicht bei leeren Strukturen!
    public static void laufzeitfehler(RuntimeException e) {
        String meldung = " (Meldung der IDE: " + e.getMessage() + ")";
        try {
            throw e;
        } catch (IllegalStateException x) {
            fail("LAUFZEITFEHLER: top, pop, head oder dequeue auf einer LEEREN Struktur - Leerabfrage mit isEmpty() vergessen?" + meldung);
        } catch (IndexOutOfBoundsException x) {
            fail("LAUFZEITFEHLER: Index ausserhalb - bei DynArray gilt 0 <= i < getLength() (insertAt: i <= getLength()). Schleifengrenze pruefen! Geloescht und danach i erhoeht?" + meldung);
        } catch (IllegalArgumentException x) {
            fail("LAUFZEITFEHLER: null in Stack, Queue oder DynArray gelegt - push, enqueue, append, insertAt und setItem verbieten null." + meldung);
        } catch (ClassCastException x) {
            fail("LAUFZEITFEHLER: falscher Cast - passt (Kiste), (Integer), (String) bzw. (Character) zum Inhaltstyp der Struktur?" + meldung);
        } catch (NullPointerException x) {
            fail("LAUFZEITFEHLER: Zugriff auf null - Attribut im Konstruktor nicht erzeugt (new Stack() ...)? Liefert eine Methode noch null?" + meldung);
        } catch (RuntimeException x) {
            fail("LAUFZEITFEHLER" + meldung);
        }
    }

    // ----- Sonstiges -----

    // Neue Zeichenkette mit demselben Inhalt wie s (Zeichen fuer Zeichen
    // zusammengesetzt). Damit pruefen die Tests, dass du Zeichenketten mit
    // equals vergleichst und nicht mit ==.
    public static String kopie(String s) {
        String t = "";
        for (int i = 0; i < s.length(); i++) {
            t = t + s.charAt(i);
        }
        return t;
    }

    // Pseudo-Zufallszahl (linearer Kongruenzgenerator): liefert aus x
    // die naechste Zahl zwischen 0 und 65536 - immer dieselbe Folge.
    public static int naechsteZufallszahl(int x) {
        return (x * 75 + 74) % 65537;
    }

    // 2D-Reihung als Text: Zeilen mit "\n" getrennt.
    public static String kartenText(char[][] karte) {
        if (karte == null) {
            return "null";
        }
        String t = "";
        for (int z = 0; z < karte.length; z++) {
            if (z > 0) {
                t = t + "\n";
            }
            for (int s = 0; s < karte[z].length; s++) {
                t = t + karte[z][s];
            }
        }
        return t;
    }

    // Zeilen als Text (wie kartenText), fuer erwartete Karten.
    public static String zeilenText(String[] zeilen) {
        String t = "";
        for (int i = 0; i < zeilen.length; i++) {
            if (i > 0) {
                t = t + "\n";
            }
            t = t + zeilen[i];
        }
        return t;
    }
}


// Die Testkarten aus Welt 3 als Reihung von Reihung von Zeichen:
// karte[z][s] = Zeile z, Spalte s. '#' Wand, '.' frei, 'S' Start, 'Z' Ziel.
// Jeder Aufruf liefert eine NEUE Karte (frisch vernebelt).
class Kartenbuch {

    // Baut aus Textzeilen eine 2D-Reihung: karte[z][s] = zeilen[z].charAt(s).
    public static char[][] karte(String[] zeilen) {
        char[][] k = new char[zeilen.length][zeilen[0].length()];
        for (int z = 0; z < zeilen.length; z++) {
            for (int s = 0; s < zeilen[z].length(); s++) {
                k[z][s] = zeilen[z].charAt(s);
            }
        }
        return k;
    }

    // Die offene Hoehle (5 x 7), Start (1, 1), Ziel (3, 5).
    public static char[][] hoehle() {
        String[] zeilen = {
            "#######",
            "#S....#",
            "#..#..#",
            "#....Z#",
            "#######"
        };
        return karte(zeilen);
    }

    // Das Labyrinth (9 x 9), Start (1, 1), Ziel (5, 7); die Kammer unten
    // rechts ist nur UEBER ECK mit dem Gang verbunden.
    public static char[][] labyrinth() {
        String[] zeilen = {
            "#########",
            "#S#...#.#",
            "#.#.#.#.#",
            "#...#...#",
            "#.###.#.#",
            "#...#.#Z#",
            "###.##.##",
            "#...#...#",
            "#########"
        };
        return karte(zeilen);
    }

    // Der Kerker (6 x 8), Start (1, 1), Ziel (2, 5) in einer eigenen Kammer.
    public static char[][] kerker() {
        String[] zeilen = {
            "########",
            "#S..#..#",
            "#.#.#Z.#",
            "#...####",
            "#.#....#",
            "########"
        };
        return karte(zeilen);
    }

    // Die Lichtung (3 x 4) OHNE Randmauer, Start (1, 1), Ziel (2, 3).
    public static char[][] lichtung() {
        String[] zeilen = {
            "..#.",
            ".S..",
            "#..Z"
        };
        return karte(zeilen);
    }
}
