// ---------------------------------------------------------------------
//  GERÜST – unveraendert
// ---------------------------------------------------------------------
//  Kiste:      die Fracht im Hafen (wird in allen Leveln benutzt).
//  Testhilfe:  baut Stapel/Schlangen/Reihungen fuer die Tests und
//              liest sie aus, OHNE sie zu veraendern. Die Tests zeigen
//              dir damit in der Meldung den ganzen Inhalt, z. B.
//              "[erwartet: [Erz (5 kg), Holz (3 kg)], tatsaechlich: ...]".
//              Stapel werden von OBEN nach unten, Schlangen von VORNE
//              nach hinten angezeigt. Ausserdem: die Standard-Ladung
//              (sechs Kisten) und die Meldung bei Laufzeitfehlern.
//  Kartenbuch: die Dungeon-Karten aus Welt 3 als 2D-Reihungen (Level 4.6).
//  Du darfst alle drei Klassen auch im Hauptprogramm benutzen.
// ---------------------------------------------------------------------

//  Klassenkarte (Notation wie im Abitur):
//      Kiste
//      ------------------------------------------------
//      - name: Zeichenkette
//      - gewicht: Ganzzahl                     // in kg
//      ------------------------------------------------
//      + c Kiste(name: Zeichenkette, gewicht: Ganzzahl)
//      + getName(): Zeichenkette
//      + getGewicht(): Ganzzahl
//      + beschreibung(): Zeichenkette          // z. B. "Erz (5 kg)"

class Kiste {

    private String name;
    private int gewicht;

    public Kiste(String name, int gewicht) {
        this.name = name;
        this.gewicht = gewicht;
    }

    public String getName() {
        return name;
    }

    public int getGewicht() {
        return gewicht;
    }

    public String beschreibung() {
        return name + " (" + gewicht + " kg)";
    }
}


class Testhilfe {

    // ----- Strukturen bauen -----

    // Stapel aus Ganzzahlen: werte[0] zuerst, der letzte Wert liegt oben.
    public static Stack zahlenStapel(int[] werte) {
        Stack s = new Stack();
        for (int i = 0; i < werte.length; i++) {
            s.push(werte[i]);
        }
        return s;
    }

    // Schlange aus Ganzzahlen: werte[0] steht vorne.
    public static Queue zahlenSchlange(int[] werte) {
        Queue q = new Queue();
        for (int i = 0; i < werte.length; i++) {
            q.enqueue(werte[i]);
        }
        return q;
    }

    // Schlange aus Zeichenketten: namen[0] steht vorne.
    public static Queue namenSchlange(String[] namen) {
        Queue q = new Queue();
        for (int i = 0; i < namen.length; i++) {
            q.enqueue(namen[i]);
        }
        return q;
    }

    // Stapel aus Kisten: kisten[0] liegt unten, die letzte Kiste oben.
    public static Stack kistenStapel(Kiste[] kisten) {
        Stack s = new Stack();
        for (int i = 0; i < kisten.length; i++) {
            s.push(kisten[i]);
        }
        return s;
    }

    // Schlange aus Kisten: kisten[0] steht vorne.
    public static Queue kistenSchlange(Kiste[] kisten) {
        Queue q = new Queue();
        for (int i = 0; i < kisten.length; i++) {
            q.enqueue(kisten[i]);
        }
        return q;
    }

    // Dynamische Reihung aus Kisten in derselben Reihenfolge.
    public static DynArray kistenReihung(Kiste[] kisten) {
        DynArray d = new DynArray();
        for (int i = 0; i < kisten.length; i++) {
            d.append(kisten[i]);
        }
        return d;
    }

    // ----- Inhalt auslesen, OHNE die Struktur zu veraendern -----

    // Inhalt eines Stapels von oben nach unten. s ist danach wieder genau
    // wie vorher (dieselben Objekte, dieselbe Reihenfolge).
    public static DynArray stapelInhalt(Stack s) {
        if (s == null) {
            return null;
        }
        DynArray d = new DynArray();
        Stack hilf = new Stack();
        while (!s.isEmpty()) {
            Object o = s.pop();
            d.append(o);
            hilf.push(o);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return d;
    }

    // Inhalt einer Schlange von vorne nach hinten. q ist danach wieder
    // genau wie vorher.
    public static DynArray schlangeInhalt(Queue q) {
        if (q == null) {
            return null;
        }
        DynArray d = new DynArray();
        while (!q.isEmpty()) {
            d.append(q.dequeue());
        }
        for (int i = 0; i < d.getLength(); i++) {
            q.enqueue(d.getItem(i));
        }
        return d;
    }

    // ----- Texte fuer die Meldungen -----

    // Zahlen oder Zeichenketten als Text, z. B. "[4, 8, 15]".
    public static String text(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + d.getItem(i);
        }
        return t + "]";
    }

    // Kisten als Text, z. B. "[Erz (5 kg), Holz (3 kg)]".
    public static String kistenText(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            Kiste k = (Kiste) d.getItem(i);
            t = t + k.beschreibung();
        }
        return t + "]";
    }

    public static String stapelText(Stack s) {
        return text(stapelInhalt(s));
    }

    public static String schlangeText(Queue q) {
        return text(schlangeInhalt(q));
    }

    public static String kistenStapelText(Stack s) {
        return kistenText(stapelInhalt(s));
    }

    public static String kistenSchlangeText(Queue q) {
        return kistenText(schlangeInhalt(q));
    }

    // ----- Vergleich: dieselben Objekte in derselben Reihenfolge? -----

    public static boolean dieselben(DynArray a, DynArray b) {
        if (a == null || b == null) {
            return false;
        }
        if (a.getLength() != b.getLength()) {
            return false;
        }
        for (int i = 0; i < a.getLength(); i++) {
            if (a.getItem(i) != b.getItem(i)) {
                return false;
            }
        }
        return true;
    }

    // Eine Kiste als Text (auch null), z. B. "Erz (5 kg)".
    public static String kisteText(Kiste k) {
        if (k == null) {
            return "null";
        }
        return k.beschreibung();
    }

    // ----- Die Standard-Ladung des Hafens (fuer Tests, AB und Hauptprogramm) -----

    // Liefert bei JEDEM Aufruf sechs NEUE Kisten:
    //     Holz (3 kg), Erz (5 kg), Gold (8 kg), Salz (3 kg), Tuch (8 kg), Wein (6 kg)
    // Gesamtgewicht 33 kg, Durchschnitt 5,5 kg.
    public static Kiste[] ladung() {
        Kiste[] k = new Kiste[6];
        k[0] = new Kiste("Holz", 3);
        k[1] = new Kiste("Erz", 5);
        k[2] = new Kiste("Gold", 8);
        k[3] = new Kiste("Salz", 3);
        k[4] = new Kiste("Tuch", 8);
        k[5] = new Kiste("Wein", 6);
        return k;
    }

    // ----- Laufzeitfehler in einem Test verstaendlich melden -----

    // Die Tests fangen Laufzeitfehler (Exceptions) ab und melden sie mit
    // einem Hinweis. In einer richtigen Loesung tritt NIE ein Laufzeitfehler
    // auf - auch nicht bei leeren Strukturen!
    public static void laufzeitfehler(RuntimeException e) {
        String meldung = " (Meldung der IDE: " + e.getMessage() + ")";
        try {
            throw e;
        } catch (IllegalStateException x) {
            fail("LAUFZEITFEHLER: top, pop, head oder dequeue auf einer LEEREN Struktur - Leerabfrage mit isEmpty() vergessen?" + meldung);
        } catch (IndexOutOfBoundsException x) {
            fail("LAUFZEITFEHLER: Index ausserhalb - bei DynArray gilt 0 <= i < getLength() (insertAt: i <= getLength()). Schleifengrenze pruefen! Geloescht und danach i erhoeht?" + meldung);
        } catch (IllegalArgumentException x) {
            fail("LAUFZEITFEHLER: null in Stack, Queue oder DynArray gelegt - push, enqueue, append, insertAt und setItem verbieten null." + meldung);
        } catch (ClassCastException x) {
            fail("LAUFZEITFEHLER: falscher Cast - passt (Kiste), (Integer), (String) bzw. (Character) zum Inhaltstyp der Struktur?" + meldung);
        } catch (NullPointerException x) {
            fail("LAUFZEITFEHLER: Zugriff auf null - Attribut im Konstruktor nicht erzeugt (new Stack() ...)? Liefert eine Methode noch null?" + meldung);
        } catch (RuntimeException x) {
            fail("LAUFZEITFEHLER" + meldung);
        }
    }

    // ----- Sonstiges -----

    // Neue Zeichenkette mit demselben Inhalt wie s (Zeichen fuer Zeichen
    // zusammengesetzt). Damit pruefen die Tests, dass du Zeichenketten mit
    // equals vergleichst und nicht mit ==.
    public static String kopie(String s) {
        String t = "";
        for (int i = 0; i < s.length(); i++) {
            t = t + s.charAt(i);
        }
        return t;
    }

    // Pseudo-Zufallszahl (linearer Kongruenzgenerator): liefert aus x
    // die naechste Zahl zwischen 0 und 65536 - immer dieselbe Folge.
    public static int naechsteZufallszahl(int x) {
        return (x * 75 + 74) % 65537;
    }

    // 2D-Reihung als Text: Zeilen mit "\n" getrennt.
    public static String kartenText(char[][] karte) {
        if (karte == null) {
            return "null";
        }
        String t = "";
        for (int z = 0; z < karte.length; z++) {
            if (z > 0) {
                t = t + "\n";
            }
            for (int s = 0; s < karte[z].length; s++) {
                t = t + karte[z][s];
            }
        }
        return t;
    }

    // Zeilen als Text (wie kartenText), fuer erwartete Karten.
    public static String zeilenText(String[] zeilen) {
        String t = "";
        for (int i = 0; i < zeilen.length; i++) {
            if (i > 0) {
                t = t + "\n";
            }
            t = t + zeilen[i];
        }
        return t;
    }
}


// Die Testkarten aus Welt 3 als Reihung von Reihung von Zeichen:
// karte[z][s] = Zeile z, Spalte s. '#' Wand, '.' frei, 'S' Start, 'Z' Ziel.
// Jeder Aufruf liefert eine NEUE Karte (frisch vernebelt).
class Kartenbuch {

    // Baut aus Textzeilen eine 2D-Reihung: karte[z][s] = zeilen[z].charAt(s).
    public static char[][] karte(String[] zeilen) {
        char[][] k = new char[zeilen.length][zeilen[0].length()];
        for (int z = 0; z < zeilen.length; z++) {
            for (int s = 0; s < zeilen[z].length(); s++) {
                k[z][s] = zeilen[z].charAt(s);
            }
        }
        return k;
    }

    // Die offene Hoehle (5 x 7), Start (1, 1), Ziel (3, 5).
    public static char[][] hoehle() {
        String[] zeilen = {
            "#######",
            "#S....#",
            "#..#..#",
            "#....Z#",
            "#######"
        };
        return karte(zeilen);
    }

    // Das Labyrinth (9 x 9), Start (1, 1), Ziel (5, 7); die Kammer unten
    // rechts ist nur UEBER ECK mit dem Gang verbunden.
    public static char[][] labyrinth() {
        String[] zeilen = {
            "#########",
            "#S#...#.#",
            "#.#.#.#.#",
            "#...#...#",
            "#.###.#.#",
            "#...#.#Z#",
            "###.##.##",
            "#...#...#",
            "#########"
        };
        return karte(zeilen);
    }

    // Der Kerker (6 x 8), Start (1, 1), Ziel (2, 5) in einer eigenen Kammer.
    public static char[][] kerker() {
        String[] zeilen = {
            "########",
            "#S..#..#",
            "#.#.#Z.#",
            "#...####",
            "#.#....#",
            "########"
        };
        return karte(zeilen);
    }

    // Die Lichtung (3 x 4) OHNE Randmauer, Start (1, 1), Ziel (2, 3).
    public static char[][] lichtung() {
        String[] zeilen = {
            "..#.",
            ".S..",
            "#..Z"
        };
        return karte(zeilen);
    }
}

// ============= Level 4.1 – Stapel und Schlange (DS 24) ==============
//  Alle Methoden sind static: Aufruf mit Klassenname, z. B.
//  Hafen.leeren(s).

class Hafen {

    // ===================== Stufe BASIS =====================

    // stapleAuf(werte: Reihung von Ganzzahl): Stapel vom Inhaltstyp Ganzzahl
    // Erzeugt einen NEUEN Stapel und legt die Werte darauf: werte[0]
    // zuerst, der letzte Wert liegt oben. {4, 8, 15} -> oben 15, unten 4.
    public static Stack stapleAuf(int[] werte) {
        Stack s = new Stack();
        for (int i = 0; i < werte.length; i++) {
            s.push(werte[i]);                           // int wird automatisch verpackt
        }
        return s;
    }

    // obersterWert(s: Stapel vom Inhaltstyp Ganzzahl): Ganzzahl
    // Liefert den obersten Wert, OHNE ihn zu entfernen (top, nicht pop).
    // Leerer Stapel -> -1.
    public static int obersterWert(Stack s) {
        if (s.isEmpty()) {
            return -1;                                  // Leer? -> kein top()!
        }
        return (Integer) s.top();                       // Cast: Object -> Integer
    }

    // leeren(s: Stapel): Ganzzahl
    // Entnimmt ALLE Elemente und gibt zurueck, wie viele es waren.
    // Danach ist s leer. Leerer Stapel -> 0.
    public static int leeren(Stack s) {
        int anzahl = 0;
        while (!s.isEmpty()) {
            s.pop();
            anzahl++;
        }
        return anzahl;
    }


    // ================== Stufe ERWEITERUNG ==================

    // warteschlange(namen: Reihung von Zeichenkette): Schlange vom Inhaltstyp Zeichenkette
    // Erzeugt eine NEUE Schlange: namen[0] stellt sich zuerst an (steht
    // vorne), der letzte Name steht hinten.
    public static Queue warteschlange(String[] namen) {
        Queue q = new Queue();
        for (int i = 0; i < namen.length; i++) {
            q.enqueue(namen[i]);
        }
        return q;
    }

    // naechster(q: Schlange vom Inhaltstyp Zeichenkette): Zeichenkette
    // Entnimmt den vordersten Namen und gibt ihn zurueck.
    // Leere Schlange -> "".
    public static String naechster(Queue q) {
        if (q.isEmpty()) {
            return "";                                  // Leer? -> kein dequeue()!
        }
        return (String) q.dequeue();
    }

    // alleAbfertigen(q: Schlange vom Inhaltstyp Zeichenkette): Zeichenkette
    // Entnimmt ALLE Namen der Reihe nach und liefert sie, mit ", "
    // verbunden: "Albatros, Moewe, Kormoran". Leere Schlange -> "".
    // Danach ist q leer.
    public static String alleAbfertigen(Queue q) {
        String text = "";
        boolean erster = true;
        while (!q.isEmpty()) {
            if (!erster) {
                text = text + ", ";                     // Trenner nur ZWISCHEN den Namen
            }
            text = text + (String) q.dequeue();
            erster = false;
        }
        return text;
    }


    // =================== Stufe EXPERTE =====================

    // umkehren(q: Schlange): nichts
    // Kehrt die Reihenfolge der Schlange um - mit Hilfe eines Stapels.
    // Vorne A, B, C hinten -> vorne C, B, A hinten.
    public static void umkehren(Queue q) {
        Stack hilf = new Stack();
        while (!q.isEmpty()) {
            hilf.push(q.dequeue());                     // A, B, C -> oben liegt C
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.pop());                      // C zuerst wieder anstellen
        }
    }

    // stapelUmdrehen(s: Stapel): Stapel
    // Liefert einen NEUEN Stapel mit umgekehrter Reihenfolge: Was in s
    // unten lag, liegt im neuen Stapel oben. s ist danach leer.
    public static Stack stapelUmdrehen(Stack s) {
        Stack neu = new Stack();
        while (!s.isEmpty()) {
            neu.push(s.pop());
        }
        return neu;
    }
}


// ============ Level 4.2 – Spielmodule (DS 25, Kuer-DS) ==============
//  Zwei Objekt-Klassen (KEINE static-Methoden): Erst ein Objekt
//  erzeugen, dann die Methoden daran aufrufen, z. B.
//      Zugbuch zb = new Zugbuch();   zb.zug("Nord");
//
//  Klassenkarten:
//      Zugbuch                                   Initiative
//      -----------------------------------       -----------------------------------
//      - erledigt: Stapel vom Inhaltstyp         - reihe: Schlange vom Inhaltstyp
//          Zeichenkette                              Zeichenkette
//      - zurueckgenommen: Stapel vom             -----------------------------------
//          Inhaltstyp Zeichenkette               + c Initiative()
//      -----------------------------------       + beitreten(name: Zeichenkette): nichts
//      + c Zugbuch()                             + amZug(): Zeichenkette
//      + zug(z: Zeichenkette): nichts            + ausscheiden(name: Zeichenkette):
//      + rueckgaengig(): Zeichenkette                Wahrheitswert
//      + wiederholen(): Zeichenkette
//      + anzahlZuege(): Ganzzahl

class Zugbuch {

    private Stack erledigt;         // ausgefuehrte Zuege, der letzte liegt oben
    private Stack zurueckgenommen;  // zurueckgenommene Zuege (fuer wiederholen)

    // ===================== Stufe BASIS =====================

    // c Zugbuch()
    // Erzeugt die beiden (leeren) Stapel.
    public Zugbuch() {
        erledigt = new Stack();
        zurueckgenommen = new Stack();
    }

    // zug(z: Zeichenkette): nichts
    // Ein neuer Zug wird ausgefuehrt: z kommt oben auf erledigt. Nach einem
    // neuen Zug kann man nichts mehr wiederholen: der Stapel zurueckgenommen
    // wird geleert.
    public void zug(String z) {
        erledigt.push(z);
        zurueckgenommen = new Stack();                  // Redo-Stapel leeren
    }

    // rueckgaengig(): Zeichenkette
    // Nimmt den letzten Zug zurueck: oberster Zug von erledigt nach
    // zurueckgenommen; Rueckgabe dieser Zug. Nichts mehr da -> "".
    public String rueckgaengig() {
        if (erledigt.isEmpty()) {
            return "";                                  // Leer? -> kein pop()!
        }
        String z = (String) erledigt.pop();
        zurueckgenommen.push(z);
        return z;
    }


    // ================== Stufe ERWEITERUNG ==================

    // wiederholen(): Zeichenkette
    // Fuehrt den zuletzt zurueckgenommenen Zug wieder aus: oberster Zug
    // von zurueckgenommen zurueck nach erledigt; Rueckgabe dieser Zug.
    // Nichts zum Wiederholen da -> "".
    public String wiederholen() {
        if (zurueckgenommen.isEmpty()) {
            return "";
        }
        String z = (String) zurueckgenommen.pop();
        erledigt.push(z);
        return z;
    }

    // anzahlZuege(): Ganzzahl
    // Anzahl der Zuege in erledigt. UNVERAENDERT: erledigt sieht danach
    // genau so aus wie vorher (Hilfsstapel).
    public int anzahlZuege() {
        Stack hilf = new Stack();
        int anzahl = 0;
        while (!erledigt.isEmpty()) {
            hilf.push(erledigt.pop());
            anzahl++;
        }
        while (!hilf.isEmpty()) {
            erledigt.push(hilf.pop());                  // zurueckschichten
        }
        return anzahl;
    }
}


class Initiative {

    private Queue reihe;            // vorne steht, wer als Naechstes am Zug ist

    // ===================== Stufe BASIS =====================

    // c Initiative()
    // Erzeugt die (leere) Schlange.
    public Initiative() {
        reihe = new Queue();
    }

    // beitreten(name: Zeichenkette): nichts
    // name stellt sich hinten an.
    public void beitreten(String name) {
        reihe.enqueue(name);
    }

    // amZug(): Zeichenkette
    // Der Vorderste ist am Zug: Er wird vorne entnommen und sofort hinten
    // wieder angestellt (rotieren). Rueckgabe: sein Name. Leer -> "".
    public String amZug() {
        if (reihe.isEmpty()) {
            return "";
        }
        String name = (String) reihe.dequeue();
        reihe.enqueue(name);                            // hinten wieder anstellen
        return name;
    }


    // =================== Stufe EXPERTE =====================

    // ausscheiden(name: Zeichenkette): Wahrheitswert
    // Entfernt das ERSTE Vorkommen von name aus der Reihe; die Reihenfolge
    // der uebrigen bleibt gleich. Rueckgabe true, wenn name dabei war,
    // sonst false (Reihe unveraendert).
    public boolean ausscheiden(String name) {
        Queue hilf = new Queue();
        boolean gefunden = false;
        while (!reihe.isEmpty()) {
            String n = (String) reihe.dequeue();
            if (!gefunden && n.equals(name)) {
                gefunden = true;                        // NICHT wieder anstellen
            } else {
                hilf.enqueue(n);
            }
        }
        while (!hilf.isEmpty()) {
            reihe.enqueue(hilf.dequeue());              // Reihenfolge bleibt
        }
        return gefunden;
    }
}


// ========== Level 4.3 – Muster "unveraendert lassen" (DS 26) ==========
//  ALLE Methoden dieser Klasse lassen die uebergebene Struktur
//  UNVERAENDERT: dieselben Objekte in derselben Reihenfolge (Ausnahme:
//  entferneAlle). Nutze Hilfsstapel bzw. Hilfsschlange und schichte
//  am Ende zurueck - auch wenn du schon vorher fertig bist!

class Unveraendert {

    // ===================== Stufe BASIS =====================

    // zaehle(s: Stapel): Ganzzahl
    // Anzahl der Elemente in s. UNVERAENDERT (Hilfsstapel).
    public static int zaehle(Stack s) {
        Stack hilf = new Stack();
        int anzahl = 0;
        while (!s.isEmpty()) {
            hilf.push(s.pop());
            anzahl++;
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());                         // zurueckschichten
        }
        return anzahl;
    }

    // summe(q: Schlange vom Inhaltstyp Ganzzahl): Ganzzahl
    // Summe aller Werte in q. UNVERAENDERT (Hilfsschlange).
    public static int summe(Queue q) {
        Queue hilf = new Queue();
        int summe = 0;
        while (!q.isEmpty()) {
            Object o = q.dequeue();
            int x = (Integer) o;                        // Cast fuers Rechnen
            summe = summe + x;
            hilf.enqueue(o);                            // dasselbe Objekt aufheben
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return summe;
    }


    // ================== Stufe ERWEITERUNG ==================

    // enthaelt(s: Stapel vom Inhaltstyp Kiste, name: Zeichenkette): Wahrheitswert
    // true, wenn eine Kiste mit diesem Namen im Stapel liegt.
    // UNVERAENDERT - auch wenn du die Kiste schon gefunden hast!
    public static boolean enthaelt(Stack s, String name) {
        Stack hilf = new Stack();
        boolean gefunden = false;
        while (!s.isEmpty()) {
            Kiste k = (Kiste) s.pop();
            if (k.getName().equals(name)) {
                gefunden = true;                        // NICHT sofort return - erst zurueckschichten
            }
            hilf.push(k);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return gefunden;
    }

    // durchschnittsGewicht(q: Schlange vom Inhaltstyp Kiste): Kommazahl
    // Durchschnittliches Gewicht der Kisten in q. Leere Schlange -> 0.0.
    // UNVERAENDERT (Hilfsschlange).
    public static double durchschnittsGewicht(Queue q) {
        Queue hilf = new Queue();
        int summe = 0;
        int anzahl = 0;
        while (!q.isEmpty()) {
            Kiste k = (Kiste) q.dequeue();
            summe = summe + k.getGewicht();
            anzahl++;
            hilf.enqueue(k);
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        if (anzahl == 0) {
            return 0.0;                                 // nie durch 0 teilen
        }
        return (double) summe / anzahl;                 // Kommazahl-Division!
    }

    // schwerste(s: Stapel vom Inhaltstyp Kiste): Kiste
    // Die schwerste Kiste im Stapel; bei Gleichstand die OBERSTE davon.
    // Leerer Stapel -> null. UNVERAENDERT.
    public static Kiste schwerste(Stack s) {
        Stack hilf = new Stack();
        Kiste beste = null;
        while (!s.isEmpty()) {
            Kiste k = (Kiste) s.pop();
            if (beste == null || k.getGewicht() > beste.getGewicht()) {
                beste = k;                              // > (nicht >=): die obere gewinnt
            }
            hilf.push(k);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return beste;
    }


    // =================== Stufe EXPERTE =====================

    // unterste(s: Stapel vom Inhaltstyp Kiste): Kiste
    // Die unterste Kiste des Stapels. Leerer Stapel -> null. UNVERAENDERT.
    public static Kiste unterste(Stack s) {
        Stack hilf = new Stack();
        Kiste letzte = null;
        while (!s.isEmpty()) {
            letzte = (Kiste) s.pop();                   // die zuletzt entnommene ist die unterste
            hilf.push(letzte);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return letzte;
    }

    // entferneAlle(q: Schlange vom Inhaltstyp Kiste, name: Zeichenkette): Ganzzahl
    // AUSNAHME - diese Methode VERAENDERT q: Alle Kisten mit diesem Namen
    // werden entfernt, die uebrigen bleiben in ihrer Reihenfolge.
    // Rueckgabe: Anzahl der entfernten Kisten.
    public static int entferneAlle(Queue q, String name) {
        Queue hilf = new Queue();
        int anzahl = 0;
        while (!q.isEmpty()) {
            Kiste k = (Kiste) q.dequeue();
            if (k.getName().equals(name)) {
                anzahl++;                               // nicht wieder anstellen
            } else {
                hilf.enqueue(k);
            }
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return anzahl;
    }
}


// ============ Level 4.4 – Die Ladeliste (DS 27) =====================
//  Alle Methoden arbeiten auf einer dynamischen Reihung vom Inhaltstyp
//  Kiste (DynArray). Index ab 0, Laenge mit getLength().
//  Index? Vor getItem, setItem und delete: 0 <= i < getLength().
//  Wer in einer Schleife loescht, darf den Index danach NICHT erhoehen.

class Ladeliste {

    // ===================== Stufe BASIS =====================

    // gesamtGewicht(d: dynamische Reihung vom Inhaltstyp Kiste): Ganzzahl
    // Summe der Gewichte aller Kisten. Leere Reihung -> 0.
    public static int gesamtGewicht(DynArray d) {
        int summe = 0;
        for (int i = 0; i < d.getLength(); i++) {
            Kiste k = (Kiste) d.getItem(i);
            summe = summe + k.getGewicht();
        }
        return summe;
    }

    // indexVon(d: dynamische Reihung vom Inhaltstyp Kiste, name: Zeichenkette): Ganzzahl
    // Index der ERSTEN Kiste mit diesem Namen, sonst -1.
    public static int indexVon(DynArray d, String name) {
        for (int i = 0; i < d.getLength(); i++) {
            Kiste k = (Kiste) d.getItem(i);
            if (k.getName().equals(name)) {
                return i;
            }
        }
        return -1;
    }

    // leichteste(d: dynamische Reihung vom Inhaltstyp Kiste): Kiste
    // Die leichteste Kiste; bei Gleichstand die ERSTE davon.
    // Leere Reihung -> null. d bleibt unveraendert.
    public static Kiste leichteste(DynArray d) {
        if (d.isEmpty()) {
            return null;                                // Leer? -> kein getItem(0)!
        }
        Kiste beste = (Kiste) d.getItem(0);
        for (int i = 1; i < d.getLength(); i++) {
            Kiste k = (Kiste) d.getItem(i);
            if (k.getGewicht() < beste.getGewicht()) {
                beste = k;                              // < (nicht <=): die erste gewinnt
            }
        }
        return beste;
    }


    // ================== Stufe ERWEITERUNG ==================

    // einfuegenSortiert(d: dynamische Reihung vom Inhaltstyp Kiste, k: Kiste): nichts
    // d ist aufsteigend nach Gewicht sortiert. k wird so eingefuegt, dass
    // d sortiert bleibt - HINTER allen Kisten, die gleich schwer sind.
    // Ist k die schwerste, kommt sie ans Ende.
    public static void einfuegenSortiert(DynArray d, Kiste k) {
        int pos = 0;
        boolean gefunden = false;
        while (pos < d.getLength() && !gefunden) {      // Index? zuerst die Grenze pruefen
            Kiste x = (Kiste) d.getItem(pos);
            if (x.getGewicht() > k.getGewicht()) {
                gefunden = true;                        // erste echt schwerere Kiste
            } else {
                pos++;
            }
        }
        d.insertAt(pos, k);                             // pos == getLength() haengt hinten an
    }

    // entferneLeichteste(d: dynamische Reihung vom Inhaltstyp Kiste): Kiste
    // Entfernt die leichteste Kiste (bei Gleichstand die ERSTE davon) aus d
    // und gibt sie zurueck. Leere Reihung -> null.
    public static Kiste entferneLeichteste(DynArray d) {
        if (d.isEmpty()) {
            return null;
        }
        int minIndex = 0;
        for (int i = 1; i < d.getLength(); i++) {
            Kiste k = (Kiste) d.getItem(i);
            Kiste m = (Kiste) d.getItem(minIndex);
            if (k.getGewicht() < m.getGewicht()) {
                minIndex = i;
            }
        }
        Kiste leicht = (Kiste) d.getItem(minIndex);     // ERST merken ...
        d.delete(minIndex);                             // ... DANN loeschen
        return leicht;
    }

    // entferneDuplikate(d: dynamische Reihung vom Inhaltstyp Kiste): nichts
    // Kisten mit gleichem Namen: Nur das ERSTE Vorkommen bleibt, alle
    // spaeteren werden geloescht. Die Reihenfolge der uebrigen bleibt.
    public static void entferneDuplikate(DynArray d) {
        for (int i = 0; i < d.getLength(); i++) {
            Kiste a = (Kiste) d.getItem(i);
            int j = i + 1;
            while (j < d.getLength()) {
                Kiste b = (Kiste) d.getItem(j);
                if (b.getName().equals(a.getName())) {
                    d.delete(j);                        // j NICHT erhoehen: der Nachfolger rueckt nach
                } else {
                    j++;
                }
            }
        }
    }


    // =================== Stufe EXPERTE =====================

    // einfuegesortierung(d: dynamische Reihung vom Inhaltstyp Kiste): nichts
    // Sortiert d aufsteigend nach Gewicht mit Sortieren durch Einfuegen.
    // STABIL: gleich schwere Kisten behalten ihre Reihenfolge.
    // Nur getItem, setItem, insertAt, delete und getLength benutzen
    // (keine Reihung, keine zweite DynArray).
    public static void einfuegesortierung(DynArray d) {
        for (int i = 1; i < d.getLength(); i++) {
            Kiste k = (Kiste) d.getItem(i);
            int j = i - 1;
            boolean weiter = true;
            while (j >= 0 && weiter) {
                Kiste v = (Kiste) d.getItem(j);
                if (v.getGewicht() > k.getGewicht()) {  // nur ECHT schwerere nach rechts -> stabil
                    d.setItem(j + 1, v);
                    j--;
                } else {
                    weiter = false;
                }
            }
            d.setItem(j + 1, k);
        }
    }

    // abspalten(d: dynamische Reihung vom Inhaltstyp Kiste, grenze: Ganzzahl):
    //     dynamische Reihung vom Inhaltstyp Kiste
    // Alle Kisten mit gewicht > grenze werden aus d entfernt und in einer
    // NEUEN dynamischen Reihung zurueckgegeben (Reihenfolge bleibt in
    // beiden erhalten). Keine solche Kiste -> leere Reihung (nicht null).
    public static DynArray abspalten(DynArray d, int grenze) {
        DynArray schwer = new DynArray();
        int i = 0;
        while (i < d.getLength()) {
            Kiste k = (Kiste) d.getItem(i);
            if (k.getGewicht() > grenze) {
                schwer.append(k);
                d.delete(i);                            // i NICHT erhoehen
            } else {
                i++;
            }
        }
        return schwer;
    }
}


// ============ Level 4.5 – Umformen und Anwendungen (DS 28) ============

class Umformen {

    // ===================== Stufe BASIS =====================

    // schlangeZuDynArray(q: Schlange): dynamische Reihung
    // Neue dynamische Reihung mit den Elementen von q in Schlangen-
    // reihenfolge (vorne -> Index 0). q bleibt UNVERAENDERT.
    public static DynArray schlangeZuDynArray(Queue q) {
        DynArray d = new DynArray();
        while (!q.isEmpty()) {
            d.append(q.dequeue());
        }
        for (int i = 0; i < d.getLength(); i++) {
            q.enqueue(d.getItem(i));                    // q wiederherstellen
        }
        return d;
    }

    // dynArrayZuStapel(d: dynamische Reihung): Stapel
    // Neuer Stapel: Element 0 wird zuerst gepusht, das letzte Element
    // liegt oben. d bleibt unveraendert.
    public static Stack dynArrayZuStapel(DynArray d) {
        Stack s = new Stack();
        for (int i = 0; i < d.getLength(); i++) {
            s.push(d.getItem(i));
        }
        return s;
    }


    // ================== Stufe ERWEITERUNG ==================

    // klammernOk(s: Zeichenkette): Wahrheitswert
    // true, wenn die Klammern ( ) [ ] { } in s richtig gesetzt sind: Jede
    // schliessende Klammer passt zur zuletzt geoeffneten, und am Ende ist
    // keine Klammer mehr offen. Andere Zeichen werden ignoriert.
    // Benutze einen Stapel vom Inhaltstyp Zeichen.
    public static boolean klammernOk(String s) {
        Stack offen = new Stack();
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '(' || c == '[' || c == '{') {
                offen.push(c);                          // char wird automatisch verpackt
            } else if (c == ')' || c == ']' || c == '}') {
                if (offen.isEmpty()) {
                    return false;                       // Leer? zu viele schliessende Klammern
                }
                char auf = (Character) offen.pop();
                if (c == ')' && auf != '(') {
                    return false;
                }
                if (c == ']' && auf != '[') {
                    return false;
                }
                if (c == '}' && auf != '{') {
                    return false;
                }
            }
        }
        return offen.isEmpty();                         // noch Klammern offen? -> false
    }


    // =================== Stufe EXPERTE =====================

    // postfix(ausdruck: Zeichenkette): Ganzzahl
    // Wertet einen Ausdruck in Postfix-Schreibweise aus: einstellige Zahlen
    // und die Operatoren + - *, ohne Leerzeichen. "34+2*" = (3 + 4) * 2 = 14.
    // Der Ausdruck ist immer gueltig. Ziffer -> Zahl: (int) c - (int) '0'.
    public static int postfix(String ausdruck) {
        Stack werte = new Stack();
        for (int i = 0; i < ausdruck.length(); i++) {
            char c = ausdruck.charAt(i);
            if (c >= '0' && c <= '9') {
                werte.push((int) c - (int) '0');        // Ziffer -> Zahlwert (ASCII)
            } else {
                int b = (Integer) werte.pop();          // ZUERST der RECHTE Operand
                int a = (Integer) werte.pop();
                if (c == '+') {
                    werte.push(a + b);
                } else if (c == '-') {
                    werte.push(a - b);
                } else {
                    werte.push(a * b);
                }
            }
        }
        return (Integer) werte.pop();                   // gueltig: genau ein Wert uebrig
    }

    // nachGewicht(q: Schlange vom Inhaltstyp Kiste, grenze: Ganzzahl):
    //     dynamische Reihung vom Inhaltstyp Kiste
    // Neue dynamische Reihung mit allen Kisten aus q mit gewicht <= grenze
    // (in Schlangenreihenfolge). q bleibt UNVERAENDERT.
    public static DynArray nachGewicht(Queue q, int grenze) {
        DynArray d = new DynArray();
        Queue hilf = new Queue();
        while (!q.isEmpty()) {
            Kiste k = (Kiste) q.dequeue();
            if (k.getGewicht() <= grenze) {
                d.append(k);
            }
            hilf.enqueue(k);
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return d;
    }
}


// ========== Level 4.6 – Expertenpfad (DS 29, Kuer-DS) ===============
//  (a) Nebel lueften wie in Welt 3 - aber OHNE Rekursion: Statt der
//      Selbstaufrufe merkst du dir die noch zu bearbeitenden Felder in
//      einer Schlange (mitSchlange == true) oder einem Stapel (false).
//  (b) Ein eigener Stapel aus Knoten: So koennte Stack innen gebaut sein.
//
//  REGELN fuer fluteIterativ (sonst stimmt die spur nicht):
//  - Begehbar sind '.', 'S' und 'Z' (im Feld!). Beim Markieren wird aus
//    '.' ein '+'; 'S' und 'Z' bleiben stehen, gelten aber als besucht
//    (Tipp: eine Reihung von Reihung von Wahrheitswert "besucht").
//  - Eine Position (z, s) wird als EINE Ganzzahl gespeichert:
//        pos = z * 1000 + s      zurueck: z = pos / 1000, s = pos % 1000
//  - MARKIEREN BEIM EINFUEGEN: Ein Feld wird markiert (und mit merke(z, s)
//    gemeldet), BEVOR es in die Schlange bzw. den Stapel kommt - so kommt
//    kein Feld zweimal hinein.
//  - Nachbarn IMMER in der Reihenfolge oben, rechts, unten, links.
//
//      fluteIterativ(karte, z, s, mitSchlange)
//      ist (z, s) nicht begehbar?  -> gib 0 zurueck
//      markiere (z, s), lege z * 1000 + s hinein
//      solange die Struktur nicht leer ist:
//          entnimm pos (dequeue bzw. pop), berechne z und s daraus
//          fuer jeden Nachbarn (oben, rechts, unten, links):
//              begehbar und noch nicht besucht?
//                  -> markieren, merke(nz, ns), hineinlegen
//      gib die Anzahl der Felder zurueck, die von '.' zu '+' wurden

class Expertenpfad {

    // ======== VORGEGEBEN – bis zur Linie "AB HIER ARBEITEST DU" nichts veraendern ========

    // Protokoll der markierten Felder in der Reihenfolge des Markierens,
    // z. B. "(1,1) (1,2) (2,1)". Die Tests setzen spur vorher auf "".
    public static String spur = "";

    // Haengt das Feld (z, s) an die spur an. Rufe merke(z, s) direkt beim
    // Markieren jedes Feldes auf (auch fuer S und Z).
    public static void merke(int z, int s) {
        if (!spur.equals("")) {
            spur = spur + " ";
        }
        spur = spur + "(" + z + "," + s + ")";
    }

    // ======== AB HIER ARBEITEST DU =============================================


    // ========= Stufe BASIS und ERWEITERUNG ==========
    // (Basis-Tests: Endkarte und Anzahl mit Schlange; Erweiterung: mit
    //  Stapel, Karte ohne Randmauer und die genaue Reihenfolge in spur.)

    // fluteIterativ(karte: Reihung von Reihung von Zeichen, z: Ganzzahl, s: Ganzzahl,
    //     mitSchlange: Wahrheitswert): Ganzzahl
    // Lueftet ab (z, s) alle erreichbaren Felder (siehe REGELN oben) mit
    // einer Schlange (mitSchlange == true) bzw. einem Stapel (false).
    // Rueckgabe: Anzahl der Felder, die von '.' zu '+' wurden.
    // Ist (z, s) ausserhalb, eine Wand oder schon '+': 0, nichts aendern.
    public static int fluteIterativ(char[][] karte, int z, int s, boolean mitSchlange) {
        int zeilen = karte.length;
        int spalten = karte[0].length;
        if (z < 0 || z >= zeilen || s < 0 || s >= spalten) {
            return 0;                                   // Index? ausserhalb
        }
        if (karte[z][s] != '.' && karte[z][s] != 'S' && karte[z][s] != 'Z') {
            return 0;                                   // Wand oder schon gelueftet
        }
        boolean[][] besucht = new boolean[zeilen][spalten];
        Queue schlange = new Queue();
        Stack stapel = new Stack();
        int anzahl = 0;
        // Start markieren und hineinlegen
        besucht[z][s] = true;
        merke(z, s);
        if (karte[z][s] == '.') {
            karte[z][s] = '+';
            anzahl++;
        }
        if (mitSchlange) {
            schlange.enqueue(z * 1000 + s);
        } else {
            stapel.push(z * 1000 + s);
        }
        int[] dz = {-1, 0, 1, 0};                       // oben, rechts, unten, links
        int[] ds = {0, 1, 0, -1};
        while (!schlange.isEmpty() || !stapel.isEmpty()) {
            int pos = 0;
            if (mitSchlange) {
                pos = (Integer) schlange.dequeue();
            } else {
                pos = (Integer) stapel.pop();
            }
            int pz = pos / 1000;
            int ps = pos % 1000;
            for (int r = 0; r < 4; r++) {
                int nz = pz + dz[r];
                int ns = ps + ds[r];
                if (nz >= 0 && nz < zeilen && ns >= 0 && ns < spalten) {
                    char c = karte[nz][ns];
                    if (!besucht[nz][ns] && (c == '.' || c == 'S' || c == 'Z')) {
                        besucht[nz][ns] = true;         // markieren BEIM Einfuegen
                        merke(nz, ns);
                        if (c == '.') {
                            karte[nz][ns] = '+';
                            anzahl++;
                        }
                        if (mitSchlange) {
                            schlange.enqueue(nz * 1000 + ns);
                        } else {
                            stapel.push(nz * 1000 + ns);
                        }
                    }
                }
            }
        }
        return anzahl;
    }
}


//  Klassenkarten (Experte):
//      Knoten                                  MeinStapel
//      -----------------------------------     -----------------------------------
//      - inhalt: Object                        - kopf: Knoten     // oberster Knoten
//      - naechster: Knoten                     -----------------------------------
//      -----------------------------------     + c MeinStapel()
//      + c Knoten(inhalt: Object,              + isEmpty(): Wahrheitswert
//            naechster: Knoten)                + push(inhalt: Object): nichts
//      + getInhalt(): Object                   + top(): Object
//      + getNaechster(): Knoten                + pop(): Object
//      + setInhalt(inhalt: Object): nichts
//      + setNaechster(naechster: Knoten): nichts
//
//      kopf -> [Gold | o-]-> [Holz | o-]-> [Erz | null]
//              (oben)                       (unten)
//
//      Alle Attribute private, alle Methoden public. MeinStapel verhaelt
//      sich wie Stack: top() und pop() auf einem LEEREN Stapel werfen
//          throw new IllegalStateException("Der Stapel ist leer.");
//      Die Tests vergleichen MeinStapel mit Stack in 400 zufaelligen
//      Befehlen (immer dieselbe Folge, Testhilfe.naechsteZufallszahl).

class Knoten {

    private Object inhalt;
    private Knoten naechster;

    // =================== Stufe EXPERTE =====================

    // c Knoten(inhalt: Object, naechster: Knoten)
    public Knoten(Object inhalt, Knoten naechster) {
        this.inhalt = inhalt;
        this.naechster = naechster;
    }

    // getInhalt(): Object
    public Object getInhalt() {
        return inhalt;
    }

    // getNaechster(): Knoten
    public Knoten getNaechster() {
        return naechster;
    }

    // setInhalt(inhalt: Object): nichts
    public void setInhalt(Object inhalt) {
        this.inhalt = inhalt;
    }

    // setNaechster(naechster: Knoten): nichts
    public void setNaechster(Knoten naechster) {
        this.naechster = naechster;
    }
}


class MeinStapel {

    private Knoten kopf;            // oberster Knoten; null = leerer Stapel

    // =================== Stufe EXPERTE =====================
    // Gleiches Verhalten wie Stack: top() und pop() auf einem leeren
    // Stapel werfen eine IllegalStateException:
    //     throw new IllegalStateException("Der Stapel ist leer.");

    // c MeinStapel()
    // Erzeugt einen leeren Stapel.
    public MeinStapel() {
        kopf = null;
    }

    // isEmpty(): Wahrheitswert
    public boolean isEmpty() {
        return kopf == null;
    }

    // push(inhalt: Object): nichts
    // Neuer Knoten wird der neue kopf; sein Nachfolger ist der alte kopf.
    public void push(Object inhalt) {
        kopf = new Knoten(inhalt, kopf);
    }

    // top(): Object
    // Inhalt des obersten Knotens (ohne Entfernen).
    public Object top() {
        if (isEmpty()) {
            throw new IllegalStateException("Der Stapel ist leer.");
        }
        return kopf.getInhalt();
    }

    // pop(): Object
    // Inhalt des obersten Knotens; der Knoten wird entfernt.
    public Object pop() {
        if (isEmpty()) {
            throw new IllegalStateException("Der Stapel ist leer.");
        }
        Object inhalt = kopf.getInhalt();
        kopf = kopf.getNaechster();
        return inhalt;
    }
}

                        assertEquals(a, b, "Schritt " + schritt + ": pop liefert bei Stack und MeinStapel verschiedene Werte");
                        hoehe--;
                    } else {
                        int a = (Integer) original.top();
                        int b = (Integer) mein.top();
                        assertEquals(a, b, "Schritt " + schritt + ": top liefert bei Stack und MeinStapel verschiedene Werte");
                    }
                }
                if (hoehe > maxHoehe) {
                    maxHoehe = hoehe;
                }
                assertEquals(original.isEmpty(), mein.isEmpty(), "Schritt " + schritt + ": isEmpty() bei Stack und MeinStapel verschieden");
            }
            while (!original.isEmpty()) {
                int a = (Integer) original.pop();
                int b = (Integer) mein.pop();
                assertEquals(a, b, "Abbauen am Ende: pop liefert verschiedene Werte");
            }
            assertTrue(mein.isEmpty(), "Am Ende muessen beide Stapel leer sein");
            assertTrue(maxHoehe >= 3, "(Selbstkontrolle des Tests: Der Stapel war mindestens 3 hoch)");
        } catch (RuntimeException e) {
            Testhilfe.laufzeitfehler(e);
        }
    }
