// ---------------------------------------------------------------------
//  GERÜST – hier nichts veraendern
// ---------------------------------------------------------------------
//  Baumschule: die Testbaeume der Welt 5 (wie in der Level-Datei).
//  Item:       ein Gegenstand der Item-Datenbank (Nummer und Name).
//  Zeichner:   zeichnet einen BinTree und faerbt Knoten ein.
// ---------------------------------------------------------------------

//  Die Testbaeume (ja = links, nein = rechts beim Monsterquiz):
//
//  beispielBST()            zahlenbaum()            termbaum()
//  (BST: 8, 3, 10, 1, 6,    (KEIN BST,              "(3+4)*(5-2)"
//   14, 4, 7, 13 eingefuegt) Duplikat 9)
//          8                       5                     *
//        /   \                   /   \                 /   \
//       3     10                9     4               +     -
//      / \      \              / \     \             / \   / \
//     1   6      14           2   7     9           3   4 5   2
//        / \    /                      /
//       4   7  13                     1
//
//  entartet(): 1 -> 2 -> 3 -> 4 -> 5 (jeder Knoten hat nur ein rechtes Kind)
//
//  monsterquiz():                 Kann es fliegen?
//                              ja /              \ nein
//                    Speit es Feuer?          Lebt es im Wasser?
//                    ja /     \ nein          ja /         \ nein
//                   Drache    Greif         Kraken         Troll
//
//  leer(): leerer Baum      blatt(x): ein einzelnes Blatt mit dem Wert x

class Baumschule {

    // knoten(inhalt: Object, links: BinTree, rechts: BinTree): BinTree
    // Neuer Knoten mit dem Inhalt und den beiden Teilbaeumen.
    public static BinTree knoten(Object inhalt, BinTree links, BinTree rechts) {
        BinTree b = new BinTree(inhalt);
        b.setLeft(links);
        b.setRight(rechts);
        return b;
    }

    // blatt(x: Ganzzahl): BinTree  - ein Knoten mit zwei leeren Teilbaeumen
    public static BinTree blatt(int x) {
        return new BinTree(x);
    }

    // leer(): BinTree  - der leere Baum
    public static BinTree leer() {
        return new BinTree();
    }

    // Der Beispiel-Suchbaum: entsteht durch Einfuegen von
    // 8, 3, 10, 1, 6, 14, 4, 7, 13 (in dieser Reihenfolge).
    public static BinTree beispielBST() {
        BinTree links = knoten(3, blatt(1), knoten(6, blatt(4), blatt(7)));
        BinTree rechts = knoten(10, leer(), knoten(14, blatt(13), leer()));
        return knoten(8, links, rechts);
    }

    // Ein Zahlenbaum, der KEIN Suchbaum ist (9 kommt zweimal vor).
    public static BinTree zahlenbaum() {
        BinTree links = knoten(9, blatt(2), blatt(7));
        BinTree rechts = knoten(4, leer(), knoten(9, blatt(1), leer()));
        return knoten(5, links, rechts);
    }

    // Entarteter Suchbaum: 1, 2, 3, 4, 5 eingefuegt -> rechte Kette.
    public static BinTree entartet() {
        BinTree b = blatt(5);
        for (int i = 4; i >= 1; i--) {
            b = knoten(i, leer(), b);
        }
        return b;
    }

    // Termbaum zu (3+4)*(5-2): Inhaltstyp Zeichenkette.
    // Innere Knoten: Rechenzeichen "+", "-", "*", "/"; Blaetter: Zahlen.
    public static BinTree termbaum() {
        BinTree plus = knoten("+", new BinTree("3"), new BinTree("4"));
        BinTree minus = knoten("-", new BinTree("5"), new BinTree("2"));
        return knoten("*", plus, minus);
    }

    // Startbaum des Akinator-Monsterquiz: innere Knoten = Fragen,
    // Blaetter = Monster; ja -> links, nein -> rechts.
    public static BinTree monsterquiz() {
        BinTree fliegt = knoten("Speit es Feuer?", new BinTree("Drache"), new BinTree("Greif"));
        BinTree laeuft = knoten("Lebt es im Wasser?", new BinTree("Kraken"), new BinTree("Troll"));
        return knoten("Kann es fliegen?", fliegt, laeuft);
    }
}


//  Klassenkarte (Notation wie im Abitur):
//      Item
//      ------------------------------------------------
//      - nummer: Ganzzahl                      // Schluessel im Suchbaum
//      - name: Zeichenkette
//      ------------------------------------------------
//      + c Item(nummer: Ganzzahl, name: Zeichenkette)
//      + getNummer(): Ganzzahl
//      + getName(): Zeichenkette
//      + beschreibung(): Zeichenkette          // z. B. "#65 Bogen"

class Item {

    private int nummer;
    private String name;

    public Item(int nummer, String name) {
        this.nummer = nummer;
        this.name = name;
    }

    public int getNummer() {
        return nummer;
    }

    public String getName() {
        return name;
    }

    public String beschreibung() {
        return "#" + nummer + " " + name;
    }
}


// Der Zeichner malt einen BinTree in die Grafik und faerbt Knoten ein.
//
// Befehle fuer dich (alle static):
//   Zeichner.zeichne(b, "Titel")        zeichnet den Baum b (Zahlen,
//                                       kurze Zeichenketten)
//   Zeichner.zeichneItems(b, "Titel")   zeichnet einen BinTree vom
//                                       Inhaltstyp Item (zeigt die Nummer)
//   Zeichner.start("Preorder")          neue Runde: alle Knoten wieder
//                                       hellblau, Ausgabezeile leer
//   Zeichner.betrete(b)                 Knoten b blau: der Aufruf fuer b
//                                       laeuft gerade
//   Zeichner.besuche(b)                 Knoten b gruen und sein Inhalt
//                                       kommt in die Ausgabezeile
//   Zeichner.hervorheben(b, farbe)      Knoten b in einer Farbe, z. B.
//                                       Zeichner.WEG (gold) oder
//                                       Zeichner.GEFUNDEN (kraeftig gruen)
//   Zeichner.neuZeichnen()              Baum neu zeichnen, nachdem er sich
//                                       veraendert hat (z. B. eingefuegt);
//                                       die Farben bleiben erhalten
//   Zeichner.meldung("Text")            Text unter dem Titel (und in der
//                                       Konsole)
//   Zeichner.fertig()                   schreibt die Ausgabezeile in die
//                                       Konsole, z. B. "Preorder: 8 3 1 ..."
//   Zeichner.setPause(n)                Wartezeit nach jedem Einfaerben
//                                       (Standard 60, 0 = keine)
//   Zeichner.koordinaten()              die berechneten Positionen als Text
//
// Leere Baeume, null oder Knoten, die nicht gezeichnet sind, werden bei
// betrete, besuche und hervorheben einfach uebergangen.
//
// So wird gezeichnet: Die Knoten sind Quadrate, Ebene i liegt in der
// Zeile i (Wurzel oben). Die x-Position ist der INORDER-INDEX: der
// kleinste Wert (ganz links in Inorder) bekommt Spalte 0, der naechste
// Spalte 1 usw. Bei einem Suchbaum stehen die Werte also von links nach
// rechts aufsteigend sortiert. Weil jeder Knoten seine eigene Spalte hat,
// ueberlappt nichts. Die Kanten sind Winkel: waagerecht aus der
// Elternmitte bis ueber das Kind, dann senkrecht hinunter.
class Zeichner {

    // ----- Farben fuer hervorheben -----
    public static String NORMAL = "#DBE7F5";       // hellblau: noch nicht besucht
    public static String AKTIV = "#5B9BD5";        // blau: Aufruf laeuft gerade
    public static String BESUCHT = "#A9D18E";      // gruen: besucht / ausgegeben
    public static String WEG = "#FFD966";          // gold: Suchweg
    public static String GEFUNDEN = "#70AD47";     // kraeftig gruen: gefunden
    public static String NEU = "#F4B183";          // orange: neu eingefuegt
    public static String FEHLT = "#FF7C80";        // rot: hier fehlt etwas

    private static String FARBE_LINIE = "#BFBFBF";
    private static String FARBE_SCHRIFT = "#1F2D3D";

    // ----- Zeichenflaeche -----
    private static int BREITE = 1000;
    private static int HOEHE = 700;
    private static int RAND = 30;
    private static int OBEN = 85;                  // y der Wurzel
    private static int UNTEN = 600;                // unterste Kante der Knoten
    private static int MAX_KNOTEN = 40;

    // ----- der gezeichnete Baum (Index = Inorder-Index) -----
    private static BinTree wurzel = null;
    private static boolean itemModus = false;
    private static BinTree[] knoten = new BinTree[40];
    private static int[] ebeneVon = new int[40];
    private static int[] kindLinks = new int[40];
    private static int[] kindRechts = new int[40];
    private static int[] posX = new int[40];
    private static int[] posY = new int[40];
    private static String[] farbe = new String[40];
    private static Rectangle[] kasten = new Rectangle[40];
    private static Text[] schrift = new Text[40];
    private static int anzahl = 0;
    private static int tiefe = 0;
    private static boolean zuGross = false;
    private static Rectangle[] linien = new Rectangle[80];
    private static int anzahlLinien = 0;

    // ----- Layout (wird an den Baum angepasst) -----
    private static int abstand = 80;               // Abstand der Spalten
    private static int zeile = 90;                 // Abstand der Ebenen
    private static int groesse = 46;               // Kantenlaenge eines Knotens
    private static int links0 = 0;                 // x der Spalte 0

    // ----- Texte -----
    private static boolean weltBereit = false;
    private static Text titel;
    private static Text info;
    private static Text ausgabe;
    private static Text legende;
    private static String titelText = "";
    private static String ausgabeText = "";
    private static int pausenLaenge = 60;


    // =============== Befehle ===============

    public static void zeichne(BinTree b, String ueberschrift) {
        itemModus = false;
        neuerBaum(b, ueberschrift);
    }

    public static void zeichneItems(BinTree b, String ueberschrift) {
        itemModus = true;
        neuerBaum(b, ueberschrift);
    }

    public static void start(String ueberschrift) {
        welt();
        warte(3);                                  // altes Bild kurz stehen lassen
        for (int i = 0; i < anzahl; i++) {
            farbe[i] = NORMAL;
            kasten[i].setFillColor(NORMAL);
        }
        titelText = ueberschrift;
        ausgabeText = "";
        titel.setText(ueberschrift);
        info.setText("");
        ausgabe.setText(ueberschrift + ": ");
    }

    public static void betrete(BinTree b) {
        hervorheben(b, AKTIV);
    }

    public static void besuche(BinTree b) {
        if (index(b) < 0) {
            return;
        }
        ausgabeText = ausgabeText + beschriftung(b.getItem()) + " ";
        ausgabe.setText(titelText + ": " + ausgabeText);
        hervorheben(b, BESUCHT);
    }

    public static void hervorheben(BinTree b, String f) {
        int i = index(b);
        if (i < 0) {
            return;
        }
        farbe[i] = f;
        kasten[i].setFillColor(f);
        warte(1);
    }

    public static void meldung(String text) {
        welt();
        info.setText(text);
        println("   " + text);
    }

    public static void fertig() {
        println(titelText + ": " + ausgabeText);
    }

    public static String getAusgabe() {
        return ausgabeText;
    }

    public static void setPause(int n) {
        pausenLaenge = n;
    }

    // Die berechneten Positionen, z. B. "8 (493|85)  3 (333|175) ..."
    // (linke obere Ecke der Knoten, in Preorder).
    public static String koordinaten() {
        return koordinaten(wurzel);
    }

    public static void neuZeichnen() {
        welt();
        BinTree[] alteKnoten = knoten;
        String[] alteFarben = farbe;
        int alteAnzahl = anzahl;
        entfernen();
        knoten = new BinTree[MAX_KNOTEN];
        farbe = new String[MAX_KNOTEN];
        anzahl = 0;
        tiefe = 0;
        zuGross = false;
        if (wurzel == null || wurzel.isEmpty()) {
            info.setText("(leerer Baum)");
            return;
        }
        platziere(wurzel, 0);
        if (!zuGross) {
            layout();
        }
        if (zuGross || groesse < 10) {
            anzahl = 0;
            info.setText("Der Baum ist zu gross oder zu tief zum Zeichnen.");
            return;
        }
        for (int i = 0; i < anzahl; i++) {
            if (kindLinks[i] >= 0) {
                verbinde(i, kindLinks[i]);
            }
            if (kindRechts[i] >= 0) {
                verbinde(i, kindRechts[i]);
            }
        }
        for (int i = 0; i < anzahl; i++) {
            farbe[i] = NORMAL;
            for (int k = 0; k < alteAnzahl; k++) {
                if (alteKnoten[k] == knoten[i]) {
                    farbe[i] = alteFarben[k];      // Farbe vom letzten Bild behalten
                }
            }
            kasten[i] = new Rectangle(posX[i], posY[i], groesse, groesse);
            kasten[i].setFillColor(farbe[i]);
            String t = beschriftung(knoten[i].getItem());
            int fs = groesse * 45 / 100;
            int textBreite = t.length() * fs * 55 / 100;
            schrift[i] = new Text(posX[i] + (groesse - textBreite) / 2, posY[i] + (groesse - fs) / 2, fs, t);
            schrift[i].setFillColor(FARBE_SCHRIFT);
        }
    }


    // =============== intern ===============

    private static void neuerBaum(BinTree b, String ueberschrift) {
        welt();
        entfernen();
        anzahl = 0;                                // keine alten Farben uebernehmen
        wurzel = b;
        neuZeichnen();
        titelText = ueberschrift;
        ausgabeText = "";
        titel.setText(ueberschrift);
        ausgabe.setText("");
    }

    private static void welt() {
        if (!weltBereit) {
            new World(BREITE, HOEHE);
            titel = new Text(RAND, 12, 28, "");
            titel.setFillColor("#DAA520");
            info = new Text(RAND, 50, 18, "");
            info.setFillColor("#8FA8C8");
            ausgabe = new Text(RAND, 622, 22, "");
            ausgabe.setFillColor("#9DC3E6");
            legende = new Text(RAND, 664, 15, "hellblau: unbesucht | blau: Aufruf laeuft | gruen: besucht | gold: Suchweg | orange: neu | rot: fehlt");
            legende.setFillColor("#8C8C8C");
            weltBereit = true;
        }
    }

    // Kurze Pause (Tempo zusaetzlich mit dem Regler der IDE einstellen).
    private static void warte(int mal) {
        int summe = 0;
        for (int i = 0; i < pausenLaenge * mal; i++) {
            summe = summe + i;
        }
    }

    private static int index(BinTree b) {
        for (int i = 0; i < anzahl; i++) {
            if (knoten[i] == b) {
                return i;
            }
        }
        return -1;
    }

    private static String beschriftung(Object o) {
        String t = "" + o;
        if (itemModus) {
            Item it = (Item) o;
            t = "" + it.getNummer();
        }
        if (t.length() > 4) {
            t = t.substring(0, 3) + ".";
        }
        return t;
    }

    // Inorder durchlaufen: Der Index eines Knotens ist seine Inorder-Nummer
    // (= Spalte). Rueckgabe: Index des Knotens, -1 fuer den leeren Baum.
    private static int platziere(BinTree b, int ebene) {
        if (b.isEmpty()) {
            return -1;
        }
        int l = platziere(b.getLeft(), ebene + 1);
        if (anzahl >= MAX_KNOTEN) {
            zuGross = true;
            return -1;
        }
        int i = anzahl;
        anzahl++;
        knoten[i] = b;
        ebeneVon[i] = ebene;
        kindLinks[i] = l;
        if (ebene > tiefe) {
            tiefe = ebene;
        }
        kindRechts[i] = platziere(b.getRight(), ebene + 1);
        return i;
    }

    // Spaltenabstand, Zeilenabstand und Knotengroesse an den Baum anpassen.
    private static void layout() {
        abstand = (BREITE - 2 * RAND) / anzahl;
        if (abstand > 80) {
            abstand = 80;
        }
        zeile = 90;
        if (tiefe > 0 && (UNTEN - OBEN - 46) / tiefe < 90) {
            zeile = (UNTEN - OBEN - 46) / tiefe;
        }
        groesse = 46;
        if (groesse > abstand - 10) {
            groesse = abstand - 10;
        }
        if (groesse > zeile - 10) {
            groesse = zeile - 10;
        }
        links0 = (BREITE - anzahl * abstand) / 2 + (abstand - groesse) / 2;
        for (int i = 0; i < anzahl; i++) {
            posX[i] = links0 + i * abstand;
            posY[i] = OBEN + ebeneVon[i] * zeile;
        }
    }

    // Kante als Winkel: waagerecht in Hoehe der Elternmitte bis ueber die
    // Mitte des Kindes, dann senkrecht hinunter bis zur Oberkante des Kindes.
    private static void verbinde(int eltern, int kind) {
        int xm = posX[kind] + groesse / 2;         // Mitte des Kindes
        int ym = posY[eltern] + groesse / 2;       // Hoehe der Elternmitte
        if (xm < posX[eltern]) {
            linie(xm - 1, ym - 1, posX[eltern] - xm + 1, 3);
        } else {
            linie(posX[eltern] + groesse, ym - 1, xm - posX[eltern] - groesse + 2, 3);
        }
        linie(xm - 1, ym - 1, 3, posY[kind] - ym + 1);
    }

    private static void linie(int x, int y, int b, int h) {
        Rectangle r = new Rectangle(x, y, b, h);
        r.setFillColor(FARBE_LINIE);
        linien[anzahlLinien] = r;
        anzahlLinien++;
    }

    private static void entfernen() {
        for (int i = 0; i < anzahl; i++) {
            kasten[i].destroy();
            schrift[i].destroy();
        }
        for (int i = 0; i < anzahlLinien; i++) {
            linien[i].destroy();
        }
        anzahlLinien = 0;
    }

    private static String koordinaten(BinTree b) {
        if (b == null || b.isEmpty()) {
            return "";
        }
        int i = index(b);
        String t = "";
        if (i >= 0) {
            t = beschriftung(b.getItem()) + " (" + posX[i] + "|" + posY[i] + ")  ";
        }
        return t + koordinaten(b.getLeft()) + koordinaten(b.getRight());
    }
}
