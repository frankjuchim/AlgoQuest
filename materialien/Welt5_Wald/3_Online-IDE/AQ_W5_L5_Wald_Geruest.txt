//  Die Testbaeume (ja = links, nein = rechts beim Monsterquiz):
//
//  beispielBST()            zahlenbaum()            termbaum()
//  (BST: 8, 3, 10, 1, 6,    (KEIN BST,              "(3+4)*(5-2)"
//   14, 4, 7, 13 eingefuegt) Duplikat 9)
//          8                       5                     *
//        /   \                   /   \                 /   \
//       3     10                9     4               +     -
//      / \      \              / \     \             / \   / \
//     1   6      14           2   7     9           3   4 5   2
//        / \    /                      /
//       4   7  13                     1
//
//  entartet(): 1 -> 2 -> 3 -> 4 -> 5 (jeder Knoten hat nur ein rechtes Kind)
//
//  monsterquiz():                 Kann es fliegen?
//                              ja /              \ nein
//                    Speit es Feuer?          Lebt es im Wasser?
//                    ja /     \ nein          ja /         \ nein
//                   Drache    Greif         Kraken         Troll
//
//  leer(): leerer Baum      blatt(x): ein einzelnes Blatt mit dem Wert x

class Baumschule {

    // knoten(inhalt: Object, links: BinTree, rechts: BinTree): BinTree
    // Neuer Knoten mit dem Inhalt und den beiden Teilbaeumen.
    public static BinTree knoten(Object inhalt, BinTree links, BinTree rechts) {
        BinTree b = new BinTree(inhalt);
        b.setLeft(links);
        b.setRight(rechts);
        return b;
    }

    // blatt(x: Ganzzahl): BinTree  - ein Knoten mit zwei leeren Teilbaeumen
    public static BinTree blatt(int x) {
        return new BinTree(x);
    }

    // leer(): BinTree  - der leere Baum
    public static BinTree leer() {
        return new BinTree();
    }

    // Der Beispiel-Suchbaum: entsteht durch Einfuegen von
    // 8, 3, 10, 1, 6, 14, 4, 7, 13 (in dieser Reihenfolge).
    public static BinTree beispielBST() {
        BinTree links = knoten(3, blatt(1), knoten(6, blatt(4), blatt(7)));
        BinTree rechts = knoten(10, leer(), knoten(14, blatt(13), leer()));
        return knoten(8, links, rechts);
    }

    // Ein Zahlenbaum, der KEIN Suchbaum ist (9 kommt zweimal vor).
    public static BinTree zahlenbaum() {
        BinTree links = knoten(9, blatt(2), blatt(7));
        BinTree rechts = knoten(4, leer(), knoten(9, blatt(1), leer()));
        return knoten(5, links, rechts);
    }

    // Entarteter Suchbaum: 1, 2, 3, 4, 5 eingefuegt -> rechte Kette.
    public static BinTree entartet() {
        BinTree b = blatt(5);
        for (int i = 4; i >= 1; i--) {
            b = knoten(i, leer(), b);
        }
        return b;
    }

    // Termbaum zu (3+4)*(5-2): Inhaltstyp Zeichenkette.
    // Innere Knoten: Rechenzeichen "+", "-", "*", "/"; Blaetter: Zahlen.
    public static BinTree termbaum() {
        BinTree plus = knoten("+", new BinTree("3"), new BinTree("4"));
        BinTree minus = knoten("-", new BinTree("5"), new BinTree("2"));
        return knoten("*", plus, minus);
    }

    // Startbaum des Akinator-Monsterquiz: innere Knoten = Fragen,
    // Blaetter = Monster; ja -> links, nein -> rechts.
    public static BinTree monsterquiz() {
        BinTree fliegt = knoten("Speit es Feuer?", new BinTree("Drache"), new BinTree("Greif"));
        BinTree laeuft = knoten("Lebt es im Wasser?", new BinTree("Kraken"), new BinTree("Troll"));
        return knoten("Kann es fliegen?", fliegt, laeuft);
    }
}


class Testhilfe {

    // Klammerdarstellung eines Baums (fuer die Meldungen der Tests):
    //   leerer Baum "-", Blatt "7", sonst inhalt(links,rechts)
    //   beispielBST -> "8(3(1,6(4,7)),10(-,14(13,-)))"
    public static String klammer(BinTree b) {
        if (b == null) {
            return "null";
        }
        if (b.isEmpty()) {
            return "-";
        }
        if (b.isLeaf()) {
            return "" + b.getItem();
        }
        return b.getItem() + "(" + klammer(b.getLeft()) + "," + klammer(b.getRight()) + ")";
    }

    // Inhalt einer dynamischen Reihung als Text, z. B. "[1, 3, 4]".
    public static String text(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + d.getItem(i);
        }
        return t + "]";
    }

    // Neue Zeichenkette mit demselben Inhalt wie s (Zeichen fuer Zeichen
    // zusammengesetzt). Damit pruefen die Tests, dass du Zeichenketten mit
    // equals vergleichst und nicht mit ==.
    public static String kopie(String s) {
        String t = "";
        for (int i = 0; i < s.length(); i++) {
            t = t + s.charAt(i);
        }
        return t;
    }

    // Die Tests fangen Laufzeitfehler (Exceptions) ab und melden sie mit
    // einem Hinweis. In einer richtigen Loesung tritt NIE ein Laufzeitfehler
    // auf - auch nicht beim leeren Baum!
    public static void laufzeitfehler(RuntimeException e) {
        String meldung = " (Meldung der IDE: " + e.getMessage() + ")";
        try {
            throw e;
        } catch (IllegalStateException x) {
            fail("LAUFZEITFEHLER: getItem, getLeft, getRight, isLeaf, setLeft oder setRight auf einem LEEREN Baum - Basisfall if (b.isEmpty()) vergessen oder an die falsche Stelle gesetzt?" + meldung);
        } catch (IllegalArgumentException x) {
            fail("LAUFZEITFEHLER: null an setItem, setLeft oder setRight bzw. an append uebergeben - nimm new BinTree() fuer einen leeren Teilbaum." + meldung);
        } catch (IndexOutOfBoundsException x) {
            fail("LAUFZEITFEHLER: Index ausserhalb - bei charAt/substring gilt 0 <= i < length(), bei Reihungen 0 <= i < length. Schleifengrenze bzw. lo/hi pruefen!" + meldung);
        } catch (ClassCastException x) {
            fail("LAUFZEITFEHLER: falscher Cast - (Integer) fuer Zahlenbaeume, (String) fuer Termbaum und Monsterquiz, (BinTree) fuer die Schlange in levelorder." + meldung);
        } catch (NullPointerException x) {
            fail("LAUFZEITFEHLER: Zugriff auf null - Attribut im Konstruktor nicht erzeugt? Liefert eine Methode noch null statt new BinTree() bzw. new DynArray()?" + meldung);
        } catch (RuntimeException x) {
            fail("LAUFZEITFEHLER" + meldung);
        }
    }
}

            assertEquals("8(3(1,6(4,7)),10(-,14(13,-)))", Testhilfe.klammer(b), "baueAus({8, 3, 10, 1, 6, 14, 4, 7, 13}): beispielBST");
            int[] kette = {1, 2, 3, 4, 5};
            assertEquals(Testhilfe.klammer(Baumschule.entartet()), Testhilfe.klammer(Suchbaum.baueAus(kette)), "baueAus({1, 2, 3, 4, 5}): sortierte Werte -> entarteter Baum");

            assertTrue(leer.isEmpty(), "baueAus mit leerer Reihung: leerer Baum");

    @Test
    public void experte_istBST() {
        try {
            assertTrue(Suchbaum.istBST(Baumschule.beispielBST()), "istBST(beispielBST): true");
            assertTrue(Suchbaum.istBST(Baumschule.entartet()), "istBST(entartet): true");
            assertTrue(Suchbaum.istBST(Baumschule.blatt(0)), "istBST eines Blatts: true");
            assertTrue(Suchbaum.istBST(Baumschule.leer()), "istBST des leeren Baums: true");
            assertFalse(Suchbaum.istBST(Baumschule.zahlenbaum()), "istBST(zahlenbaum): false (9 links von 5)");
            BinTree falle1 = Baumschule.knoten(8, Baumschule.knoten(3, Baumschule.blatt(1), Baumschule.blatt(9)), Baumschule.blatt(10));
            assertFalse(Suchbaum.istBST(falle1), "istBST(8(3(1,9),10)): false - 9 > 3 passt zum Elternknoten, steht aber im LINKEN Teilbaum der 8 (nur Kinder pruefen reicht nicht!)");

            assertFalse(Suchbaum.istBST(Baumschule.knoten(5, Baumschule.leer(), Baumschule.blatt(5))), "istBST(5(-,5)): false - keine Duplikate");
            assertTrue(Suchbaum.istBST(Baumschule.knoten(0, Baumschule.leer(), Baumschule.blatt(1))), "istBST(0(-,1)): true (die 0 ist erlaubt)");
