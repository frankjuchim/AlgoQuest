//  Baumschule: die Testbaeume der Welt 5 (in allen Leveln, auf den
//              Arbeitsblaettern und im Boss benutzt). Jeder Aufruf
//              liefert einen NEUEN Baum - du darfst ihn veraendern.
//  Testhilfe:  zeigt Baeume und dynamische Reihungen als Text (fuer die

//  Du darfst beide Klassen auch im Hauptprogramm benutzen.

//  Die Testbaeume (ja = links, nein = rechts beim Monsterquiz):
//
//  beispielBST()            zahlenbaum()            termbaum()
//  (BST: 8, 3, 10, 1, 6,    (KEIN BST,              "(3+4)*(5-2)"
//   14, 4, 7, 13 eingefuegt) Duplikat 9)
//          8                       5                     *
//        /   \                   /   \                 /   \
//       3     10                9     4               +     -
//      / \      \              / \     \             / \   / \
//     1   6      14           2   7     9           3   4 5   2
//        / \    /                      /
//       4   7  13                     1
//
//  entartet(): 1 -> 2 -> 3 -> 4 -> 5 (jeder Knoten hat nur ein rechtes Kind)
//
//  monsterquiz():                 Kann es fliegen?
//                              ja /              \ nein
//                    Speit es Feuer?          Lebt es im Wasser?
//                    ja /     \ nein          ja /         \ nein
//                   Drache    Greif         Kraken         Troll
//
//  leer(): leerer Baum      blatt(x): ein einzelnes Blatt mit dem Wert x

class Baumschule {

    // knoten(inhalt: Object, links: BinTree, rechts: BinTree): BinTree
    // Neuer Knoten mit dem Inhalt und den beiden Teilbaeumen.
    public static BinTree knoten(Object inhalt, BinTree links, BinTree rechts) {
        BinTree b = new BinTree(inhalt);
        b.setLeft(links);
        b.setRight(rechts);
        return b;
    }

    // blatt(x: Ganzzahl): BinTree  - ein Knoten mit zwei leeren Teilbaeumen
    public static BinTree blatt(int x) {
        return new BinTree(x);
    }

    // leer(): BinTree  - der leere Baum
    public static BinTree leer() {
        return new BinTree();
    }

    // Der Beispiel-Suchbaum: entsteht durch Einfuegen von
    // 8, 3, 10, 1, 6, 14, 4, 7, 13 (in dieser Reihenfolge).
    public static BinTree beispielBST() {
        BinTree links = knoten(3, blatt(1), knoten(6, blatt(4), blatt(7)));
        BinTree rechts = knoten(10, leer(), knoten(14, blatt(13), leer()));
        return knoten(8, links, rechts);
    }

    // Ein Zahlenbaum, der KEIN Suchbaum ist (9 kommt zweimal vor).
    public static BinTree zahlenbaum() {
        BinTree links = knoten(9, blatt(2), blatt(7));
        BinTree rechts = knoten(4, leer(), knoten(9, blatt(1), leer()));
        return knoten(5, links, rechts);
    }

    // Entarteter Suchbaum: 1, 2, 3, 4, 5 eingefuegt -> rechte Kette.
    public static BinTree entartet() {
        BinTree b = blatt(5);
        for (int i = 4; i >= 1; i--) {
            b = knoten(i, leer(), b);
        }
        return b;
    }

    // Termbaum zu (3+4)*(5-2): Inhaltstyp Zeichenkette.
    // Innere Knoten: Rechenzeichen "+", "-", "*", "/"; Blaetter: Zahlen.
    public static BinTree termbaum() {
        BinTree plus = knoten("+", new BinTree("3"), new BinTree("4"));
        BinTree minus = knoten("-", new BinTree("5"), new BinTree("2"));
        return knoten("*", plus, minus);
    }

    // Startbaum des Akinator-Monsterquiz: innere Knoten = Fragen,
    // Blaetter = Monster; ja -> links, nein -> rechts.
    public static BinTree monsterquiz() {
        BinTree fliegt = knoten("Speit es Feuer?", new BinTree("Drache"), new BinTree("Greif"));
        BinTree laeuft = knoten("Lebt es im Wasser?", new BinTree("Kraken"), new BinTree("Troll"));
        return knoten("Kann es fliegen?", fliegt, laeuft);
    }
}


class Testhilfe {

    // Klammerdarstellung eines Baums (fuer die Meldungen der Tests):
    //   leerer Baum "-", Blatt "7", sonst inhalt(links,rechts)
    //   beispielBST -> "8(3(1,6(4,7)),10(-,14(13,-)))"
    public static String klammer(BinTree b) {
        if (b == null) {
            return "null";
        }
        if (b.isEmpty()) {
            return "-";
        }
        if (b.isLeaf()) {
            return "" + b.getItem();
        }
        return b.getItem() + "(" + klammer(b.getLeft()) + "," + klammer(b.getRight()) + ")";
    }

    // Inhalt einer dynamischen Reihung als Text, z. B. "[1, 3, 4]".
    public static String text(DynArray d) {
        if (d == null) {
            return "null";
        }
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + d.getItem(i);
        }
        return t + "]";
    }

    // Neue Zeichenkette mit demselben Inhalt wie s (Zeichen fuer Zeichen
    // zusammengesetzt). Damit pruefen die Tests, dass du Zeichenketten mit
    // equals vergleichst und nicht mit ==.
    public static String kopie(String s) {
        String t = "";
        for (int i = 0; i < s.length(); i++) {
            t = t + s.charAt(i);
        }
        return t;
    }

    // Die Tests fangen Laufzeitfehler (Exceptions) ab und melden sie mit
    // einem Hinweis. In einer richtigen Loesung tritt NIE ein Laufzeitfehler
    // auf - auch nicht beim leeren Baum!
    public static void laufzeitfehler(RuntimeException e) {
        String meldung = " (Meldung der IDE: " + e.getMessage() + ")";
        try {
            throw e;
        } catch (IllegalStateException x) {
            fail("LAUFZEITFEHLER: getItem, getLeft, getRight, isLeaf, setLeft oder setRight auf einem LEEREN Baum - Basisfall if (b.isEmpty()) vergessen oder an die falsche Stelle gesetzt?" + meldung);
        } catch (IllegalArgumentException x) {
            fail("LAUFZEITFEHLER: null an setItem, setLeft oder setRight bzw. an append uebergeben - nimm new BinTree() fuer einen leeren Teilbaum." + meldung);
        } catch (IndexOutOfBoundsException x) {
            fail("LAUFZEITFEHLER: Index ausserhalb - bei charAt/substring gilt 0 <= i < length(), bei Reihungen 0 <= i < length. Schleifengrenze bzw. lo/hi pruefen!" + meldung);
        } catch (ClassCastException x) {
            fail("LAUFZEITFEHLER: falscher Cast - (Integer) fuer Zahlenbaeume, (String) fuer Termbaum und Monsterquiz, (BinTree) fuer die Schlange in levelorder." + meldung);
        } catch (NullPointerException x) {
            fail("LAUFZEITFEHLER: Zugriff auf null - Attribut im Konstruktor nicht erzeugt? Liefert eine Methode noch null statt new BinTree() bzw. new DynArray()?" + meldung);
        } catch (RuntimeException x) {
            fail("LAUFZEITFEHLER" + meldung);
        }
    }
}

//      getItem(): Inhaltstyp        setItem(o: Inhaltstyp): nichts
//      getLeft(): BinTree           setLeft(b: BinTree): nichts
//      getRight(): BinTree          setRight(b: BinTree): nichts
//      setEmpty(): nichts
//  Ein NICHT leerer Baum hat IMMER zwei Teilbaeume (evtl. leere Baeume,
//  nie null). getItem, getLeft, getRight, isLeaf, setLeft und setRight
//  auf einem LEEREN Baum -> Laufzeitfehler. setItem auf einem leeren
//  Baum macht ihn zum Blatt (mit zwei leeren Teilbaeumen).
//
//  Muster "REKURSION AUF BAEUMEN" (Beispiel: alle Werte zaehlen, die
//  gerade sind):
//
//      public static int anzahlGerade(BinTree b) {
//          if (b.isEmpty()) {                  // 1. Basisfall: leerer Baum
//              return 0;
//          }
//          int wert = (Integer) b.getItem();   // 2. Wurzel auswerten (Cast!)
//          int anzahl = 0;
//          if (wert % 2 == 0) {
//              anzahl = 1;
//          }
//          return anzahl + anzahlGerade(b.getLeft())     // 3. Teilbaeume
//                        + anzahlGerade(b.getRight());   //    rekursiv
//      }
//
//  Traversierungen als Zeichenkette: jedes Element gefolgt von EINEM
//  Leerzeichen, z. B. "8 3 1 6 4 7 10 14 13 "; leerer Baum -> "".
//  Zeichenkette + Object geht direkt: b.getItem() + " "
// ---------------------------------------------------------------------

// ============ Level 5.1 – Baeume ablaufen (DS 31) ===================
//  Alle Methoden sind static: Aufruf mit Klassenname, z. B.
//  Baum.anzahlKnoten(b). ALLE Methoden (ausser zahl) sind rekursiv und
//  beginnen mit dem Basisfall if (b.isEmpty()).

class Baum {

    // ===================== Stufe BASIS =====================

    // anzahlKnoten(b: BinTree): Ganzzahl
    // Anzahl aller Knoten. Leerer Baum -> 0.
    public static int anzahlKnoten(BinTree b) {
        if (b.isEmpty()) {
            return 0;                                   // Basisfall
        }
        return 1 + anzahlKnoten(b.getLeft()) + anzahlKnoten(b.getRight());
    }

    // preorder(b: BinTree): Zeichenkette
    // Wurzel - links - rechts. beispielBST -> "8 3 1 6 4 7 10 14 13 "
    public static String preorder(BinTree b) {
        if (b.isEmpty()) {
            return "";
        }
        return b.getItem() + " " + preorder(b.getLeft()) + preorder(b.getRight());
    }

    // inorder(b: BinTree): Zeichenkette
    // links - Wurzel - rechts. beispielBST -> "1 3 4 6 7 8 10 13 14 "
    public static String inorder(BinTree b) {
        if (b.isEmpty()) {
            return "";
        }
        return inorder(b.getLeft()) + b.getItem() + " " + inorder(b.getRight());
    }

    // postorder(b: BinTree): Zeichenkette
    // links - rechts - Wurzel. beispielBST -> "1 4 7 6 3 13 14 10 8 "
    public static String postorder(BinTree b) {
        if (b.isEmpty()) {
            return "";
        }
        return postorder(b.getLeft()) + postorder(b.getRight()) + b.getItem() + " ";
    }


    // ================== Stufe ERWEITERUNG ==================

    // hoehe(b: BinTree): Ganzzahl
    // Anzahl der Ebenen: leerer Baum -> 0, Blatt -> 1,
    // sonst 1 + groessere Hoehe der beiden Teilbaeume.
    public static int hoehe(BinTree b) {
        if (b.isEmpty()) {
            return 0;
        }
        int links = hoehe(b.getLeft());
        int rechts = hoehe(b.getRight());
        if (links > rechts) {
            return 1 + links;
        }
        return 1 + rechts;
    }

    // anzahlBlaetter(b: BinTree): Ganzzahl
    // Anzahl der Knoten ohne Kinder. Leerer Baum -> 0.
    public static int anzahlBlaetter(BinTree b) {
        if (b.isEmpty()) {
            return 0;
        }
        if (b.isLeaf()) {
            return 1;                                   // isLeaf erst NACH isEmpty!
        }
        return anzahlBlaetter(b.getLeft()) + anzahlBlaetter(b.getRight());
    }

    // summe(b: BinTree vom Inhaltstyp Ganzzahl): Ganzzahl
    // Summe aller Werte. Leerer Baum -> 0.
    public static int summe(BinTree b) {
        if (b.isEmpty()) {
            return 0;
        }
        int wert = (Integer) b.getItem();               // Cast: Object -> Integer
        return wert + summe(b.getLeft()) + summe(b.getRight());
    }


    // =================== Stufe EXPERTE =====================

    // zahl(s: Zeichenkette): Ganzzahl
    // Wandelt eine Ziffernfolge in eine Ganzzahl um: "12" -> 12.
    // NUR mit length() und charAt() - KEIN Integer.parseInt (im Abitur
    // nicht erlaubt). Tipp: (int) c - (int) '0' ist der Wert der Ziffer c.
    // Diese Hilfsmethode ist NICHT rekursiv (Schleife).
    public static int zahl(String s) {
        int z = 0;
        for (int i = 0; i < s.length(); i++) {
            int ziffer = (int) s.charAt(i) - (int) '0';
            z = z * 10 + ziffer;                        // "907": 9 -> 90 -> 907
        }
        return z;
    }

    // berechne(t: BinTree vom Inhaltstyp Zeichenkette): Ganzzahl
    // Wert eines Termbaums: Blaetter sind Zahlen ("3", "12"), innere
    // Knoten die Rechenzeichen "+", "-", "*", "/" (ganzzahlig).
    // termbaum() -> (3+4)*(5-2) = 21. Leerer Baum -> 0.
    public static int berechne(BinTree t) {
        if (t.isEmpty()) {
            return 0;
        }
        String s = (String) t.getItem();
        if (t.isLeaf()) {
            return zahl(s);                             // Blatt: eine Zahl
        }
        int links = berechne(t.getLeft());
        int rechts = berechne(t.getRight());
        if (s.equals("+")) {                            // equals, nicht ==
            return links + rechts;
        }
        if (s.equals("-")) {
            return links - rechts;                      // Reihenfolge: links - rechts
        }
        if (s.equals("*")) {
            return links * rechts;
        }
        if (s.equals("/")) {
            return links / rechts;                      // ganzzahlig: 7 / 2 = 3
        }
        return 0;                                       // unbekanntes Zeichen
    }

    // spiegeln(b: BinTree): nichts
    // Vertauscht in JEDEM Knoten den linken und den rechten Teilbaum
    // (mit setLeft/setRight). Der Baum wird dabei veraendert.
    public static void spiegeln(BinTree b) {
        if (b.isEmpty()) {
            return;
        }
        BinTree links = b.getLeft();                    // merken, sonst ist er weg
        b.setLeft(b.getRight());
        b.setRight(links);
        spiegeln(b.getLeft());
        spiegeln(b.getRight());
    }
}


// ========== Level 5.2 – Rekursive Baumalgorithmen (DS 32) ===========
//  Zahlenbaeume (BinTree vom Inhaltstyp Ganzzahl, Werte >= 0), KEIN
//  Suchbaum: Ein Wert kann ueberall stehen, auch mehrfach. Alle
//  Methoden sind rekursiv (die Methoden mit dem Parameter d sind
//  rekursive Hilfsmethoden).

class Rekursiv {

    // ===================== Stufe BASIS =====================

    // enthaelt(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Wahrheitswert
    // true, wenn x irgendwo im Baum vorkommt.
    public static boolean enthaelt(BinTree b, int x) {
        if (b.isEmpty()) {
            return false;
        }
        int wert = (Integer) b.getItem();
        if (wert == x) {
            return true;
        }
        return enthaelt(b.getLeft(), x) || enthaelt(b.getRight(), x);
    }

    // maximum(b: BinTree vom Inhaltstyp Ganzzahl): Ganzzahl
    // Groesster Wert im Baum. Leerer Baum -> -1.
    public static int maximum(BinTree b) {
        if (b.isEmpty()) {
            return -1;                                  // kleiner als jeder Wert >= 0
        }
        int max = (Integer) b.getItem();
        int links = maximum(b.getLeft());
        int rechts = maximum(b.getRight());
        if (links > max) {
            max = links;
        }
        if (rechts > max) {
            max = rechts;
        }
        return max;
    }

    // anzahlGroesser(b: BinTree vom Inhaltstyp Ganzzahl, grenze: Ganzzahl): Ganzzahl
    // Anzahl der Werte, die GROESSER als grenze sind.
    public static int anzahlGroesser(BinTree b, int grenze) {
        if (b.isEmpty()) {
            return 0;
        }
        int anzahl = 0;
        int wert = (Integer) b.getItem();
        if (wert > grenze) {
            anzahl = 1;
        }
        return anzahl + anzahlGroesser(b.getLeft(), grenze) + anzahlGroesser(b.getRight(), grenze);
    }


    // ================== Stufe ERWEITERUNG ==================

    // sammleInorder(b: BinTree): dynamische Reihung vom Inhaltstyp Ganzzahl
    // Liefert eine NEUE DynArray mit allen Werten in Inorder-Reihenfolge.
    // beispielBST -> [1, 3, 4, 6, 7, 8, 10, 13, 14]. Leerer Baum -> leere
    // DynArray (nicht null). Erzeuge die DynArray hier und lass die
    // rekursive Hilfsmethode darunter die Werte anhaengen.
    public static DynArray sammleInorder(BinTree b) {
        DynArray d = new DynArray();
        sammleInorder(b, d);
        return d;
    }

    // sammleInorder(b: BinTree, d: dynamische Reihung): nichts
    // Rekursive Hilfsmethode: haengt die Werte von b in Inorder an d an.
    public static void sammleInorder(BinTree b, DynArray d) {
        if (b.isEmpty()) {
            return;
        }
        sammleInorder(b.getLeft(), d);
        d.append(b.getItem());                          // Wurzel ZWISCHEN den Teilbaeumen
        sammleInorder(b.getRight(), d);
    }

    // blaetter(b: BinTree): dynamische Reihung vom Inhaltstyp Ganzzahl
    // NEUE DynArray mit den Werten aller Blaetter von links nach rechts.
    // beispielBST -> [1, 4, 7, 13]. Leerer Baum -> leere DynArray.
    public static DynArray blaetter(BinTree b) {
        DynArray d = new DynArray();
        blaetter(b, d);
        return d;
    }

    // blaetter(b: BinTree, d: dynamische Reihung): nichts
    // Rekursive Hilfsmethode: haengt die Blattwerte von b an d an.
    public static void blaetter(BinTree b, DynArray d) {
        if (b.isEmpty()) {
            return;
        }
        if (b.isLeaf()) {
            d.append(b.getItem());
            return;
        }
        blaetter(b.getLeft(), d);
        blaetter(b.getRight(), d);
    }

    // anzahlAufEbene(b: BinTree, ebene: Ganzzahl): Ganzzahl
    // Anzahl der Knoten auf der Ebene (Wurzel = Ebene 0).
    // Idee: Ebene e im Baum = Ebene e - 1 in den beiden Teilbaeumen.
    public static int anzahlAufEbene(BinTree b, int ebene) {
        if (b.isEmpty()) {
            return 0;
        }
        if (ebene == 0) {
            return 1;                                   // die Wurzel selbst
        }
        return anzahlAufEbene(b.getLeft(), ebene - 1) + anzahlAufEbene(b.getRight(), ebene - 1);
    }


    // =================== Stufe EXPERTE =====================

    // pfadZu(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Zeichenkette
    // Weg von der Wurzel zum ERSTEN Vorkommen von x in Preorder als Folge
    // von 'L' (links) und 'R' (rechts). zahlenbaum: 7 -> "LR", 1 -> "RRL".
    // x in der Wurzel -> "", x kommt nicht vor -> "-".
    public static String pfadZu(BinTree b, int x) {
        if (b.isEmpty()) {
            return "-";
        }
        int wert = (Integer) b.getItem();
        if (wert == x) {
            return "";
        }
        String links = pfadZu(b.getLeft(), x);
        if (!links.equals("-")) {
            return "L" + links;                         // links zuerst (Preorder)
        }
        String rechts = pfadZu(b.getRight(), x);
        if (!rechts.equals("-")) {
            return "R" + rechts;
        }
        return "-";
    }

    // istGleich(a: BinTree, b: BinTree): Wahrheitswert   (Inhaltstyp Ganzzahl)
    // true, wenn beide Baeume dieselbe Struktur UND an jeder Stelle
    // denselben Wert haben. Zwei leere Baeume sind gleich.
    public static boolean istGleich(BinTree a, BinTree b) {
        if (a.isEmpty() && b.isEmpty()) {
            return true;
        }
        if (a.isEmpty() || b.isEmpty()) {
            return false;                               // nur einer ist leer
        }
        int x = (Integer) a.getItem();
        int y = (Integer) b.getItem();
        if (x != y) {
            return false;
        }
        return istGleich(a.getLeft(), b.getLeft()) && istGleich(a.getRight(), b.getRight());
    }
}


// ========== Level 5.3 – Akinator: der Baum, der lernt (DS 33, Kuer) ==
//  Objekt-Klasse: Entscheidungsbaum vom Inhaltstyp Zeichenkette.
//  Innere Knoten = Ja/Nein-Fragen, Blaetter = Monsternamen.
//  Antwort 'j' -> linker Teilbaum, sonst (z. B. 'n') -> rechter Teilbaum.
//
//  Klassenkarte:
//      Akinator
//      ------------------------------------------------------------
//      - baum: BinTree                          // Inhaltstyp Zeichenkette
//      ------------------------------------------------------------
//      + c Akinator(start: Zeichenkette)        // ein Blatt
//      + c Akinator(b: BinTree)                 // VORGEGEBEN
//      + getBaum(): BinTree                     // VORGEGEBEN
//      + raten(antworten: Zeichenkette): Zeichenkette
//      + lernen(antworten: Zeichenkette, neu: Zeichenkette,
//               frage: Zeichenkette, jaFuerNeu: Wahrheitswert): nichts
//      + anzahlMonster(): Ganzzahl
//      + anzahlFragen(): Ganzzahl
//      + levelorder(b: BinTree): Zeichenkette   // static
//  Die Hilfsmethoden (private static, rekursiv) bekommen den Baum als
//  Parameter - so geht Rekursion auch in einer Objekt-Klasse.

class Akinator {

    private BinTree baum;

    // ===================== Stufe BASIS =====================

    // c Akinator(start: Zeichenkette)
    // Der Akinator kennt am Anfang nur ein Monster: baum ist ein Blatt
    // mit dem Namen start.
    public Akinator(String start) {
        baum = new BinTree(start);
    }

    // c Akinator(b: BinTree)  - VORGEGEBEN: startet mit einem fertigen Baum,
    // z. B. new Akinator(Baumschule.monsterquiz())
    public Akinator(BinTree b) {
        baum = b;
    }

    // getBaum(): BinTree  - VORGEGEBEN
    public BinTree getBaum() {
        return baum;
    }

    // raten(antworten: Zeichenkette): Zeichenkette
    // Geht von der Wurzel aus je Zeichen von antworten einen Schritt
    // ('j' links, sonst rechts), bis ein Blatt erreicht ist, und liefert
    // dessen Namen. Ueberzaehlige Antworten werden ignoriert, reichen die
    // Antworten nicht bis zu einem Blatt -> "?".
    // monsterquiz: "nj" -> "Kraken", "nnj" -> "Troll", "j" -> "?"
    public String raten(String antworten) {
        return raten(baum, antworten);
    }

    // raten(b: BinTree, antworten: Zeichenkette): Zeichenkette
    // Rekursive Hilfsmethode: antworten.charAt(0) ist die Antwort auf die
    // Frage in b, der Rest antworten.substring(1, antworten.length())
    // gehoert zum Teilbaum.
    private static String raten(BinTree b, String antworten) {
        if (b.isEmpty()) {
            return "?";
        }
        if (b.isLeaf()) {
            return (String) b.getItem();                // Monster gefunden
        }
        if (antworten.length() == 0) {
            return "?";                                 // Frage, aber keine Antwort mehr
        }
        String rest = antworten.substring(1, antworten.length());
        if (antworten.charAt(0) == 'j') {
            return raten(b.getLeft(), rest);
        }
        return raten(b.getRight(), rest);
    }


    // ================== Stufe ERWEITERUNG ==================

    // lernen(antworten: Zeichenkette, neu: Zeichenkette, frage: Zeichenkette,
    //        jaFuerNeu: Wahrheitswert): nichts
    // Der Akinator hat falsch geraten: Das ueber antworten erreichte Blatt
    // (Name alt) wird zum Frageknoten frage mit den Kindern neu und alt.
    // jaFuerNeu = true: neu links (ja), alt rechts; sonst umgekehrt.
    // Ueberzaehlige Antworten werden ignoriert; reichen die Antworten
    // nicht bis zu einem Blatt, passiert nichts.
    public void lernen(String antworten, String neu, String frage, boolean jaFuerNeu) {
        lernen(baum, antworten, neu, frage, jaFuerNeu);
    }

    // lernen(b: BinTree, antworten: Zeichenkette, neu: Zeichenkette,
    //        frage: Zeichenkette, jaFuerNeu: Wahrheitswert): nichts
    // Rekursive Hilfsmethode (wie raten, aber am Blatt wird umgebaut).
    private static void lernen(BinTree b, String antworten, String neu, String frage, boolean jaFuerNeu) {
        if (b.isEmpty()) {
            return;
        }
        if (b.isLeaf()) {
            String alt = (String) b.getItem();
            b.setItem(frage);                           // Blatt wird Frageknoten
            if (jaFuerNeu) {
                b.setLeft(new BinTree(neu));
                b.setRight(new BinTree(alt));
            } else {
                b.setLeft(new BinTree(alt));
                b.setRight(new BinTree(neu));
            }
            return;
        }
        if (antworten.length() == 0) {
            return;                                     // zu wenige Antworten
        }
        String rest = antworten.substring(1, antworten.length());
        if (antworten.charAt(0) == 'j') {
            lernen(b.getLeft(), rest, neu, frage, jaFuerNeu);
        } else {
            lernen(b.getRight(), rest, neu, frage, jaFuerNeu);
        }
    }

    // anzahlMonster(): Ganzzahl  - Anzahl der Blaetter von baum
    public int anzahlMonster() {
        return monster(baum);
    }

    // monster(b: BinTree): Ganzzahl  - rekursive Hilfsmethode (Blaetter zaehlen)
    private static int monster(BinTree b) {
        if (b.isEmpty()) {
            return 0;
        }
        if (b.isLeaf()) {
            return 1;
        }
        return monster(b.getLeft()) + monster(b.getRight());
    }

    // anzahlFragen(): Ganzzahl  - Anzahl der inneren Knoten von baum
    public int anzahlFragen() {
        return fragen(baum);
    }

    // fragen(b: BinTree): Ganzzahl  - rekursive Hilfsmethode (innere Knoten)
    private static int fragen(BinTree b) {
        if (b.isEmpty()) {
            return 0;
        }
        if (b.isLeaf()) {
            return 0;                                   // ein Blatt ist keine Frage
        }
        return 1 + fragen(b.getLeft()) + fragen(b.getRight());
    }


    // =================== Stufe EXPERTE =====================

    // levelorder(b: BinTree): Zeichenkette   (static, NICHT rekursiv)
    // Ebene fuer Ebene, jede Ebene von links nach rechts, Format wie die
    // anderen Traversierungen. beispielBST -> "8 3 10 1 6 14 4 7 13 ".
    // Benutze eine Schlange vom Inhaltstyp BinTree:
    //     BinTree k = (BinTree) q.dequeue();
    public static String levelorder(BinTree b) {
        String text = "";
        Queue q = new Queue();
        if (!b.isEmpty()) {
            q.enqueue(b);                               // nur nicht leere Baeume einreihen
        }
        while (!q.isEmpty()) {
            BinTree k = (BinTree) q.dequeue();
            text = text + k.getItem() + " ";
            if (!k.getLeft().isEmpty()) {
                q.enqueue(k.getLeft());
            }
            if (!k.getRight().isEmpty()) {
                q.enqueue(k.getRight());
            }
        }
        return text;
    }
}


// =========== Level 5.4 – Binaere Suchbaeume (DS 34 – 35) ============
//  BST vom Inhaltstyp Ganzzahl: Fuer JEDEN Knoten gilt: alle Werte im
//  linken Teilbaum < Wert < alle Werte im rechten Teilbaum. Duplikate
//  werden nicht eingefuegt. Die Methoden duerfen das ausnutzen: Suchen
//  geht in jedem Schritt nur in EINEN Teilbaum.

class Suchbaum {

    // ===================== Stufe BASIS =====================

    // suche(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Wahrheitswert
    // Rekursiv: in jedem Schritt NUR in den passenden Teilbaum absteigen.
    public static boolean suche(BinTree b, int x) {
        if (b.isEmpty()) {
            return false;                               // hier muesste x stehen
        }
        int wert = (Integer) b.getItem();
        if (x == wert) {
            return true;
        }
        if (x < wert) {
            return suche(b.getLeft(), x);
        }
        return suche(b.getRight(), x);
    }

    // minimum(b: BinTree vom Inhaltstyp Ganzzahl): Ganzzahl
    // Kleinster Wert: immer links, bis es links nicht weitergeht.
    // Leerer Baum -> -1.
    public static int minimum(BinTree b) {
        if (b.isEmpty()) {
            return -1;
        }
        if (b.getLeft().isEmpty()) {
            return (Integer) b.getItem();               // links ist nichts mehr
        }
        return minimum(b.getLeft());
    }

    // einfuegen(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): nichts
    // Rekursiv: Ist b leer, wird x hier eingefuegt (setItem macht den
    // leeren Baum zum Blatt). Sonst links bzw. rechts weiter. Ist x schon
    // vorhanden, passiert nichts.
    public static void einfuegen(BinTree b, int x) {
        if (b.isEmpty()) {
            b.setItem(x);                               // leerer Teilbaum -> Blatt
            return;
        }
        int wert = (Integer) b.getItem();
        if (x < wert) {
            einfuegen(b.getLeft(), x);
        } else if (x > wert) {
            einfuegen(b.getRight(), x);
        }
        // x == wert: Duplikat - nichts tun
    }


    // ================== Stufe ERWEITERUNG ==================

    // sucheIterativ(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Wahrheitswert
    // Wie suche, aber mit einer Schleife statt Rekursion.
    public static boolean sucheIterativ(BinTree b, int x) {
        BinTree k = b;
        while (!k.isEmpty()) {
            int wert = (Integer) k.getItem();
            if (x == wert) {
                return true;
            }
            if (x < wert) {
                k = k.getLeft();
            } else {
                k = k.getRight();
            }
        }
        return false;
    }

    // vergleiche(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Ganzzahl
    // Anzahl der Knoten, die die Suche nach x untersucht (bis x gefunden
    // ist bzw. bis die Suche auf einen leeren Baum trifft).
    // beispielBST: 7 -> 4 (8, 3, 6, 7); 5 -> 4 (8, 3, 6, 4, dann leer).
    public static int vergleiche(BinTree b, int x) {
        if (b.isEmpty()) {
            return 0;                                   // leerer Baum: kein Knoten
        }
        int wert = (Integer) b.getItem();
        if (x == wert) {
            return 1;
        }
        if (x < wert) {
            return 1 + vergleiche(b.getLeft(), x);
        }
        return 1 + vergleiche(b.getRight(), x);
    }

    // baueAus(r: Reihung von Ganzzahl): BinTree
    // NEUER Suchbaum, die Werte werden in der Reihenfolge von r eingefuegt.
    // {8, 3, 10, 1, 6, 14, 4, 7, 13} -> beispielBST.
    public static BinTree baueAus(int[] r) {
        BinTree b = new BinTree();                      // leerer Baum, nicht null
        for (int i = 0; i < r.length; i++) {
            einfuegen(b, r[i]);
        }
        return b;
    }


    // =================== Stufe EXPERTE =====================

    // istBST(b: BinTree vom Inhaltstyp Ganzzahl): Wahrheitswert
    // true, wenn b ein binaerer Suchbaum ist. ACHTUNG: Es reicht NICHT,
    // jeden Knoten nur mit seinen Kindern zu vergleichen! Jeder Wert muss
    // zwischen den Grenzen liegen, die alle Vorfahren vorgeben.
    public static boolean istBST(BinTree b) {
        return istBST(b, -1, -1);                       // untere Grenze -1, keine obere
    }

    // istBST(b: BinTree, min: Ganzzahl, max: Ganzzahl): Wahrheitswert
    // Rekursive Hilfsmethode: Alle Werte in b muessen GROESSER als min und
    // KLEINER als max sein. max = -1 bedeutet: keine obere Grenze.
    // (Werte sind >= 0, also ist min = -1 "keine untere Grenze".)
    public static boolean istBST(BinTree b, int min, int max) {
        if (b.isEmpty()) {
            return true;
        }
        int wert = (Integer) b.getItem();
        if (wert <= min) {
            return false;
        }
        if (max != -1 && wert >= max) {
            return false;
        }
        return istBST(b.getLeft(), min, wert) && istBST(b.getRight(), wert, max);
    }

    // ausgeglichenAus(sortiert: Reihung von Ganzzahl): BinTree
    // Baut aus einer aufsteigend sortierten Reihung einen moeglichst
    // niedrigen Suchbaum: Das mittlere Element wird die Wurzel, die
    // linke Haelfte der linke, die rechte Haelfte der rechte Teilbaum.
    // Leere Reihung -> leerer Baum (nicht null).
    public static BinTree ausgeglichenAus(int[] sortiert) {
        return ausgeglichenAus(sortiert, 0, sortiert.length - 1);
    }

    // ausgeglichenAus(r: Reihung von Ganzzahl, lo: Ganzzahl, hi: Ganzzahl): BinTree
    // Rekursive Hilfsmethode fuer den Bereich r[lo] .. r[hi]:
    // mitte = (lo + hi) / 2. Basisfall: lo > hi -> leerer Baum.
    public static BinTree ausgeglichenAus(int[] r, int lo, int hi) {
        if (lo > hi) {
            return new BinTree();                       // Bereich leer -> leerer Baum
        }
        int mitte = (lo + hi) / 2;
        BinTree b = new BinTree(r[mitte]);
        b.setLeft(ausgeglichenAus(r, lo, mitte - 1));
        b.setRight(ausgeglichenAus(r, mitte + 1, hi));
        return b;
    }

    // nachfolger(b: BinTree vom Inhaltstyp Ganzzahl, x: Ganzzahl): Ganzzahl
    // Kleinster Wert im BST, der GROESSER als x ist; -1, wenn es keinen
    // gibt. (x muss nicht im Baum stehen.) Vorbereitung auf das Loeschen:
    // nachfolger(beispielBST, 8) = 10 ersetzt beim Loeschen die 8.
    public static int nachfolger(BinTree b, int x) {
        if (b.isEmpty()) {
            return -1;
        }
        int wert = (Integer) b.getItem();
        if (wert <= x) {
            return nachfolger(b.getRight(), x);         // Kandidaten nur rechts
        }
        int kandidat = nachfolger(b.getLeft(), x);      // wert > x: links gibt es evtl. kleinere
        if (kandidat == -1) {
            return wert;                                // links nichts Passendes: wert selbst
        }
        return kandidat;
    }
}

            assertEquals("8(3(1,6(4,7)),10(-,14(13,-)))", Testhilfe.klammer(b), "nachfolger darf den Baum NICHT veraendern");
        } catch (RuntimeException e) {
            Testhilfe.laufzeitfehler(e);
        }
    }
}

// =====================================================================
//  HÄUFIGE SCHÜLERFEHLER (Lehrerhinweise) – in Klammern der Test, der
//  den Fehler aufdeckt (diese Zuordnungen sind per Mutationstest geprueft:
//  114 typische Fehler, jeder macht mindestens einen Test rot oder
//  verhindert das Kompilieren)
// =====================================================================
//
//  QUER DURCH ALLE LEVEL
//  - BASISFALL VERGESSEN (kein if (b.isEmpty())): Spaetestens bei den
//    leeren Teilbaeumen eines Blatts ruft die Methode getItem/getLeft auf
//    einem leeren Baum auf -> "LAUFZEITFEHLER: ... auf einem LEEREN Baum
//    - Basisfall if (b.isEmpty()) vergessen?" (basis_anzahlKnoten,
//    experte_berechne, experte_spiegeln, erweiterung_blaetter,
//    basis_suche u. a.). Merksatz: "Erst fragen, ob da was ist."
//  - isLeaf VOR isEmpty: isLeaf auf einem leeren Baum ist ebenfalls ein
//    Laufzeitfehler (erweiterung_anzahlBlaetter, erweiterung_anzahlMonster-
//    Fragen). Reihenfolge immer: leer? -> Blatt? -> innerer Knoten.
//  - CAST VERGESSEN -> Kompilierfehler ("Object kann nicht ...
//    umgewandelt werden"): int wert = b.getItem() (summe), return
//    b.getItem() in raten, BinTree k = q.dequeue() in levelorder.
//  - "+1" FUER DIE WURZEL VERGESSEN: anzahlKnoten, hoehe, vergleiche
//    (basis_anzahlKnoten, erweiterung_hoehe, erweiterung_vergleiche).
//  - LINKS/RECHTS VERTAUSCHT bzw. einen Teilbaum vergessen: preorder ohne
//    rechten Teilbaum, inorder rechts vor links, enthaelt nur links,
//    maximum ohne linken Teilbaum ("im BST steht das Maximum rechts" -
//    der Zahlenbaum ist aber KEIN BST!), blaetter rechts vor links
//    (basis_preorder, basis_inorder, basis_enthaelt, basis_maximum,
//    erweiterung_blaetter).
//  - null STATT LEERER STRUKTUR: Basisfall liefert null statt "" bzw.
//    sammleInorder/baueAus liefern bei leerem Baum/leerer Reihung null
//    (basis_preorder, erweiterung_sammleInorder, erweiterung_baueAus).
//
//  Level 5.1 – Baum
//  - Leerzeichen vergessen oder VOR das Element gesetzt (" 8 3 1 ...")
//    -> basis_preorder. Format: Element, dann EIN Leerzeichen.
//  - postorder = preorder (Wurzel zuerst) (basis_postorder).
//  - hoehe: Minimum statt Maximum, Summe statt Maximum, nur links,
//    Kantenzahl (leer -> -1, Blatt -> 0) (erweiterung_hoehe; der Baum
//    1(2(3,-),-) entlarvt "nur rechts"/"nur links").
//  - anzahlBlaetter: Blatt-Fall vergessen (liefert immer 0) oder "ein
//    leeres Kind reicht" (Knoten mit nur einem Kind als Blatt gezaehlt:
//    1(-,2(3,4)) -> 1 statt 2) (erweiterung_anzahlBlaetter). Achtung: Bei
//    beispielBST, zahlenbaum und entartet liefert dieser Fehler zufaellig
//    die richtige Zahl!
//  - summe ohne Wurzelwert (erweiterung_summe).
//  - zahl: Zeichencode statt Ziffer ((int) '7' = 55, "- (int) '0'"
//    vergessen), Ziffern nur addiert (Quersumme), nur erste Ziffer
//    (experte_zahl, experte_berechne).
//  - berechne: links und rechts vertauscht (5-2 -> -3, Ergebnis -21),
//    Blatt-Fall vergessen (Zahl wird als Rechenzeichen behandelt -> 0),
//    * und / verwechselt (experte_berechne).
//  - spiegeln ohne Zwischenspeichern: b.setLeft(b.getRight());
//    b.setRight(b.getLeft()); -> beide Seiten gleich; nur die Wurzel
//    getauscht (Rekursion vergessen) (experte_spiegeln).
//
//  Level 5.2 – Rekursiv
//  - enthaelt mit && statt || oder "wie im Suchbaum" nur in EINEN
//    Teilbaum (basis_enthaelt: 7 und 1 stehen "auf der falschen Seite").
//  - anzahlGroesser mit >= statt > (basis_anzahlGroesser: Grenze 9).
//  - sammleInorder: Wurzel vor dem linken Teilbaum angehaengt (Preorder),
//    Hilfsmethode nicht aufgerufen, EINE DynArray als static-Attribut fuer
//    alle Aufrufe (zweiter Aufruf enthaelt die alten Werte)
//    (erweiterung_sammleInorder).
//  - blaetter haengt alle Knoten an (erweiterung_blaetter).
//  - anzahlAufEbene: ebene + 1 statt ebene - 1, Wurzel als Ebene 1,
//    Knoten auf dem Weg mitgezaehlt (erweiterung_anzahlAufEbene).
//  - pfadZu: rechts zuerst gesucht (zahlenbaum, 9 -> "RR" statt "L"),
//    "-" des Teilbaums nicht abgefangen ("L-"), Buchstabe hinten
//    angehaengt (Weg rueckwaerts: 13 -> "LRR" statt "RRL") (experte_pfadZu).
//  - istGleich: nur Preorder/Werte verglichen (1(2,-) und 1(-,2) haben
//    dieselbe Preorder), "nur einer leer" nicht abgefangen (Laufzeit-
//    fehler), || statt && (experte_istGleich).
//
//  Level 5.3 – Akinator
//  - Konstruktor: baum = new BinTree() (ohne start) oder gar nicht
//    erzeugt (basis_konstruktor, erweiterung_lernen).
//  - raten: 'j' nach rechts, Laenge nicht geprueft (charAt(0) auf "" ->
//    "LAUFZEITFEHLER: Index ausserhalb"), Rest falsch gebildet
//    (substring(0, length() - 1) schneidet die LETZTE Antwort ab),
//    Laenge VOR isLeaf geprueft (raten("") bei nur einem Monster liefert
//    "?" statt "Drache") (basis_raten, basis_raten_randfaelle,
//    basis_konstruktor).
//  - lernen: jaFuerNeu falsch herum, setItem(frage) vergessen, den alten
//    Namen erst NACH setItem(frage) gelesen (dann steht die Frage auch
//    im Blatt), ganze Antwort statt Rest weitergegeben, bei zu wenig
//    Antworten einen inneren Knoten ueberschrieben (erweiterung_lernen,
//    erweiterung_lernen_tief).
//  - anzahlMonster zaehlt alle Knoten, anzahlFragen zaehlt Blaetter mit
//    (erweiterung_anzahlMonsterFragen).
//  - levelorder mit Stapel statt Schlange (= Preorder mit rechts zuerst:
//    "8 10 14 13 3 6 7 4 1 "), rechts vor links eingereiht, leere
//    Teilbaeume eingereiht und dann getItem auf leerem Baum
//    (experte_levelorder).
//
//  Level 5.4 – Suchbaum
//  - suche: < und > vertauscht, Gleichheit nicht geprueft, BEIDE
//    Teilbaeume durchsucht (basis_suche_nurEinWeg - nur mit dem
//    ungueltigen Baum 8(3,10(5,-)) erkennbar, siehe oben).
//  - minimum: nach rechts gelaufen oder "bis zum Blatt" statt "bis links
//    nichts mehr ist" (5(2(-,3),9): Minimum 2 ist kein Blatt) (basis_minimum).
//  - einfuegen: b = new BinTree(x) statt b.setItem(x) - der neue Baum
//    haengt nirgends im Baum, es wird NICHTS eingefuegt (basis_einfuegen,
//    basis_einfuegen_leer); else statt else if -> Duplikate rechts
//    eingefuegt (basis_einfuegen); links/rechts vertauscht.
//  - sucheIterativ: Hilfsvariable im else-Zweig nicht weitergesetzt ->
//    ENDLOSSCHLEIFE (erweiterung_sucheIterativ -> TIMEOUT bzw. haengender
//    Testrunner).
//  - vergleiche: leerer Baum zaehlt 1, Fund zaehlt 0, +1 in einem Zweig
//    vergessen (erweiterung_vergleiche).
//  - baueAus: Schleife ab 1, Werte von hinten eingefuegt (erweiterung_
//    baueAus).
//  - istBST NUR MIT DEN KINDERN verglichen: 8(3(1,9),10) gilt dann als
//    Suchbaum, obwohl 9 im linken Teilbaum der 8 steht (experte_istBST,
//    DIE klassische Falle). Ebenso: <= statt < (Duplikate),
//    Grenzen im rechten Aufruf vertauscht, Grenze nicht weitergegeben.
//  - ausgeglichenAus: Mitte nicht ausgelassen (lo..mitte) -> unendliche
//    Rekursion (Stapelueberlauf); Basisfall lo >= hi -> Blaetter fehlen;
//    hi = length (Index ausserhalb); mitte = hi / 2 (experte_ausgeglichenAus).
//  - nachfolger: linken Teilbaum ignoriert (4 -> 8 statt 6), < statt <=
//    (liefert x selbst), -1 aus dem linken Teilbaum nicht abgefangen,
//    nur "Minimum des rechten Teilbaums" (fuer 7 gibt es keinen rechten
//    Teilbaum - der Nachfolger 8 steht WEITER OBEN) (experte_nachfolger).
//
//  CODE-REVIEW (die Tests erkennen das NICHT – per Mutationstest belegt)
//  - == STATT EQUALS bei den Rechenzeichen: im Pruef-Harness rot
//    (experte_berechne_equals), in der ONLINE-IDE aber GRUEN, weil sie
//    Zeichenketten auch mit == inhaltlich vergleicht. Im Abitur falsch!
//  - Vorgabe "rekursiv" missachtet: raten, suche (oder andere) mit einer
//    Schleife geloest -> alle Tests gruen. Im Code-Review nachfragen;
//    es gilt die Vorgabe (Abi-Aufgaben verlangen oft ausdruecklich
//    "rekursiv").
//  - Gleichwertige, richtige Alternativen (gruen und in Ordnung):
//    hoehe mit Math.max; blaetter ohne return nach dem Anhaengen (die
//    leeren Kinder tun nichts); levelorder mit Einreihen aller Teilbaeume
//    und Pruefung nach dem Entnehmen; istBST ueber sammleInorder
//    ("Inorder streng aufsteigend"); einfuegen in der Abi-Variante
//    "ist links leer? dann setLeft(new BinTree(x)), sonst rekursiv".
//
//  MUSTER-TRACES (fuer AB 1 bis 3, Warm-up und Boss 5)
//
//  Levelorder beispielBST mit Schlange (vorne -> hinten):
//      Start                     Schlange [8]
//      entnehme 8   Ausgabe "8 "                   Schlange [3, 10]
//      entnehme 3   Ausgabe "8 3 "                 Schlange [10, 1, 6]
//      entnehme 10  Ausgabe "8 3 10 "              Schlange [1, 6, 14]
//      entnehme 1   Ausgabe "8 3 10 1 "            Schlange [6, 14]
//      entnehme 6   Ausgabe "8 3 10 1 6 "          Schlange [14, 4, 7]
//      entnehme 14  Ausgabe "8 3 10 1 6 14 "       Schlange [4, 7, 13]
//      entnehme 4   Ausgabe "8 3 10 1 6 14 4 "     Schlange [7, 13]
//      entnehme 7   Ausgabe "8 3 10 1 6 14 4 7 "   Schlange [13]
//      entnehme 13  Ausgabe "8 3 10 1 6 14 4 7 13 " Schlange []
//
//  hoehe(beispielBST) - Rueckgabewerte von unten nach oben:
//      Blaetter 1, 4, 7, 13: je 1 + max(0, 0) = 1
//      6: 1 + max(1, 1) = 2      14: 1 + max(1, 0) = 2
