class Ladevorgang {

    private String kennzeichen;
    private int energie;   // in kWh
    private int dauer;     // in Minuten

    public Ladevorgang(String kennzeichen, int energie, int dauer) {
        this.kennzeichen = kennzeichen;
        this.energie = energie;
        this.dauer = dauer;
    }

    public String getKennzeichen() {
        return kennzeichen;
    }

    public int getEnergie() {
        return energie;
    }

    public int getDauer() {
        return dauer;
    }

    public String toString() {
        return kennzeichen + " (" + energie + " kWh, " + dauer + " min)";
    }
}

class Ladesaeule {

    private int nummer;
    private int leistung;          // maximale Ladeleistung in kW
    private Ladevorgang aktuell;   // laufender Ladevorgang, null, wenn die Saeule frei ist

    public Ladesaeule(int nummer) {
        this.nummer = nummer;
        this.leistung = 11;
        this.aktuell = null;
    }

    public Ladesaeule(int nummer, int leistung) {
        this.nummer = nummer;
        this.leistung = leistung;
        this.aktuell = null;
    }

    public int getNummer() {
        return nummer;
    }

    public int getLeistung() {
        return leistung;
    }

    public boolean istFrei() {
        return aktuell == null;
    }

    public void starten(Ladevorgang v) {
        aktuell = v;
    }

    public Ladevorgang beenden() {
        Ladevorgang v = aktuell;
        aktuell = null;
        return v;
    }

    // Preis pro kWh in Cent
    public int preisProKWh() {
        return 49;
    }
}

class Ladepark {

    private int[][] energie;           // [Saeule][Stunde], geladene Energie in kWh
    private BinTree karten;            // binaerer Suchbaum vom Inhaltstyp Ganzzahl
    private int netzleistung;          // kWh, die pro Stunde hoechstens aus dem Netz bezogen werden
    private int kapazitaet;            // Kapazitaet des Pufferspeichers in kWh

    public Ladepark(int netzleistung, int kapazitaet) {
        this.netzleistung = netzleistung;
        this.kapazitaet = kapazitaet;
        saeulen = new DynArray();
        warteschlange = new Queue();
        protokoll = new DynArray();
        energie = new int[1][24];
        karten = new BinTree();
    }

        saeulen.append(s);
    }

    }

    public int[][] getEnergie() {

    }

    public void setEnergie(int[][] e) {
        energie = e;
    }

    public BinTree getKarten() {

    }

    // =================================================================
    //  VORGEGEBEN in der Klausur (Struktogramme Abbildung 4, 7 und 10),
    //  hier zeilengetreu in Java – zur Kontrolle von f), i) und m)
    // =================================================================

    // Abbildung 4: sortiert protokoll absteigend nach der Energie
    public void sortiereProtokoll() {
        int pos = 1;
        while (pos < protokoll.getLength()) {
            if (pos == 0) {
                pos = pos + 1;
            } else {
                Ladevorgang links = (Ladevorgang) protokoll.getItem(pos - 1);
                Ladevorgang rechts = (Ladevorgang) protokoll.getItem(pos);
                if (links.getEnergie() < rechts.getEnergie()) {
                    protokoll.setItem(pos - 1, rechts);
                    protokoll.setItem(pos, links);
                    pos = pos - 1;
                } else {
                    pos = pos + 1;
                }
            }
        }
    }

    // Abbildung 7: erste Stunde mit Ueberlast oder -1
    public int ueberlast(int[] bedarf, int h, int puffer) {
        if (h == bedarf.length) {
            return -1;
        }
        puffer = puffer + netzleistung - bedarf[h];
        if (puffer < 0) {
            return h;
        }
        if (puffer > kapazitaet) {
            puffer = kapazitaet;
        }
        return ueberlast(bedarf, h + 1, puffer);
    }

    // Abbildung 10: Anzahl der Kartennummern x mit von <= x <= bis im BST b
    public int anzahlImBereich(BinTree b, int von, int bis) {
        if (b.isEmpty()) {
            return 0;
        }
        int x = (Integer) b.getItem();
        if (x < von) {
            return anzahlImBereich(b.getRight(), von, bis);
        }
        if (x > bis) {
            return anzahlImBereich(b.getLeft(), von, bis);
        }
        return 1 + anzahlImBereich(b.getLeft(), von, bis) + anzahlImBereich(b.getRight(), von, bis);
    }

    // =================================================================
    //  HIER DIE HANDSCHRIFT UNVERAENDERT ABTIPPEN (Teilaufgaben c, d, e, n)
    // =================================================================

    // Teilaufgabe c) besteFreieSaeule(): Ladesaeule

    // Teilaufgabe d) wartezeit(kennzeichen: Zeichenkette): Ganzzahl
    public int wartezeit(String kennzeichen) {
        // TODO Handschrift abtippen
        return -2;
    }

    // Teilaufgabe e) spitzenstunde(): Ganzzahl  (Struktogramm Zeile fuer Zeile uebertragen)
    public int spitzenstunde() {
        // TODO Struktogramm uebertragen
        return -2;
    }

    // Teilaufgabe n) entferneVorgaenge(kennzeichen: Zeichenkette): Ganzzahl
    // So steht die Operation in Abbildung 11 der Klausur (mit drei Fehlern).
    // Die KORRIGIERTE Fassung des Prueflings hier abtippen.
    public int entferneVorgaenge(String kennzeichen) {
        int anzahl = 0;
        for (int i = 0; i <= protokoll.getLength(); i++) {
            Ladevorgang v = (Ladevorgang) protokoll.getItem(i);
            if (v.getKennzeichen() == kennzeichen) {
                protokoll.delete(i);
                anzahl++;
            }
        }
        return anzahl;
    }

class Testhilfe {

    // Zeichenkette zur Laufzeit neu zusammensetzen: In echtem Java faellt
    // == statt equals damit auf. ACHTUNG: Die Online-IDE vergleicht
    // Zeichenketten auch mit == inhaltlich -> dort von Hand pruefen (P5)!
    public static String kopie(String s) {
        String k = "";
        for (int i = 0; i < s.length(); i++) {
            k = k + s.charAt(i);
        }
        return k;
    }

    // Schlange aus Ladevorgaengen: r[0] steht vorne.
    public static Queue schlange(Ladevorgang[] r) {
        Queue q = new Queue();
        for (int i = 0; i < r.length; i++) {
            q.enqueue(r[i]);
        }
        return q;
    }

    // true, wenn q GENAU die Objekte aus r in dieser Reihenfolge enthaelt.
    // q ist danach wieder wie vorher.
    public static boolean gleicheSchlange(Queue q, Ladevorgang[] r) {
        Queue hilf = new Queue();
        int n = 0;
        boolean ok = true;
        while (!q.isEmpty()) {
            Object o = q.dequeue();
            if (n >= r.length || o != r[n]) {
                ok = false;
            }
            n++;
            hilf.enqueue(o);
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return ok && n == r.length;
    }

    // Kennzeichen der dynamischen Reihung als Text, z. B. "[OL-EV 7, DEL-LP 42]"
    public static String kennzeichen(DynArray d) {
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + ((Ladevorgang) d.getItem(i)).getKennzeichen();
        }
        return t + "]";
    }

    // Kartennummer in den binaeren Suchbaum b einfuegen (Duplikate ignorieren)
    public static void einfuegen(BinTree b, int x) {
        if (b.isEmpty()) {
            b.setItem(x);
        } else if (x < (Integer) b.getItem()) {
            einfuegen(b.getLeft(), x);
        } else if (x > (Integer) b.getItem()) {
            einfuegen(b.getRight(), x);
        }
    }

    public static Ladepark park() {
        return new Ladepark(100, 120);
    }

    public static void laufzeitfehler(RuntimeException e) {
        String meldung = " (Meldung: " + e.getMessage() + ")";
        try {
            throw e;
        } catch (IllegalStateException x) {
            fail("LAUFZEITFEHLER: head/dequeue/top/pop/getItem auf LEERER Struktur - isEmpty() vergessen? (P1)" + meldung);
        } catch (IndexOutOfBoundsException x) {
            fail("LAUFZEITFEHLER: Index ausserhalb - 0 <= i < getLength() bzw. < length; geloescht und danach i erhoeht? (P1/P6)" + meldung);
        } catch (IllegalArgumentException x) {
            fail("LAUFZEITFEHLER: null in Queue oder DynArray gelegt" + meldung);
        } catch (ClassCastException x) {
            fail("LAUFZEITFEHLER: falscher Cast (P4)" + meldung);
        } catch (NullPointerException x) {
            fail("LAUFZEITFEHLER: Zugriff auf null - z. B. beste.getLeistung(), solange beste noch null ist (P1)" + meldung);
        } catch (RuntimeException x) {
            fail("LAUFZEITFEHLER" + meldung);
        }
    }
}
