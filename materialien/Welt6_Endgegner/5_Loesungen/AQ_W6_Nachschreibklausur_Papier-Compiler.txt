class Route {

    private String name;
    private int grad;          // Schwierigkeitsgrad 3 bis 11
    private String sektor;

    public Route(String name, int grad, String sektor) {
        this.name = name;
        this.grad = grad;
        this.sektor = sektor;
    }

    public String getName() {
        return name;
    }

    public int getGrad() {
        return grad;
    }

    public String getSektor() {
        return sektor;
    }
}


class Mitglied {

    private String name;
    private int punkte;
    private Stack tourenbuch;  // Stapel vom Inhaltstyp Route, oben: zuletzt geklettert

    public Mitglied(String name) {
        this.name = name;
        this.punkte = 0;
        this.tourenbuch = new Stack();
    }

    public Mitglied(String name, int punkte) {
        this.name = name;
        this.punkte = punkte;
        this.tourenbuch = new Stack();
    }

    public String getName() {
        return name;
    }

    public int getPunkte() {
        return punkte;
    }

    public void klettern(Route r) {
        tourenbuch.push(r);
        punkte = punkte + r.getGrad();
    }

    // nur fuer die Pruefhilfe (nicht im Klassendiagramm der Klausur)
    public Stack getTourenbuch() {
        return tourenbuch;
    }

    // >>> ABTIPPEN c) hoechsterGrad - Handschrift unveraendert hier einsetzen
    public int hoechsterGrad(int k) {
        return -99;   // Platzhalter
    }
    // <<< ENDE c)
}


class Station {

    private int nummer;
    private Route route;
    private Queue warteliste;  // Schlange vom Inhaltstyp Mitglied

    public Station(int nummer, Route route) {
        this.nummer = nummer;
        this.route = route;
        this.warteliste = new Queue();
    }

    public void anstellen(Mitglied m) {
        warteliste.enqueue(m);
    }

    public Mitglied naechster() {
        if (warteliste.isEmpty()) {
            return null;
        }
        return (Mitglied) warteliste.dequeue();
    }

    // nur fuer die Pruefhilfe (nicht im Klassendiagramm der Klausur)
    public Queue getWarteliste() {
        return warteliste;
    }

    // >>> ABTIPPEN d) abmelden - Java-Uebersetzung des Struktogramms
    public boolean abmelden(String name) {
        return false;   // Platzhalter
    }
    // <<< ENDE d)
}

    // Abbildung 7 der Klausur (vorgegeben, nicht zu implementieren)
    public int aufstieg(int z, int s) {
        if (z == 0) {
            return wand[z][s];
        }
        int ns = s;
        if (s > 0 && wand[z - 1][s - 1] > wand[z - 1][ns]) {
            ns = s - 1;
        }
        if (s < wand[0].length - 1 && wand[z - 1][s + 1] > wand[z - 1][ns]) {
            ns = s + 1;
        }
        if (wand[z - 1][ns] == 0) {
            return wand[z][s];
        }
        return wand[z][s] + aufstieg(z - 1, ns);
    }

    // Abbildung 10 der Klausur (vorgegeben, nicht zu implementieren);

        for (int i = 0; i <= n - 2; i++) {
            int pos = i;
            for (int j = i + 1; j <= n - 1; j++) {
                if (t[j].getPunkte() > t[pos].getPunkte()) {
                    pos = j;
                }
            }
            Mitglied hilf = t[i];
            t[i] = t[pos];
            t[pos] = hilf;
            println("i = " + i + ", pos = " + pos + ":  " + Pruefhilfe.reihungText(t));
        }
    }

    public Route schwersteRoute(String sektor) {

            Route r = (Route) routen.getItem(i);
            if (r.getSektor() == sektor && r.getGrad() > beste.getGrad()) {
                beste = r;
            }
        }
        return beste;

// Mitglied, das sich merkt, ob getPunkte() aufgerufen wurde (fuer m)
class ZaehlMitglied extends Mitglied {

    private boolean besucht;

    public ZaehlMitglied(String name, int punkte) {
        super(name, punkte);
        besucht = false;
    }

    public int getPunkte() {
        besucht = true;
        return super.getPunkte();
    }

    public boolean istBesucht() {
        return besucht;
    }

    public void zuruecksetzen() {
        besucht = false;
    }
}

    // Einfuegen in den Suchbaum (Vergleich ueber das Attribut, OHNE
    // getPunkte, damit die Besucht-Markierung nicht verfaelscht wird)
    private static void einfuegen(BinTree b, ZaehlMitglied m) {
        if (b.isEmpty()) {
            b.setItem(m);
        } else {
            ZaehlMitglied w = (ZaehlMitglied) b.getItem();
            if (punkteOhneMarke(m) < punkteOhneMarke(w)) {
                einfuegen(b.getLeft(), m);
            } else {
                einfuegen(b.getRight(), m);
            }
        }
    }

    private static int punkteOhneMarke(ZaehlMitglied m) {
        boolean vorher = m.istBesucht();
        int p = m.getPunkte();
        if (!vorher) {
            m.zuruecksetzen();
        }
        return p;
    }

    // Namen aller Mitglieder, bei denen getPunkte() aufgerufen wurde (Preorder)
    public static String besucht(BinTree b) {
        if (b.isEmpty()) {
            return "";
        }
        String t = "";
        Object o = b.getItem();
        if (o instanceof ZaehlMitglied && ((ZaehlMitglied) o).istBesucht()) {
            t = ((ZaehlMitglied) o).getName() + " ";
        }
        return t + besucht(b.getLeft()) + besucht(b.getRight());
    }

    public static void alleZuruecksetzen(BinTree b) {
        if (!b.isEmpty()) {
            ((ZaehlMitglied) b.getItem()).zuruecksetzen();
            alleZuruecksetzen(b.getLeft());
            alleZuruecksetzen(b.getRight());
        }
    }

    // Name eines Mitglieds oder "null" (fuer die Meldungen der Tests)
    public static String name(Mitglied m) {
        if (m == null) {
            return "null";
        }
        return m.getName();
    }

    // Inhalt des Suchbaums in Inorder als Text (Struktur-Kontrolle)
    public static String inorder(BinTree b) {
        if (b.isEmpty()) {
            return "";
        }
        return inorder(b.getLeft()) + ((Mitglied) b.getItem()).getName() + " " + inorder(b.getRight());
    }

    // ----- Auslesen, OHNE zu veraendern -----

    // Tourenbuch von oben nach unten, z. B. "[Sandmann, Spinne]"
    public static String stapelText(Stack s) {
        Stack hilf = new Stack();
        String t = "[";
        while (!s.isEmpty()) {
            Object o = s.pop();
            if (hilf.isEmpty()) {
                t = t + ((Route) o).getName();
            } else {
                t = t + ", " + ((Route) o).getName();
            }
            hilf.push(o);
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return t + "]";
    }

    // true, wenn s von oben nach unten GENAU die Objekte erwartet[0], erwartet[1], ...
    // enthaelt (dieselben Objekte, dieselbe Reihenfolge); s bleibt unveraendert
    public static boolean stapelGleich(Stack s, Route[] erwartet) {
        Stack hilf = new Stack();
        boolean gleich = true;
        int i = 0;
        while (!s.isEmpty()) {
            Object o = s.pop();
            if (i >= erwartet.length || o != erwartet[i]) {
                gleich = false;
            }
            hilf.push(o);
            i++;
        }
        while (!hilf.isEmpty()) {
            s.push(hilf.pop());
        }
        return gleich && i == erwartet.length;
    }

    // Warteliste von vorn nach hinten als Namen
    public static String schlangeText(Queue q) {
        Queue hilf = new Queue();
        String t = "[";
        boolean erstes = true;
        while (!q.isEmpty()) {
            Object o = q.dequeue();
            if (erstes) {
                t = t + ((Mitglied) o).getName();
                erstes = false;
            } else {
                t = t + ", " + ((Mitglied) o).getName();
            }
            hilf.enqueue(o);
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return t + "]";
    }

    // true, wenn q von vorn GENAU die Objekte erwartet[0], ... enthaelt
    public static boolean schlangeGleich(Queue q, Mitglied[] erwartet) {
        Queue hilf = new Queue();
        boolean gleich = true;
        int i = 0;
        while (!q.isEmpty()) {
            Object o = q.dequeue();
            if (i >= erwartet.length || o != erwartet[i]) {
                gleich = false;
            }
            hilf.enqueue(o);
            i++;
        }
        while (!hilf.isEmpty()) {
            q.enqueue(hilf.dequeue());
        }
        return gleich && i == erwartet.length;
    }

    // Routen einer dynamischen Reihung als Namen
    public static String reihungText(DynArray d) {
        String t = "[";
        for (int i = 0; i < d.getLength(); i++) {
            if (i > 0) {
                t = t + ", ";
            }
            t = t + ((Route) d.getItem(i)).getName();
        }
        return t + "]";
    }

    // Mitglieder einer statischen Reihung mit Punkten
    public static String reihungText(Mitglied[] t) {
        String s = "";
        for (int i = 0; i < t.length; i++) {
            if (i > 0) {
                s = s + ", ";
            }
            s = s + t[i].getName() + " " + t[i].getPunkte();
        }
        return s;
    }

    // Neue Zeichenkette mit demselben Inhalt (zur Laufzeit zusammengesetzt):
    // In echtem Java faellt damit == statt equals auf (P5).
    public static String kopie(String s) {
        String t = "";
        for (int i = 0; i < s.length(); i++) {
            t = t + s.charAt(i);
        }
