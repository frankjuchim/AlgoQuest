class Paket {

    private String nummer;        // Sendungsnummer, z. B. "P-4127"
    private String empfaenger;
    private char groesse;         // 'S', 'M' oder 'L'

    public Paket(String nummer, String empfaenger, char groesse) {
        this.nummer = nummer;
        this.empfaenger = empfaenger;
        this.groesse = groesse;
    }

    public String getNummer() {
        return nummer;
    }

    public String getEmpfaenger() {
        return empfaenger;
    }

    public char getGroesse() {
        return groesse;
    }

    // S: 1, M: 2, L: 3
    public int getVolumen() {
        if (groesse == 'S') {
            return 1;
        }
        if (groesse == 'M') {
            return 2;
        }
        return 3;
    }
}

class Fach {

    private int nummer;           // Fachnummer, z. B. 104
    private char groesse;         // 'S', 'M' oder 'L'
    private Paket inhalt;         // null, wenn das Fach frei ist

    public Fach(int nummer, char groesse) {
        this.nummer = nummer;
        this.groesse = groesse;
        this.inhalt = null;
    }

    public int getNummer() {
        return nummer;
    }

    public char getGroesse() {
        return groesse;
    }

    public boolean istFrei() {
        return inhalt == null;
    }

    // null, wenn das Fach frei ist
    public Paket getInhalt() {
        return inhalt;
    }

    // nur fuer ein freies Fach aufrufen
    public void einlegen(Paket p) {
        inhalt = p;
    }

    public Paket entnehmen() {
        Paket p = inhalt;
        inhalt = null;
        return p;
    }
}

class Paketstation {

    private String standort;
    private DynArray faecher;          // dynamische Reihung vom Inhaltstyp Fach
    private Queue auftraege;           // Schlange vom Inhaltstyp Paket
    private Stack ruecksendungen;      // Stapel vom Inhaltstyp Paket

    public Paketstation(String standort) {
        this.standort = standort;
        faecher = new DynArray();
        auftraege = new Queue();
        ruecksendungen = new Stack();
    }

    // erzeugt anzahlS Faecher der Groesse S, dann anzahlM der Groesse M,
    // dann anzahlL der Groesse L; Fachnummern fortlaufend ab 101
    public Paketstation(String standort, int anzahlS, int anzahlM, int anzahlL) {
        this(standort);
        int nr = 101;
        for (int i = 0; i < anzahlS; i++) {
            faecher.append(new Fach(nr, 'S'));
            nr++;
        }
        for (int i = 0; i < anzahlM; i++) {
            faecher.append(new Fach(nr, 'M'));
            nr++;
        }
        for (int i = 0; i < anzahlL; i++) {
            faecher.append(new Fach(nr, 'L'));
            nr++;
        }
    }

    public void fachHinzufuegen(Fach f) {
        faecher.append(f);
    }

    public void auftragAnnehmen(Paket p) {
        auftraege.enqueue(p);
    }

    public void ruecksendungAnnehmen(Paket p) {
        ruecksendungen.push(p);
    }

    public Stack getRuecksendungen() {
        return ruecksendungen;
    }

    // Abbildung 3 der Probeklausur (vorgegeben)
    public int zustellen() {
        int anzahl = 0;
        while (!auftraege.isEmpty()) {
            Paket p = (Paket) auftraege.dequeue();
            int i = 0;
            boolean gefunden = false;
            while (!gefunden && i < faecher.getLength()) {
                Fach f = (Fach) faecher.getItem(i);
                if (f.istFrei() && f.getGroesse() == p.getGroesse()) {
                    f.einlegen(p);
                    gefunden = true;
                    anzahl = anzahl + 1;
                } else {
                    i = i + 1;
                }
            }
            if (!gefunden) {
                ruecksendungen.push(p);
            }
        }
        return anzahl;
    }

    // Abbildung 5 der Probeklausur (vorgegeben, rekursiv)
    public int sammle(Stack s, int platz) {
        if (s.isEmpty()) {
            return 0;
        }
        Paket p = (Paket) s.top();
        if (p.getVolumen() > platz) {
            return 0;
        }
        s.pop();
        return 1 + sammle(s, platz - p.getVolumen());
    }

    // Testhilfe: Sendungsnummern der Auftraege von vorn nach hinten,
    // durch Leerzeichen getrennt, z. B. "P-4127 P-9054"; Schlange bleibt
    // unveraendert
    public String auftraegeAlsText() {
        String text = "";
        Queue hilf = new Queue();
        while (!auftraege.isEmpty()) {
            Paket p = (Paket) auftraege.dequeue();
            if (text.length() > 0) {
                text = text + " ";
            }
            text = text + p.getNummer();
            hilf.enqueue(p);
        }
        while (!hilf.isEmpty()) {
            auftraege.enqueue(hilf.dequeue());
        }
        return text;
    }

    // Testhilfe: Sendungsnummern der Ruecksendungen von oben nach unten;
    // Stapel bleibt unveraendert
    public String ruecksendungenAlsText() {
        String text = "";
        Stack hilf = new Stack();
        while (!ruecksendungen.isEmpty()) {
            Paket p = (Paket) ruecksendungen.pop();
            if (text.length() > 0) {
                text = text + " ";
            }
            text = text + p.getNummer();
            hilf.push(p);
        }
        while (!hilf.isEmpty()) {
            ruecksendungen.push(hilf.pop());
        }
        return text;
    }

    // =================================================================
    //  DEIN TEIL – hier arbeitest du!
    // =================================================================
    //  Tippe deine Handschrift UNVERAENDERT ab - auch mit Fehlern.
    //  Ersetze dabei jeweils die Zeile "// TODO ..." UND die vorlaeufige

    // AUFGABE B
    // sucheFach(sendungsnr: Zeichenkette): Ganzzahl
    // Fachnummer des Fachs, in dem das Paket mit der Sendungsnummer
    // liegt; -1, wenn es in keinem Fach liegt.
    public int sucheFach(String sendungsnr) {
        // TODO Aufgabe b) - deine Implementierung unveraendert abtippen

    }

    // AUFGABE C (Struktogramm -> Java)
    // anzahlAuftraege(groesse: Zeichen): Ganzzahl
    // Anzahl der Auftraege, deren Paket die Groesse groesse hat;
    // die Schlange auftraege ist danach UNVERAENDERT.
    // Uebersetze dein Struktogramm Block fuer Block:
    //   x <- wert             ->  x = wert;

    //   wenn ... wahr/falsch  ->  if (...) { ... } else { ... }
