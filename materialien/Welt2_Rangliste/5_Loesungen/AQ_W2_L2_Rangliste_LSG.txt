// ---------------------------------------------------------------------
//  Klassenkarte Eintrag (eine Zeile der Highscore-Liste):
//
//      Eintrag
//      ---------------------------------------------------------------
//      - name: Zeichenkette
//      - punkte: Ganzzahl
//      - zeit: Ganzzahl                          // Spielzeit in Sekunden
//      ---------------------------------------------------------------
//      + c Eintrag(name: Zeichenkette, punkte: Ganzzahl, zeit: Ganzzahl)
//      + getName(): Zeichenkette
//      + getPunkte(): Ganzzahl
//      + getZeit(): Ganzzahl
//      + beschreibung(): Zeichenkette            // "Aria: 1200 Punkte (95 s)"
// ---------------------------------------------------------------------

class Eintrag {

    private String name;
    private int punkte;
    private int zeit;

    public Eintrag(String name, int punkte, int zeit) {
        this.name = name;
        this.punkte = punkte;
        this.zeit = zeit;
    }

    public String getName() {
        return name;
    }

    public int getPunkte() {
        return punkte;
    }

    public int getZeit() {
        return zeit;
    }

    public String beschreibung() {
        return name + ": " + punkte + " Punkte (" + zeit + " s)";
    }
}


// ---------------------------------------------------------------------
//  LÖSUNGEN
// ---------------------------------------------------------------------
//  Klassenkarte Rangliste (Notation wie im Abitur):
//
//      Rangliste
//      ---------------------------------------------------------------
//      - eintraege: Reihung von Eintrag
//      - anzahl: Ganzzahl
//      - vergleiche: Ganzzahl
//      - vertauschungen: Ganzzahl
//      ---------------------------------------------------------------
//      + c Rangliste(kapazitaet: Ganzzahl)
//      + getAnzahl(): Ganzzahl
//      + getEintrag(i: Ganzzahl): Eintrag
//      + hinzufuegen(e: Eintrag): Wahrheitswert
//      + getVergleiche(): Ganzzahl
//      + getVertauschungen(): Ganzzahl
//      + zaehlerZuruecksetzen(): nichts
//      - besser(a: Eintrag, b: Eintrag): Wahrheitswert
//      + vertausche(i: Ganzzahl, j: Ganzzahl): nichts
//      + hoechstePunktzahl(): Ganzzahl                  Level 2.1
//      + besterEintrag(): Eintrag                       Level 2.1
//      + anzahlUeber(grenze: Ganzzahl): Ganzzahl        Level 2.1
//      + indexVon(name: Zeichenkette): Ganzzahl         Level 2.1
//      + durchschnitt(): Kommazahl                      Level 2.1
//      + hatDoppelteNamen(): Wahrheitswert              Level 2.1
//      + istSortiert(): Wahrheitswert                   Level 2.2
//      + selectionSort(): nichts                        Level 2.2
//      + insertionSort(): nichts                        Level 2.2
//      + bubbleSort(): nichts                           Level 2.2
//      + einfuegenSortiert(e: Eintrag): Wahrheitswert   Level 2.2
// ---------------------------------------------------------------------

class Rangliste {

    // ======== VORGEGEBEN – bis zur Linie "AB HIER ARBEITEST DU" nichts veraendern ========

    private Eintrag[] eintraege;
    private int anzahl;               // belegte Plaetze, zugleich naechster freier Index
    private int vergleiche;           // zaehlt die Aufrufe von besser(...)
    private int vertauschungen;       // zaehlt die Aufrufe von vertausche(...)

    // Leere Rangliste mit Platz fuer kapazitaet Eintraege.
    public Rangliste(int kapazitaet) {
        eintraege = new Eintrag[kapazitaet];
        anzahl = 0;
        vergleiche = 0;
        vertauschungen = 0;
    }

    public int getAnzahl() {
        return anzahl;
    }

    // Eintrag an Position i - oder null, wenn i kein gueltiger Index ist.
    public Eintrag getEintrag(int i) {
        if (i < 0 || i >= anzahl) {
            return null;
        }
        return eintraege[i];
    }

    // false, wenn e null ist oder die Liste voll ist. Sonst: e ans Ende,
    // anzahl erhoehen, true liefern. (Kennst du aus Welt 1: aufnehmen.)
    public boolean hinzufuegen(Eintrag e) {
        if (e == null || anzahl >= eintraege.length) {
            return false;
        }
        eintraege[anzahl] = e;
        anzahl++;
        return true;
    }

    public int getVergleiche() {
        return vergleiche;
    }

    public int getVertauschungen() {
        return vertauschungen;
    }

    public void zaehlerZuruecksetzen() {
        vergleiche = 0;
        vertauschungen = 0;
    }

    // besser(a: Eintrag, b: Eintrag): Wahrheitswert
    // true, wenn a MEHR Punkte hat als b - dann gehoert a weiter nach vorne.
    // Jeder Aufruf zaehlt einen Vergleich.
    // REGEL: Beim Sortieren und beim sortierten Einfuegen vergleichst du
    // Eintraege AUSSCHLIESSLICH mit besser(...) - nur so stimmen die Zaehler.
    private boolean besser(Eintrag a, Eintrag b) {
        vergleiche++;
        return a.getPunkte() > b.getPunkte();
    }

    // vertausche(i: Ganzzahl, j: Ganzzahl): nichts
    // Tauscht die Eintraege an den Positionen i und j (Dreieckstausch mit
    // Hilfsvariable) und zaehlt eine Vertauschung.
    // REGEL: Die drei Sortierverfahren vertauschen NUR mit vertausche(...).
    public void vertausche(int i, int j) {
        Eintrag hilf = eintraege[i];
        eintraege[i] = eintraege[j];
        eintraege[j] = hilf;
        vertauschungen++;
    }

    // ======== AB HIER ARBEITEST DU =============================================


    // ---------- Level 2.1: Standardbausteine (DS 13) ----------

    // Maximum-Baustein: mit dem ERSTEN Eintrag starten, nicht mit 0
    // (Punkte koennen negativ sein - Strafpunkte!).
    public int hoechstePunktzahl() {
        if (anzahl == 0) {
            return -1;
        }
        int max = eintraege[0].getPunkte();
        for (int i = 1; i < anzahl; i++) {
            if (eintraege[i].getPunkte() > max) {
                max = eintraege[i].getPunkte();
            }
        }
        return max;
    }

    // Wie oben, aber wir merken uns den Eintrag selbst.
    // ">" (nicht ">="): Bei Gleichstand bleibt der ERSTE.
    public Eintrag besterEintrag() {
        if (anzahl == 0) {
            return null;
        }
        Eintrag bester = eintraege[0];
        for (int i = 1; i < anzahl; i++) {
            if (eintraege[i].getPunkte() > bester.getPunkte()) {
                bester = eintraege[i];
            }
        }
        return bester;
    }

    // Zaehl-Baustein.
    public int anzahlUeber(int grenze) {
        int zaehler = 0;
        for (int i = 0; i < anzahl; i++) {
            if (eintraege[i].getPunkte() > grenze) {
                zaehler++;
            }
        }
        return zaehler;
    }

    // Lineare Suche: erstes Vorkommen, sofort zurueckgeben.
    // Das "return -1" steht NACH der Schleife, nicht in einem else.
    public int indexVon(String name) {
        for (int i = 0; i < anzahl; i++) {
            if (eintraege[i].getName().equals(name)) {
                return i;
            }
        }
        return -1;
    }

    // Summe als Ganzzahl, dann Kommazahl-Division durch den Cast.
    public double durchschnitt() {
        if (anzahl == 0) {
            return 0.0;
        }
        int summe = 0;
        for (int i = 0; i < anzahl; i++) {
            summe = summe + eintraege[i].getPunkte();
        }
        return (double) summe / anzahl;
    }

    // Jeder mit jedem: die innere Schleife beginnt bei i + 1, damit kein
    // Eintrag mit sich selbst (und kein Paar doppelt) verglichen wird.
    public boolean hatDoppelteNamen() {
        for (int i = 0; i < anzahl - 1; i++) {
            for (int j = i + 1; j < anzahl; j++) {
                if (eintraege[i].getName().equals(eintraege[j].getName())) {
                    return true;
                }
            }
        }
        return false;
    }


    // ---------- Level 2.2: Sortieren (DS 14) ----------

    // Jedes Nachbarpaar pruefen: "Fehlstand", wenn der rechte Nachbar MEHR
    // Punkte hat. Gleiche Nachbarn sind erlaubt. Kein besser(...) - zaehlt nicht.
    public boolean istSortiert() {
        for (int i = 0; i < anzahl - 1; i++) {
            if (eintraege[i].getPunkte() < eintraege[i + 1].getPunkte()) {
                return false;
            }
        }
        return true;
    }

    // Selectionsort mit Maximum-Auswahl:
    // Suche im unsortierten Rest i .. anzahl-1 den besten Eintrag und tausche
    // ihn an Position i. Vergleiche immer (n-1) + (n-2) + ... + 1 = n(n-1)/2.
    public void selectionSort() {
        for (int i = 0; i < anzahl - 1; i++) {
            int maxIndex = i;
            for (int j = i + 1; j < anzahl; j++) {
                if (besser(eintraege[j], eintraege[maxIndex])) {
                    maxIndex = j;
                }
            }
            if (maxIndex != i) {
                vertausche(i, maxIndex);
            }
        }
    }

    // Insertionsort: Der Eintrag an Position i wandert so lange nach links,
    // wie er BESSER ist als sein linker Nachbar.
    // j > 0 MUSS vor besser(...) stehen (Kurzschluss-Auswertung von &&).
    // Strikt "besser" -> gleich gute Eintraege werden nicht getauscht (stabil).
    public void insertionSort() {
        for (int i = 1; i < anzahl; i++) {
            int j = i;
            while (j > 0 && besser(eintraege[j], eintraege[j - 1])) {
                vertausche(j, j - 1);
                j--;
            }
        }
    }

    // Bubblesort mit Abbruch: Nach jedem Durchlauf steht der schlechteste
    // Eintrag des Restes ganz hinten -> innere Schleife wird um 1 kuerzer.
    // Blieb ein Durchlauf ohne Vertauschung, ist die Liste sortiert.
    public void bubbleSort() {
        boolean getauscht = true;
        int durchlauf = 0;
        while (getauscht) {
            getauscht = false;
            for (int j = 0; j < anzahl - 1 - durchlauf; j++) {
                if (besser(eintraege[j + 1], eintraege[j])) {
                    vertausche(j, j + 1);
                    getauscht = true;
                }
            }
            durchlauf++;
        }
    }

    // Sortiertes Einfuegen (wie ein Schritt von Insertionsort, aber mit
    // Verschieben statt Tauschen): Von hinten her jeden schlechteren
    // Eintrag eine Stelle nach rechts schieben, dann e in die Luecke.
    // Strikt "besser": Bei Gleichstand stoppen -> e landet HINTER gleich
    // guten Eintraegen (stabil).
    public boolean einfuegenSortiert(Eintrag e) {
        if (e == null || anzahl >= eintraege.length) {
            return false;
        }
        int i = anzahl;
        while (i > 0 && besser(e, eintraege[i - 1])) {
            eintraege[i] = eintraege[i - 1];
            i--;
        }
        eintraege[i] = e;
        anzahl++;
        return true;
    }
}


// ================== Level 2.3: Suche (DS 16) =========================

class Suche {

    // Zahl der Schleifendurchlaeufe der letzten Suche (zum Nachzaehlen).
    public static int durchlaeufe = 0;

    // Lineare Suche: jedes gepruefte Element zaehlt einen Durchlauf.
    public static int linear(int[] r, int x) {
        durchlaeufe = 0;
        for (int i = 0; i < r.length; i++) {
            durchlaeufe++;
            if (r[i] == x) {
                return i;
            }
        }
        return -1;
    }

    // Binaere Suche (iterativ) auf einer AUFSTEIGEND sortierten Reihung.
    // Der Suchbereich lo .. hi halbiert sich in jedem Durchlauf.
    public static int binaer(int[] r, int x) {
        durchlaeufe = 0;
        int lo = 0;
        int hi = r.length - 1;
        while (lo <= hi) {
            durchlaeufe++;
            int mid = (lo + hi) / 2;
            if (r[mid] == x) {
                return mid;
            }
            if (r[mid] < x) {
                lo = mid + 1;
            } else {
                hi = mid - 1;
            }
        }
        return -1;
    }

    // Experte: kleinster Index i mit r[i] >= x (r.length, falls x groesser
    // als alle ist). Der Bereich lo .. hi enthaelt immer die Antwort; hi darf
    // dabei r.length sein ("hinter dem letzten Element").
    public static int einfuegeposition(int[] r, int x) {
        durchlaeufe = 0;
        int lo = 0;
        int hi = r.length;
        while (lo < hi) {
            durchlaeufe++;
            int mid = (lo + hi) / 2;
            if (r[mid] < x) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }
        return lo;
    }
}

// Hilfsklasse fuer die Tests: baut Ranglisten und liest sie aus.
class TestHilfe {

    // Baut eine Rangliste mit der Kapazitaet kapazitaet aus den Namen und
    // Punkten (gleiche Position = gleicher Eintrag). Die Spielzeit ist fuer
    // die Tests unwichtig und betraegt einfach 100 + Position.
    public static Rangliste liste(int kapazitaet, String[] namen, int[] punkte) {
        Rangliste r = new Rangliste(kapazitaet);
        for (int i = 0; i < namen.length; i++) {
            r.hinzufuegen(new Eintrag(namen[i], punkte[i], 100 + i));
        }
        return r;
    }

    // Namen in Listenreihenfolge, z. B. "Cyra,Aria,Brann"
    public static String namen(Rangliste r) {
        String s = "";
        for (int i = 0; i < r.getAnzahl(); i++) {
            if (i > 0) {
                s = s + ",";
            }
            s = s + r.getEintrag(i).getName();
        }
        return s;
    }

    // Punkte in Listenreihenfolge, z. B. "1500,1200,800"
    public static String punkte(Rangliste r) {
        String s = "";
        for (int i = 0; i < r.getAnzahl(); i++) {
            if (i > 0) {
                s = s + ",";
            }
            s = s + r.getEintrag(i).getPunkte();
        }
        return s;
    }

    // Die Standardliste der Tests (Kapazitaet 10, also NICHT voll):
    //   Index:   0          1           2           3
    //            Aria 1200  Brann 800   Cyra 1500   Dorn 950
    public static Rangliste standard() {
        String[] namen = {"Aria", "Brann", "Cyra", "Dorn"};
        int[] punkte = {1200, 800, 1500, 950};
        return liste(10, namen, punkte);
    }
}


// ============== Level 2.1 – Standardbausteine (DS 13) =================

class L21StandardTest {

    @Test
    public void basis_hoechstePunktzahl() {
        Rangliste r = TestHilfe.standard();
        assertEquals(1500, r.hoechstePunktzahl(), "Standardliste (1200, 800, 1500, 950): hoechste Punktzahl ist 1500");
    }

    @Test
    public void basis_besterEintrag() {
        Rangliste r = TestHilfe.standard();
        assertTrue(r.besterEintrag() == r.getEintrag(2), "Standardliste: Cyra (1500 Punkte, Index 2) ist der beste Eintrag");
    }

    @Test
    public void basis_anzahlUeber() {
        Rangliste r = TestHilfe.standard();
        assertEquals(3, r.anzahlUeber(900), "Ueber 900 Punkte liegen Aria (1200), Cyra (1500) und Dorn (950): 3 Eintraege");
    }

    @Test
    public void basis_indexVon() {
        Rangliste r = TestHilfe.standard();
        assertEquals(1, r.indexVon("Brann"), "Brann steht an Index 1 - der Index beginnt bei 0");
        assertEquals(0, r.indexVon("Aria"), "Aria steht ganz vorne an Index 0");
    }

    @Test
    public void erweiterung_maximumAmAnfangUndAmEnde() {
        String[] n1 = {"Cyra", "Brann", "Dorn"};
        int[] p1 = {1500, 800, 950};
        Rangliste vorne = TestHilfe.liste(10, n1, p1);
        assertEquals(1500, vorne.hoechstePunktzahl(), "Das Maximum (1500) steht ganz vorne");
        assertTrue(vorne.besterEintrag() == vorne.getEintrag(0), "Der beste Eintrag (Cyra) steht ganz vorne an Index 0");

        String[] n2 = {"Brann", "Dorn", "Elin"};
        int[] p2 = {800, 950, 1600};
        Rangliste hinten = TestHilfe.liste(10, n2, p2);
        assertEquals(1600, hinten.hoechstePunktzahl(), "Das Maximum (1600) steht ganz hinten - laeuft deine Schleife bis anzahl - 1?");
        assertTrue(hinten.besterEintrag() == hinten.getEintrag(2), "Der beste Eintrag (Elin) steht ganz hinten - laeuft deine Schleife bis anzahl - 1?");
    }

    @Test
    public void erweiterung_anzahlUeber_grenzeGenau() {
        Rangliste r = TestHilfe.standard();
        assertEquals(1, r.anzahlUeber(1200), "Ueber 1200 liegt nur Cyra (1500) - 1200 selbst ist NICHT ueber 1200 (> statt >=)");
        assertEquals(0, r.anzahlUeber(1500), "Ueber 1500 liegt niemand");
        assertEquals(4, r.anzahlUeber(0), "Ueber 0 liegen alle 4 Eintraege");
    }

    @Test
    public void erweiterung_indexVon_letzterUndFehlender() {
        Rangliste r = TestHilfe.standard();
        assertEquals(3, r.indexVon("Dorn"), "Dorn steht ganz hinten an Index 3 - wird das letzte Element geprueft?");
        assertEquals(-1, r.indexVon("Zora"), "Zora steht nicht in der Liste: indexVon muss -1 liefern");
    }

    @Test
    public void erweiterung_indexVon_erstesVorkommen() {
        String[] namen = {"Brann", "Aria", "Cyra", "Aria"};
        int[] punkte = {800, 1200, 1500, 700};
        Rangliste r = TestHilfe.liste(10, namen, punkte);
        assertEquals(1, r.indexVon("Aria"), "Aria kommt an Index 1 und 3 vor: gesucht ist das ERSTE Vorkommen (Index 1)");
    }

    @Test
    public void erweiterung_durchschnitt() {
        String[] namen = {"Aria", "Brann", "Cyra", "Dorn"};
        int[] punkte = {100, 200, 250, 125};
        Rangliste r = TestHilfe.liste(10, namen, punkte);
        assertEquals(168.75, r.durchschnitt(), "(100 + 200 + 250 + 125) / 4 = 168.75 - Vorsicht: Ganzzahl / Ganzzahl rechnet ganzzahlig (168)!");
    }

    @Test
    public void erweiterung_leereListe() {
        Rangliste r = new Rangliste(5);

// ================ Level 2.3 – Suchen (DS 16) =========================

class L23SucheTest {

    // Aufsteigend sortiert, n = 9:
    //   Index: 0  1   2   3   4   5   6   7   8
    //   Wert:  3  8  15  23  42  57  61  77  90
    private int[] neun() {
        int[] r = {3, 8, 15, 23, 42, 57, 61, 77, 90};
        return r;
    }

    // Aufsteigend sortiert: 0, 2, 4, ..., 2 * (n - 1)
    private int[] geradeZahlen(int n) {
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            r[i] = 2 * i;
        }
        return r;
    }

    // groesste Durchlaufzahl von binaer ueber alle Werte 0, 1, ..., 2n
    // (also alle vorhandenen UND alle fehlenden Werte)
    private int maxDurchlaeufe(int[] r) {
        int max = 0;
        for (int x = -1; x <= 2 * r.length; x++) {
            Suche.binaer(r, x);
            if (Suche.durchlaeufe > max) {
                max = Suche.durchlaeufe;
            }
        }
        return max;
    }

    @Test
    public void basis_linear() {
        int[] r = {3, 8, 15, 23, 42};
        assertEquals(2, Suche.linear(r, 15), "linear({3, 8, 15, 23, 42}, 15) muss 2 liefern");
        assertEquals(0, Suche.linear(r, 3), "linear({3, 8, 15, 23, 42}, 3) muss 0 liefern");
        assertEquals(-1, Suche.linear(r, 50), "50 kommt nicht vor: linear muss -1 liefern");
    }

    @Test
    public void basis_binaer_gefunden() {
        int[] r = neun();
        assertEquals(4, Suche.binaer(r, 42), "binaer(..., 42) muss 4 liefern - 42 steht genau in der Mitte");
        assertEquals(6, Suche.binaer(r, 61), "binaer(..., 61) muss 6 liefern");
    }

    @Test
    public void basis_binaer_nichtGefunden() {
        int[] r = neun();
        assertEquals(1, Suche.binaer(r, 8), "binaer(..., 8) muss 1 liefern");
        assertEquals(-1, Suche.binaer(r, 50), "50 kommt nicht vor: binaer muss -1 liefern (und terminieren!)");
    }

    @Test
    public void erweiterung_linear_durchlaeufe() {
        int[] r = {3, 8, 15, 23, 42};
        Suche.linear(r, 15);
        assertEquals(3, Suche.durchlaeufe, "linear nach 15: 3, 8, 15 geprueft -> durchlaeufe = 3");
        Suche.linear(r, 50);
        assertEquals(5, Suche.durchlaeufe, "linear nach 50 (fehlt): alle 5 Elemente geprueft -> durchlaeufe = 5 (zu Beginn auf 0 setzen!)");
    }

    @Test
    public void erweiterung_linear_erstesUndLetztes() {
        int[] r = {4, 7, 7, 9};
        assertEquals(1, Suche.linear(r, 7), "7 steht an Index 1 und 2: linear muss das ERSTE Vorkommen liefern");
        assertEquals(3, Suche.linear(r, 9), "9 steht ganz hinten an Index 3 - wird das letzte Element geprueft?");
    }

// =====================================================================
//  HÄUFIGE SCHÜLERFEHLER (Lehrerhinweise) – in Klammern der Test, der
//  den Fehler aufdeckt (diese Zuordnungen sind per Mutationstest geprueft)
// =====================================================================
//
//  Level 2.1 – Standardbausteine
//  - Schleife bis eintraege.length statt bis anzahl -> NullPointer-
//    Exception an den leeren Plaetzen (basis_hoechstePunktzahl u. a.).
//  - Off-by-one: "i <= anzahl" (Indexfehler/null) bzw. "i < anzahl - 1"
//    (letzter Eintrag fehlt: erweiterung_maximumAmAnfangUndAmEnde,
//    erweiterung_indexVon_letzterUndFehlender).
//  - max mit 0 initialisiert (bzw. bester = null und Vergleich mit 0) ->
//    bei negativen Punkten falsch (experte_negativePunkte), bei leerer
//    Liste 0 statt -1 (erweiterung_leereListe). Richtig: mit dem ersten
//    Eintrag starten, Leer-Abfrage davor.
//  - besterEintrag mit ">=" -> bei Gleichstand der LETZTE
//    (experte_besterEintrag_gleichstand).
//  - anzahlUeber mit ">=" (erweiterung_anzahlUeber_grenzeGenau).
//  - indexVon: "else return -1;" in der Schleife -> nur der erste Eintrag
//    wird geprueft; ohne sofortiges return wird das LETZTE Vorkommen
//    geliefert (erweiterung_indexVon_erstesVorkommen); contains statt
//    equals findet "Cyr" (experte_indexVonMitEquals).
//  - == statt equals: In der Online-IDE GRUEN (sie vergleicht mit ==
//    inhaltlich), in echtem Java und im Abitur falsch. Die Tests
//    experte_indexVonMitEquals und experte_hatDoppelteNamen_randfaelle
//    bauen den Namen zur Laufzeit zusammen und wuerden in echtem Java
//    rot - in der Online-IDE also im Code-Review pruefen!
//  - durchschnitt: "summe / anzahl" oder "(double) (summe / anzahl)" ->
//    ganzzahlige Division (168 statt 168.75, erweiterung_durchschnitt);
//    ohne Leer-Abfrage Division durch 0 (erweiterung_leereListe).
//  - hatDoppelteNamen: innere Schleife ab j = i -> jeder Eintrag ist
//    sein eigenes Duplikat, immer true (experte_hatDoppelteNamen_ja);
//    innere Schleife bis anzahl - 1 -> letztes Paar fehlt
//    (experte_hatDoppelteNamen_randfaelle).
//
//  Level 2.2 – Sortieren
//  - istSortiert mit "<=" statt "<" -> gleiche Nachbarn gelten als
//    unsortiert; Schleife bis anzahl - 1 einschliesslich -> Zugriff auf
//    eintraege[anzahl] (erweiterung_istSortiert_randfaelle); istSortiert
//    ueber besser(...) -> zaehlt mit (erweiterung_istSortiert_zaehltNicht).
//  - Sortierrichtung vertauscht (aufsteigend statt absteigend) - alle
//    basis_-Tests zeigen die Reihenfolge, z. B. "erwartet: Cyra,Aria,...".
//  - Selectionsort: innere Schleife ab j = i (zusaetzliche Vergleiche,
//    erweiterung_selectionSort_vergleiche); Tausch IN der inneren
//    Schleife (= kein Selectionsort, zu viele Vertauschungen) oder
//    Tausch auch bei maxIndex == i (experte_selectionSort_vertauschungen).
//  - Insertionsort: Bedingung "besser(...) && j > 0" in falscher
//    Reihenfolge -> besser(eintraege[0], eintraege[-1]) -> Indexfehler.
//    Kein Abbruch der inneren Schleife -> bester Fall 15 statt 5
//    Vergleiche (erweiterung_insertionSort_besterFall). Bedingung
//    "!besser(eintraege[j - 1], eintraege[j])" tauscht gleich gute
//    Eintraege -> nicht stabil (experte_insertionSort_stabil).
//    Vergleich direkt ueber getPunkte() -> Zaehler bleibt 0.
//  - Bubblesort: ohne Abbruch (erweiterung_bubbleSort_besterFall), innere
//    Schleife nicht verkuerzt (experte_bubbleSort_schlechtesterFall),
//    "j < anzahl - durchlauf" -> eintraege[j + 1] ist null; getauscht
//    nie auf true gesetzt -> nur ein Durchlauf.
//  - einfuegenSortiert: Schleife "!besser(eintraege[i - 1], e)" setzt e
//    VOR gleich gute Eintraege (experte_einfuegenSortiert_gleichstand);
//    Voll- oder null-Abfrage fehlt (erweiterung_einfuegenSortiert_
//    nullUndVoll); anzahl++ vergessen; Verschieben mit vertausche
//    (experte_einfuegenSortiert_ohneVertausche); Schieben von vorn nach
//    hinten statt von hinten nach vorn -> Eintraege werden ueberschrieben.
//
//  Level 2.3 – Suchen
//  - durchlaeufe nicht zu Beginn auf 0 gesetzt -> Werte summieren sich
//    (erweiterung_linear_durchlaeufe).
//  - Lineare Suche ohne sofortiges return -> letztes Vorkommen
//    (erweiterung_linear_erstesUndLetztes).
//  - Binaere Suche mit "while (lo < hi)" -> Reihung {7} oder das letzte
//    Element wird nicht gefunden (erweiterung_binaer_raender).
//  - "hi = mid" statt "hi = mid - 1" bei lo <= hi -> ENDLOSSCHLEIFE bei
//    fehlenden Werten: Der Testrunner der Online-IDE haengt dann - Tests
//    einzeln starten lassen und den Trace auf Papier machen!
//  - Fehlende Abbruchbedingung (Treffer nur gemerkt, nicht return) ->
//    zu viele Durchlaeufe (erweiterung_binaer_traceDurchlaeufe).
//  - Vergleich r[mid] < x und r[mid] > x vertauscht -> sucht in der
//    falschen Haelfte (basis_binaer_gefunden).
//  - einfuegeposition: hi = r.length - 1 -> fuer x groesser als alle
//    falsch; "<=" statt "<" -> Position HINTER gleichen Werten
//    (experte_einfuegeposition); lineare statt binaerer Suche
//    (experte_einfuegeposition_leerUndBinaer).
//
//  ABI-BEZUG
//  - Trace Insertionsort (absteigend), 50 80 30 90 70 - Zustand nach
//    jedem aeusseren Durchlauf (Testfall experte_insertionSort_traceZaehlen):
//        i   Reihung              Vergl.  Vert.   (kumuliert)
//        1   80 50 30 90 70         1       1
//        2   80 50 30 90 70         2       1
//        3   90 80 50 30 70         5       4
//        4   90 80 70 50 30         8       6
//  - Dieselbe Eingabe mit Bubblesort (mit Abbruch, Verkuerzung):
//        nach Durchlauf 1: 80 50 90 70 30  (4 Vergl., 3 Vert.)
//        nach Durchlauf 2: 80 90 70 50 30  (7 Vergl., 5 Vert.)
//        nach Durchlauf 3: 90 80 70 50 30  (9 Vergl., 6 Vert.)
//        Durchlauf 4 ohne Tausch -> Abbruch (10 Vergl., 6 Vert.)
//    Mit Selectionsort: 10 Vergleiche (immer n(n-1)/2), 2 Vertauschungen.
//    Merke: Bubble- und Insertionsort tauschen gleich oft (6 = Zahl der
//    Fehlstaende), Selectionsort tauscht hoechstens n - 1 Mal.
//  - Trace Selectionsort (Maximum-Auswahl) auf 10 20 30 40 50 60:
//        i = 0: 60 20 30 40 50 10   i = 1: 60 50 30 40 20 10
//        i = 2: 60 50 40 30 20 10   i = 3, 4: kein Tausch
//        -> 15 Vergleiche, nur 3 Vertauschungen (n/2 bei umgekehrter Liste)
//  - Vergleiche zaehlen (n Eintraege):
//        Selectionsort                immer          n(n-1)/2
//        Insertion-/Bubblesort+       bester Fall    n - 1   (schon sortiert)
//                                     schlechtester  n(n-1)/2 (umgekehrt)
//        n = 6: 15 · n = 40: 780 · n = 100: 4950 · n = 200: 19900
//    Verdopplung: n(n-1)/2 wird etwa VIERmal so gross (4950 -> 19900),
//    der beste Fall n - 1 nur doppelt so gross.
//  - Trace binaere Suche nach 50 in 3 8 15 23 42 57 61 77 90:
//        lo  hi  mid  r[mid]
//        0   8   4    42      42 < 50 -> lo = 5
//        5   8   6    61      61 > 50 -> hi = 5
//        5   5   5    57      57 > 50 -> hi = 4
//        lo > hi -> -1 nach 3 Durchlaeufen
//  - Maximale Durchlaeufe der binaeren Suche: floor(log2 n) + 1, also
//    n = 100 -> 7 (2^6 = 64 <= 100 < 128 = 2^7), n = 200 -> 8.
//    Verdopplung -> nur EIN Durchlauf mehr (lineare Suche: doppelt so
//    viele). Begruendung im Abi: "Jeder Durchlauf halbiert den
//    Suchbereich; nach 7 Halbierungen ist von 100 Elementen hoechstens
//    eines uebrig."
//  - Voraussetzung binaere Suche: Reihung SORTIERT - im Abitur oft als
//    Teilaufgabe "Begruende, warum ... nicht angewendet werden kann".
//  - Papier-Compiler-Kandidaten: anzahlUeber, indexVon, einfuegenSortiert
//    und binaer erst handschriftlich, dann unveraendert abtippen.
//  - Abi-Check: Leer-Abfrage (anzahl == 0) vor dem Zugriff auf
//    eintraege[0]; Schleifen bis anzahl - 1; Zeichenketten mit equals;
//    Liste nach den Standardbausteinen unveraendert
//    (experte_volleListeUnveraendert).
// =====================================================================
